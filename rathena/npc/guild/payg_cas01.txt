// Guild Flags (Outside Castle)
//============================================================
-	script	Bright Arbor#1::OutsideFlagsPA1	722,{

	set .@GID, GetCastleData("payg_cas01",1);

	if (.@GID == 0) {
		mes "[ Édito do Divino Reino de Rune-Midgard ]";
		mes " ";
		mes "1. Conforme a ordenação do Divino Reino de Rune-Midgard, ";
		mes "declaramos que";
		mes "este feudo não possui mestre formal.";
		mes " ";
		mes "2. Àquele que for capaz de ";
		mes "vencer todas as provações";
		mes "e destruir o Emperium,";
		mes "o rei concederá";
		mes "a posse deste feudo.";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ Voz que Ecoa ]";
		mes "Bravos...";
		mes "Desejam retornar ao seu honrado lugar?";
		next;
		if(select("Voltar ao feudo do clã.:Sair.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("payg_cas01",1)) {
					warp "payg_cas01",54,144;
				end;
			}
			end;
		}
		close;
	}
	mes "[ Édito do Divino Reino de Rune-Midgard ]";
	mes " ";
	mes "1. Conforme a ordenação do Divino Reino de Rune-Midgard, ";
	mes "reconhecemos que este lugar está";
	mes "em posse exclusiva do clã ^ff0000"+GetGuildName(.@GID)+"^000000.";
	mes " ";
	mes "2. O líder do clã ^ff0000"+GetGuildName(.@GID)+"^000000 é";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "Se houver quem se oponha a isto,";
	mes "prove sua força e honra com uma lâmina de aço em punho.";
	close;

OnRecvCastlePy01:
	FlagEmblem GetCastleData("payg_cas01",1);
	end;
}
