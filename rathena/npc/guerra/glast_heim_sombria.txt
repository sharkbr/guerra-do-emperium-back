// browiki 1 - a entrada, em Glast Heim. O sósia fica ao lado do
// Hugin da Maldição normal (204,273), que continua onde estava.
glast_01,179,282,5	script	Sósia de Hugin::SombriaHugin	4_M_SAGE_C,{
	.@md$ = "Maldição de Glast Heim Sombria";

	if (BaseLevel < 160) {
		mes "[Sósia de Hugin]";
		mes "Existe uma segunda fenda, mais funda que a do meu original, e ela não perdoa. Volte quando tiver nível 160.";
		close;
	}
	if (isbegin_quest(12321) != 2) {
		mes "[Sósia de Hugin]";
		mes "Você ainda não atravessou a Maldição de Glast Heim. Quem não viu a ruína de perto não tem o que fazer do outro lado.";
		next;
		mes "[Sósia de Hugin]";
		mes "Fale com o meu original, ali adiante. Quando terminar aquilo, volte aqui.";
		close;
	}

	mes "[Sósia de Hugin]";
	mes "Eu não sou ele. Sou o que sobrou de uma travessia que deu errado - o Hugin que ficou do lado de dentro quando a fenda fechou.";
	next;
	mes "[Sósia de Hugin]";
	mes "O que você viu foi a maldição da Himmelmez enfraquecendo. Existe uma versão daquela noite em que ela não enfraqueceu.";
	next;
	mes "[Sósia de Hugin]";
	mes "Os mesmos corredores, a mesma gente morta, e o Amdarais inteiro. Quer ver?";
	next;

	switch (select("Quero entrar.", "Quero gerar a fenda.", "Agora não.")) {
	case 1:
		// Nao ha `has_instance` neste rAthena - so `instance_enter`,
		// e o IE_NOINSTANCE dele ja responde "a fenda nao esta
		// aberta". Uma checagem a menos, e uma verdade a menos para
		// divergir.
		if (isbegin_quest(12334) == 1) {
			mes "[Sósia de Hugin]";
			mes "Você atravessou faz pouco. O corpo leva três dias para aguentar de novo.";
			close;
		}
		switch (instance_enter(.@md$)) {
		case IE_OK:
			if (isbegin_quest(12335) == 0)
				setquest 12335;
			mapannounce "glast_01", strcharinfo(0) + " atravessou para a Glast Heim Sombria.", bc_map, 0x99CCFF;
			close;
		case IE_NOMEMBER:
			mes "[Sósia de Hugin]";
			mes "Você não é do grupo que abriu esta fenda.";
			close;
		case IE_NOINSTANCE:
			mes "[Sósia de Hugin]";
			mes "A fenda não está aberta.";
			close;
		default:
			mes "[Sósia de Hugin]";
			mes "Alguma coisa empurrou você de volta. Tente outra vez.";
			close;
		}

	case 2:
		if (is_party_leader() == false) {
			mes "[Sósia de Hugin]";
			mes "Só quem lidera o grupo consegue rasgar o tempo. Chame o seu líder.";
			close;
		}
		switch (instance_create(.@md$)) {
		case IE_NOMEMBER:
			mes "[Sósia de Hugin]";
			mes "Você precisa de um grupo, mesmo que seja só você nele.";
			close;
		case IE_NOINSTANCE:
			mes "[Sósia de Hugin]";
			mes "Falhei ao abrir a fenda. Tente daqui a pouco.";
			close;
		case IE_OTHER:
			mes "[Sósia de Hugin]";
			mes "Esta fenda já está aberta.";
			close;
		default:
			mes "[Sósia de Hugin]";
			mes "Pronto. Fale comigo de novo para atravessar - e leve gente.";
			close;
		}

	case 3:
		close;
	}
}

// O CONTROLADOR
//
// Guarda o estágio da instância e é quem liga e desliga tudo.
// Mora em 1,1 porque não tem corpo: o `HIDDEN_WARP_NPC` é o
// mesmo que o OldGlastHeim.txt usa para os `#talkinstance`.
//
// `OnInstanceInit` e NÃO `OnInit`: o `disablenpc` do OnInit
// esconde o NPC do mapa-molde, onde ninguém entra, e o clone
// de dentro da instância nasce LIGADO - o `npc_duplicate_sub`
// copia `nd->state`, que o `disablenpc` nunca escreve
// (CLAUDE.md seção 5). É seguro porque o `instance_addnpc`
// cria todos os clones ANTES de rodar qualquer
// `OnInstanceInit` (instance.cpp:586-598).
// ---------------------------------------------------------
1@gl_he,1,1,0	script	#sombria_controle	HIDDEN_WARP_NPC,{
	end;

OnInstanceInit:
	'mapa$[0] = instance_mapname("1@gl_he");
	'mapa$[1] = instance_mapname("1@gl_he2");
	'estagio = 0;

	disablenpc instance_npcname("Henrich#sombria");
	disablenpc instance_npcname("Coroinha de Domun#sombria");
	disablenpc instance_npcname("Ferreira Hollgrehenn#sombria");
	disablenpc instance_npcname("Himmelmez#sombria1");
	disablenpc instance_npcname("Himmelmez#sombria2");
	disablenpc instance_npcname("Hugin#sombria");
	disablenpc instance_npcname("#sombria_portal_oeste");
	disablenpc instance_npcname("#sombria_portal_leste");
	disablenpc instance_npcname("#sombria_portal_salao");
	disablenpc instance_npcname("#sombria_portal_andar2");
	disablenpc instance_npcname("#sombria_estatua_chave");

	for (.@i = 1; .@i <= 6; .@i++)
		disablenpc instance_npcname("Homem caído#sombria" + .@i);
	for (.@i = 1; .@i <= 7; .@i++)
		disablenpc instance_npcname("Estátua Estranha#sombria" + .@i);
	end;

// Cada setor termina do mesmo jeito: quando o último monstro
// com o rótulo cai. Um rótulo por setor, e o `mobcount` conta
// por rótulo - nunca por "all", que devolve 0 e faz o setor
// nunca fechar, calado (CLAUDE.md seção 5).
OnSetorLimpo:
	if (mobcount('mapa$[0], instance_npcname("#sombria_controle") + "::OnSetorLimpo") > 0)
		end;
	donpcevent instance_npcname("#sombria_controle") + "::OnAvanca";
	end;

OnAvanca:
	'estagio++;
	switch ('estagio) {
	// browiki 6 - o Coroinha de Domun, depois do setor 1
	case 3:
		mapannounce 'mapa$[0], "A maldição da Himmelmez enfraqueceu neste trecho. Procurem sobreviventes.", bc_map, 0xFFFF00;
		enablenpc instance_npcname("Coroinha de Domun#sombria");
		break;
	// browiki 8 - a Ferreira Hollgrehenn, depois do setor 2
	case 5:
		mapannounce 'mapa$[0], "O setor oeste está limpo. Alguém chama por socorro no corredor.", bc_map, 0xFFFF00;
		enablenpc instance_npcname("Ferreira Hollgrehenn#sombria");
		break;
	// browiki 10 - a Himmelmez, depois do setor 3
	case 7:
		mapannounce 'mapa$[0], "O setor leste está limpo. O portal do salão se abriu.", bc_map, 0xFFFF00;
		enablenpc instance_npcname("#sombria_portal_salao");
		enablenpc instance_npcname("Himmelmez#sombria1");
		break;
	}
	end;

// browiki 5 - setor 1, no corredor central.
OnSetor1:
	.@rot$ = instance_npcname("#sombria_controle") + "::OnSetorLimpo";
	areamonster 'mapa$[0],149,52,20,20,"--ja--",3139,8,.@rot$;
	areamonster 'mapa$[0],149,52,20,20,"--ja--",3141,6,.@rot$;
	areamonster 'mapa$[0],149,52,20,20,"--ja--",3140,4,.@rot$;
	mapannounce 'mapa$[0], "Os mortos do corredor se levantaram.", bc_map, 0xFF6666;
	end;

// browiki 7 - setor 2, ala oeste. Os Homens caídos ficam de pé
// junto com os monstros: quem chegar perto deles leva Arclouses.
OnSetor2:
	.@rot$ = instance_npcname("#sombria_controle") + "::OnSetorLimpo";
	areamonster 'mapa$[0],129,104,35,35,"--ja--",3139,12,.@rot$;
	areamonster 'mapa$[0],129,104,35,35,"--ja--",3141,10,.@rot$;
	for (.@i = 1; .@i <= 6; .@i++)
		enablenpc instance_npcname("Homem caído#sombria" + .@i);
	mapannounce 'mapa$[0], "Cuidado com os Homens caídos: eles não vieram sozinhos.", bc_map, 0xFF6666;
	end;

// browiki 9 - setor 3, ala leste.
OnSetor3:
	.@rot$ = instance_npcname("#sombria_controle") + "::OnSetorLimpo";
	areamonster 'mapa$[0],191,107,35,35,"--ja--",3143,10,.@rot$;
	areamonster 'mapa$[0],191,107,35,35,"--ja--",3144,8,.@rot$;
	mapannounce 'mapa$[0], "A guarda do reino ainda patrulha o setor leste.", bc_map, 0xFF6666;
	end;

// browiki 13 - o andar 2. Aqui os monstros não abrem portal:
// eles só precisam ser derrubados até a mensagem aparecer, que
// é o que a browiki descreve ("as chances da mensagem aparecer
// são aleatórias, é importante continuar matando").
OnAndar2:
	.@rot$ = instance_npcname("#sombria_controle") + "::OnAndar2Morto";
	areamonster 'mapa$[1],129,104,35,35,"--ja--",3146,8,.@rot$;
	areamonster 'mapa$[1],129,104,35,35,"--ja--",3145,6,.@rot$;
	areamonster 'mapa$[1],191,107,35,35,"--ja--",3147,6,.@rot$;
	areamonster 'mapa$[1],191,107,35,35,"--ja--",3143,8,.@rot$;
	areamonster 'mapa$[1],191,107,35,35,"--ja--",3144,6,.@rot$;
	'mortos = 0;
	end;

OnAndar2Morto:
	'mortos++;
	// 12 mortes garantem, e antes disso é sorte. O `rand` nunca
	// recebe menos de 2 - `rand(1)` MATA o script (seção 5).
	if ('mortos < 12) {
		if (rand(6) != 0)
			end;
	}
	if ('estagio >= 11)
		end;
	'estagio = 11;
	mapannounce 'mapa$[1], "Um dos Comandantes da Destruição se revelou na ala oeste.", bc_map, 0xFFFF00;
	donpcevent instance_npcname("#sombria_controle") + "::OnDesmorto";
	end;

// browiki 14 - o Desmorto Sombrio, num de quatro pontos.
OnDesmorto:
	setarray .@x[0], 118, 141, 100, 129, -1;
	setarray .@y[0], 118,  90, 130, 104, -1;
	.@n = 4;
	.@i = rand(.@n);
	monster 'mapa$[1],.@x[.@i],.@y[.@i],"--ja--",3148,1,
		instance_npcname("#sombria_controle") + "::OnDesmortoMorto";
	end;

OnDesmortoMorto:
	'estagio = 12;
	mapannounce 'mapa$[1], "O Comandante caiu. A irmã dele acordou na ala leste.", bc_map, 0xFFFF00;
	setarray .@x[0], 191, 205, 178, 196, -1;
	setarray .@y[0], 107,  95, 120, 130, -1;
	.@n = 4;
	.@i = rand(.@n);
	monster 'mapa$[1],.@x[.@i],.@y[.@i],"--ja--",3149,1,
		instance_npcname("#sombria_controle") + "::OnDesmortaMorta";
	end;

// browiki 16 - a Himmelmez de novo, e o Amdarais Sombrio.
OnDesmortaMorta:
	'estagio = 13;
	mapannounce 'mapa$[1], "O caminho para o norte se abriu. A Himmelmez espera.", bc_map, 0xFFFF00;
	enablenpc instance_npcname("#sombria_portal_andar2");
	enablenpc instance_npcname("Himmelmez#sombria2");
	end;

// O Amdarais chama Arclouses o tempo todo, e é isso que a
// browiki avisa: "esta versão do MVP invocará Arclouses
// Sombrios". Um temporizador do controlador, e não uma linha
// de mob_skill_db, porque a invocação é do encontro e não do
// bicho - fora daqui ele não deve chamar nada.
OnAmdaraisComeca:
	initnpctimer;
	end;

OnTimer25000:
	if (mobcount('mapa$[1], instance_npcname("#sombria_controle") + "::OnAmdaraisMorto") == 0) {
		stopnpctimer;
		end;
	}
	areamonster 'mapa$[1],148,172,12,12,"--ja--",3142,4,
		instance_npcname("#sombria_controle") + "::OnArclouseMorto";
	initnpctimer;
	end;

OnArclouseMorto:
	end;

OnAmdaraisMorto:
	stopnpctimer;
	'estagio = 14;
	killmonster 'mapa$[1], instance_npcname("#sombria_controle") + "::OnArclouseMorto";
	mapannounce 'mapa$[1], "O Amdarais Sombrio caiu. Falem com Hugin.", bc_map, 0x99FF99;
	enablenpc instance_npcname("Hugin#sombria");
	enablenpc instance_npcname("#sombria_estatua_chave");
	end;
}
