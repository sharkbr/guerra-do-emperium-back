//===== Guerra do Emperium ==================================================
//= Bandeiras do feudo - todas com o emblema de quem tomou o Kriemhild
//===========================================================================
//= 279 bandeiras em 27 mapas, ARRANCADAS SEM QUERER em 2026-08-13 e
//= devolvidas no mesmo dia. Ver "o que aconteceu" logo abaixo.
//===========================================================================
//
//  O QUE ACONTECEU
//  ===============
//
//  Desligar os dezenove castelos em npc/scripts_guild.conf tirou muito mais
//  que o Emperium: cada arquivo de castelo do rAthena define TAMBEM as
//  bandeiras dele - quatro do lado de fora, no feudo; doze dentro do castelo;
//  e uma na cidade correspondente. Dezenove arquivos comentados levaram 279
//  bandeiras junto, e o dono percebeu na tela antes de qualquer log acusar.
//
//  A pergunta que ele fez e a que este arquivo responde: "elas eram atreladas
//  ao castelo NECESSARIAMENTE?". Nao. O que prende uma bandeira a um castelo e
//  UMA LINHA - o `FlagEmblem GetCastleData("<mapa>", CD_GUILD_ID)` do rotulo
//  de atualizacao. O resto e um NPC de sprite 722 parado numa celula. Trocar o
//  mapa dentro daquele `GetCastleData` e tudo que e preciso para a bandeira
//  passar a hastear outro cla.
//
//  E e o que se faz aqui: as 279 leem `prtg_cas01`, o Kriemhild. Quem toma o
//  Kriemhild passa a ter o emblema em Al De Baran, Geffen, Payon, Prontera e
//  nos quatro feudos, de uma vez. Decisao do dono em 2026-08-13.
//
//  AS POSICOES NAO FORAM DIGITADAS
//  ===============================
//
//  Sao extraidas dos dezenove arquivos de castelo por
//  ferramentas/../gera_bandeiras.py (scratchpad), que le as linhas
//  `duplicate(OutsideFlags*)` e `duplicate(InsideFlags*)` de cada um. As
//  bandeiras voltam exatamente onde a Gravity as pos, e se o upstream mover
//  alguma um dia, a diferenca aparece ao regerar.
//
//  Uma armadilha ficou registrada dessa extracao: NOME DE NPC PODE TER ESPACO.
//  Os cinco castelos de Payon chamam as bandeiras de `Bright Arbor#1-2`, com
//  espaco no meio, e um `\S+` no lugar do nome perde Payon INTEIRO - devolve
//  219 bandeiras em vez de 279, sem erro, e o numero e plausivel demais para
//  levantar suspeita. Corta-se por TAB, que e o separador de verdade.
//
//  QUEM MANDA ATUALIZAR
//  ====================
//
//  Tres gatilhos, e cada um cobre um buraco do outro:
//
//  1. `OnAgitInit` - a subida do servidor, quando o char-server entrega os
//     dados de castelo.
//  2. `OnRecvCastlePt01` - o evento que o proprio agit_main.txt dispara para
//     as bandeiras do Kriemhild quando ele troca de dono. E o mesmo `donpcevent
//     "::OnRecvCastlePt01"` de sempre, e as nossas so precisam ter o rotulo
//     para entrar na lista.
//  3. Um temporizador de dez segundos no `ControleDasBandeiras` la embaixo,
//     que so dispara quando o dono MUDOU.
//
//  O terceiro nao e paranoia: o `OnRecvCastlePt01` do rAthena so e chamado no
//  ramo do castelo COM DONO (npc/guild/agit_main.txt, o `else` do
//  `OnRecvCastle`). Quando o cla dono se desfaz, o `OnGuildBreak` poe o
//  CD_GUILD_ID em 0 e chama o `OnRecvCastle`, que dessa vez cai no ramo do
//  castelo vazio e NAO avisa bandeira nenhuma. No rAthena puro as bandeiras
//  ficam com o emblema do cla morto ate o servidor reiniciar; aqui elas se
//  apagam em dez segundos.
//
//  O temporizador e barato de proposito: ele so LE `GetCastleData`, que e
//  memoria do map-server, e compara com um inteiro. So quando o numero muda e
//  que ele gasta um `donpcevent` nas 279.
//
//  SE UM CASTELO VOLTAR A GUERRA
//  =============================
//
//  Descomentar a linha dele em npc/scripts_guild.conf devolve as bandeiras
//  ORIGINAIS dele, que leem o proprio castelo - e ai as duas ficam empilhadas
//  na mesma celula. Ao religar um castelo, tirar as linhas dele daqui. O nome
//  unico das nossas termina em `#feudo<N>` justamente para nao colidir com o
//  das originais: sem isso o `npc_parsename` renomearia uma das duas e o
//  empilhamento seria mais dificil de ver.
//===========================================================================

// A bandeira. O dialogo conta a mesma coisa que o edito do rAthena, so que
// sobre o Kriemhild - nao ha mais "este feudo" no plural, ha um so.
//
// O que NAO tem, de proposito: o atalho de voltar ao castelo que as bandeiras
// de fora do rAthena oferecem ao membro do cla dono. Aquilo e uma porta de
// entrada, e 279 portas para dentro do Kriemhild seria outra coisa. Quem quer
// o atalho usa as bandeiras do proprio Kriemhild, que continuam de pe.
-	script	Bandeira do Feudo::BandeiraDoFeudo	722,{
	.@dono = GetCastleData("prtg_cas01", CD_GUILD_ID);

	if (.@dono == 0) {
		mes "[ Édito do Divino Reino de Rune-Midgard ]";
		mes " ";
		mes "1. Conforme a ordenação do Divino Reino de Rune-Midgard, declaramos que o feudo de Prontera não possui mestre formal.";
		mes " ";
		mes "2. Àquele que for capaz de vencer todas as provações e destruir o Emperium do Kriemhild, o rei concederá a posse destas terras - e o direito de hastear seu estandarte em todas elas.";
		close;
	}

	mes "[ Édito do Divino Reino de Rune-Midgard ]";
	mes " ";
	mes "1. Conforme a ordenação do Divino Reino de Rune-Midgard, reconhecemos que o Kriemhild está em posse exclusiva do clã ^ff0000" + GetGuildName(.@dono) + "^000000, e que este estandarte é o dele.";
	mes " ";
	mes "2. O líder do clã ^ff0000" + GetGuildName(.@dono) + "^000000 é ^ff0000" + GetGuildMaster(.@dono) + "^000000.";
	mes " ";
	mes "3. Se houver quem se oponha a isto, prove sua força e honra com uma lâmina de aço em punho, na Guerra do Emperium.";
	close;

// `OnRecvCastlePt01` e o evento do proprio rAthena - basta ter o rotulo para
// entrar na lista dele. `OnBandeiraAtualiza` e o nosso, do temporizador.
OnAgitInit:
OnRecvCastlePt01:
OnBandeiraAtualiza:
	FlagEmblem GetCastleData("prtg_cas01", CD_GUILD_ID);
	end;
}

// Guarda o dono conhecido e so acorda as 279 quando ele muda - ver o
// cabecalho, "quem manda atualizar".
-	script	ControleDasBandeiras	-1,{
	end;

OnInit:
	// -1 e nao 0: `0` e um valor legitimo (castelo sem dono), entao comecar
	// nele faria a primeira conferencia achar que nada mudou e deixar as
	// bandeiras sem emblema ate a primeira conquista.
	.dono = -1;
	initnpctimer;
	end;

OnAgitInit:
OnTimer10000:
	.@dono = GetCastleData("prtg_cas01", CD_GUILD_ID);
	if (.@dono != .dono) {
		.dono = .@dono;
		donpcevent "::OnBandeiraAtualiza";
	}
	setnpctimer 0;
	end;
}
