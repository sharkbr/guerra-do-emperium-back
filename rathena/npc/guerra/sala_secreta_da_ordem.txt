//===== A Roleta Magica ======================================
//= prt_in 136,114, sprite 506 (4_VENDING_MACHINE), virada para
//= oeste. Custa 10 MOEDAS NOVAS o giro e devolve UM de cinco
//= itens, com peso:
//=
//=     600  Album de Cartas de Taro          20/10000   0,20%
//=   12658  Disfarce de Deviruchi          2495/10000  24,95%
//=   12663  Disfarce de Poring             2495/10000  24,95%

//= e o cabecalho de la explica cada uma dessas escolhas por
//= extenso. O que muda aqui e o anuncio.
//===== O ANUNCIO, e por que ele tem coluna propria ==========
//= So o Album de Cartas de Taro anuncia para o servidor inteiro.
//= Seria facil escrever `if (.@premio == 600) announce ...`, e
//= seria a divergencia da regra 11 do CLAUDE.md nascendo: o dia em
//= que o premio raro mudar de ID, a tabela muda e o `if` fica
//= apontando para o item velho, calado.
//=
//= Entao o anuncio e a QUARTA COLUNA da mesma tabela.
//=
//= ELA E NUMERADA A PARTIR DE 1, e isso nao e estetica: em array
//= de inteiro do rAthena, gravar 0 REMOVE o elemento, e o
//= `getarraysize` passaria a devolver um tamanho menor que o das
//= outras tres colunas - a conferencia que existe para pegar
//= desalinhamento passaria a inventar um. Por isso 1 = calado e
//= 2 = anuncia (CLAUDE.md secao 5).
//===== Por que checkweight ANTES de cobrar ==================
//= `getitem` com a mochila cheia LARGA O ITEM NO CHAO - o
//= buildin cai num `map_addflooritem` quando o pc_additem falha
//= (CLAUDE.md secao 5). Quem segura a queda e o `NoDrop` do item,
//= e AQUI UM DOS CINCO NAO O TEM:
//=
//=     600  Album de Cartas de Taro      SEM TRAVA - CAI
//=   12658  Disfarce de Deviruchi        NoDrop    - segura
//=   12663  Disfarce de Poring           NoDrop    - segura
//=   12660  Disfarce de Mavka            NoDrop    - segura
//=   22753  Disfarce de Kobold Arqueiro  SEM TRAVA - CAI
//=
//= DOIS DOS CINCO CAEM, e um deles e justamente o premio de 0,2%.
//= O Album foi criado SEM trava de troca porque a descricao do bRO
//= nao traz a linha "Intransferivel" que a Moeda Nova e a Caveira
//= Humana trazem - por-la seria inventar, contra a regra 4.3.
//=
//= Entao o que impede a queda e o `checkweight` na frente do
//= `delitem`, e SO ELE. Quem um dia puser `NoDrop` no 600 esta
//= trocando a fonte (bRO) por gosto; a decisao e legitima, mas e
//= decisao, e o cabecalho do item em db/guerra/item_db.yml repete
//= este mesmo aviso do outro lado.
//=
//= E DAI O SORTEIO VIR ANTES DA COBRANCA: o checkweight precisa
//= saber QUAL item vai entregar. Os cinco sao empilhaveis, mas
//= empilhavel so dispensa slot livre se o jogador JA tem aquele
//= item - checar um e entregar outro passa no teste e larga no
//= chao assim mesmo. A ordem e: sorteia, confere o premio

//===== ESPACO NO INICIO DO `mes` COLA A LINHA NA ANTERIOR ===

// ---------------------------------------------------------------
// A Roleta Magica - sorteia, e por isso e script puro e nao loja
// ---------------------------------------------------------------
prt_in,136,114,2	script	Roleta Mágica	506,{
	// Guarda da tabela: com as colunas fora de sincronia o OnInit
	// deixa .total em 0, e sortear com ela seria sortear fora da
	// tabela. Ver a secao "A tabela sai de UM lugar so".
	if (!.total) {
		mes "[Roleta Mágica]";
		mes "O mostrador pisca em vermelho.";
		mes "A máquina está fora de serviço.";
		close;
	}

	mes "[Roleta Mágica]";
	mes "Um tambor de vidro cheio de fichas gira devagar";
	mes "atrás do painel, sem começo e sem fim.";
	mes "A fenda pede ^0000FF" + .preco + " Moedas Novas^000000.";
	next;

	// A lista e a porcentagem saem dos MESMOS arrays do sorteio,
	// por laco - nenhum numero destes esta escrito no arquivo.
	// CLAUDE.md secao 4.11.
	mes "[Roleta Mágica]";
	mes "O mostrador lista o que a gaveta solta:";
	for (.@i = 0; .@i < getarraysize(.premio); .@i++) {
		// Duas casas decimais com aritmetica inteira: o +.total/2
		// arredonda em vez de truncar, e o zero a esquerda entra a
		// mao porque nao ha formatacao de casa decimal no script.
		.@pct = (.peso[.@i] * 10000 + .total / 2) / .total;
		.@dec = .@pct % 100;
		mes "^0000FF" + getitemname(.premio[.@i]) + "^000000  " +
		    (.@pct / 100) + "," +
		    ((.@dec < 10) ? "0" + .@dec : "" + .@dec) + "%";
	}
	next;

	.@saldo = countitem(.moeda);
	mes "[Roleta Mágica]";
	mes "Você tem ^0000FF" + .@saldo + "^000000 " +
	    ((.@saldo == 1) ? "Moeda Nova" : "Moedas Novas") + ".";
	if (.@saldo < .preco) {
		mes "Não dá para um giro.";
		mes "O tambor continua girando sozinho.";
		close;
	}
	next;

	if (select("Girar por " + .preco + " Moedas Novas",
	           "Deixa para depois") == 2) {
		mes "[Roleta Mágica]";
		mes "O tambor desacelera até parar.";
		close;
	}

	// SORTEIO ANTES DA COBRANCA, e de proposito - o checkweight
	// abaixo precisa saber qual item vai entregar. Ver a secao
	// "Por que checkweight ANTES de cobrar".
	//
	// rand(.total) devolve 0..total-1; o laco anda pelos pesos
	// acumulados ate passar do sorteado.
	.@sorte = rand(.total);
	.@i = 0;
	.@ate = .peso[0];
	while (.@sorte >= .@ate) {
		.@i++;
		.@ate += .peso[.@i];
	}
	.@premio = .premio[.@i];

	if (!checkweight(.@premio, 1)) {
		mes "[Roleta Mágica]";
		mes "A gaveta tenta abrir e trava.";
		mes "Não há espaço na sua mochila,";
		mes "e a máquina não cobrou nada.";
		close;
	}

	// Segunda conferencia de saldo, colada no delitem: `delitem`
	// sem saldo nao falha limpo, aborta o script com erro no
	// console. Entre a primeira e esta ha um `select`.
	if (countitem(.moeda) < .preco) {
		mes "[Roleta Mágica]";
		mes "A fenda cospe de volta o que sobrou.";
		mes "Faltou ^0000FFMoeda Nova^000000.";
		close;
	}

	delitem .moeda, .preco;
	getitem .@premio, 1;

	// O anuncio sai da QUARTA COLUNA da tabela, nunca de um `if`
	// contra o ID - ver a secao "O ANUNCIO". 2 = anuncia.
	if (.anuncia[.@i] == 2)
		announce strcharinfo(0) + " recebeu " + getitemname(.@premio) +
		         " na Sala Secreta da Ordem!", bc_all;

	mes "[Roleta Mágica]";
	mes "O tambor trava, e uma ficha cai na gaveta.";
	mes "Você recebeu ^0000FF" + getitemname(.@premio) + "^000000.";
	close;

OnInit:
	.moeda = 30998;   // Moeda Nova, db/guerra/item_db.yml
	.preco = 10;

	// A TABELA. As QUATRO colunas andam juntas, e sao a unica fonte
	// da lista na tela, da porcentagem, do sorteio e do anuncio.
	// Premio novo e uma coluna em cada uma das quatro.
	//
	// O peso esta em decimos de milesimo (soma 10000) para o 0,2%
	// do Album caber exato. A coluna de anuncio e 1 = calado,
	// 2 = anuncia - numerada a partir de 1 de proposito, ver a
	// secao "O ANUNCIO" no cabecalho.
	//
	//                       600   12658  12663  12660  22753
	setarray .premio[0],     600,  12658, 12663, 12660, 22753;
	setarray .peso[0],        20,   2495,  2495,  2495,  2495;
	setarray .anuncia[0],      2,      1,     1,     1,     1;

	.@n = getarraysize(.premio);
	if (.@n != getarraysize(.peso))    .@n = 0;
	if (.@n != getarraysize(.anuncia)) .@n = 0;
	if (.@n == 0) {
		// .total fica 0 e o corpo recusa a abrir. Melhor a maquina
		// calada que a maquina sorteando fora da tabela.
		debugmes "Roleta Magica: .premio (" + getarraysize(.premio) +
		         "), .peso (" + getarraysize(.peso) +
		         ") e .anuncia (" + getarraysize(.anuncia) +
		         ") com tamanhos diferentes - SORTEIO DESALINHADO.";
		end;
	}

	.total = 0;
	for (.@i = 0; .@i < .@n; .@i++)
		.total += .peso[.@i];
	end;
}

prt_in,137,108,2	script	Máquina Especial	564,{
	mes "[Máquina Especial]";
	mes "Uma máquina baixa, de porta de vidro embaçado.";
	mes "Atrás dela, prateleiras de comida quente";
	mes "que ninguém sabe dizer quem repõe.";
	next;

	.@saldo = countitem(.moeda);
	mes "[Máquina Especial]";
	if (.@saldo == 0) {
		mes "Você não tem nenhuma ^0000FFMoeda Nova^000000.";
		mes "A vitrine acende assim mesmo,";
		mes "para você saber para o que juntar.";
	} else {
		mes "Você tem ^0000FF" + .@saldo + "^000000 " +
		    ((.@saldo == 1) ? "Moeda Nova" : "Moedas Novas") + ".";
		mes "A vitrine acende.";
	}

	// `close2` e nao `close`: fecha a caixa de dialogo mas mantem o
	// jogador preso ao NPC, que e o que permite a janela de troca
	// abrir em seguida. Com `close` a loja nao abriria.
	close2;
	callshop "MaquinaEspecial#loja";
	end;

OnInit:
	// A conta de saldo aqui e INFORMATIVA, nao e trava: mesmo com
	// zero moedas a loja abre. Quem recusa a compra sem saldo e a
	// propria janela de troca.
	.moeda = 30998;   // Moeda Nova, db/guerra/item_db.yml
	end;
}
