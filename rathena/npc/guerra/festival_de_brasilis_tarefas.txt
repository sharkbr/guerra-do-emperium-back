//===== Guerra do Emperium ===================================
//= Festival de Brasilis - as cinco tarefas
//===== Onde ================================================
//= Costureira          brasilis 142,258   Criar Fantasias
//= Pai Desesperado     brasilis 291,203   Moleque perdido
//= Guarda Salva-Vidas  brasilis 263,56    Invasao de Tubaroes
//= Tereza Limpeza      brasilis 243,197   Limpando a cidade
//= Montoeira de Sujeira brasilis 241,197  (da tarefa da Tereza)
//= Chef                brasilis 343,243   Estoque de Comida
//= Crianca             brasilis 207,216 / 178,229 / 202,270
//===========================================================
//= ESTE ARQUIVO E A FONTE PRINCIPAL DE INSIGNIA, e sem ele o
//= Festival nao tem economia: a Maquina cobra de 5 a 50 por
//= peca, e o drop por morte e de 5%. O par deste arquivo e o
//= `festival_de_brasilis.txt`, que tem os Agentes, a Maquina, a
//= Mae de Santo e a TABELA - a insignia de cada Liga sai de la,
//= por `getvariableofnpc(...,"FestivalMae")`.
//=
//= Os dois sao uma peca so: desligar um sem o outro deixa ou o
//= evento sem economia, ou tarefas que pagam numa moeda que nao
//= tem onde ser gasta.
//===========================================================
//= AS COORDENADAS SAO AS DO bRO, e isso e sorte, nao projeto.
//= A tabela do bROWiki da as cinco tarefas em `brz_n`, o mapa
//= exclusivo do Festival que nao temos e nao vamos ter - tudo
//= aqui acontece na propria Brasilis (decisao do dono). As oito
//= coordenadas de la (cinco NPCs mais os tres esconderijos da
//= crianca) foram testadas uma a uma no `map_cache.dat` e as
//= OITO caem em chao andavel na Brasilis, sem NPC do vendor por
//= cima. Entao ficaram como estao: quem conhece o evento do bRO
//= procura no lugar certo.
//=
//= A unica coordenada inventada e a da Montoeira de Sujeira
//= (241,197), que o bROWiki nao da - fica coladinha na Tereza.
//= As tres celulas ao lado (247,197 / 240,200 / 246,194) sao
//= bloqueadas; esta nao.
//===========================================================
//= DUAS ADAPTACOES, e as duas sao visiveis em jogo
//=
//= 1. OS SHARKIRAS VIRARAM CARRAPATOS DE TENTACULO, e a caca
//=    saiu das cavernas para a PRAIA DA PROPRIA CIDADE. O bicho
//=    da tarefa original nao existe no nosso mob_db - e mob
//=    exclusivo do bRO. O substituto e o Id 2081, que o vendor
//=    ja nasce na orla de brasilis, de oeste a leste, e que
//=    parece uma copia de Hidra. O Guarda DIZ que nao sao os
//=    Sharkiras de antes, a pedido do dono.
//=
//=    O 2081 SE CHAMAVA "Strange Hydra" EM JOGO, e nao pelo
//=    mob_db: com `override_mob_names: 0` quem nomeia o monstro
//=    na tela e a terceira coluna da linha de spawn
//=    (npc.cpp:5371). As quatro linhas do vendor traziam o nome
//=    em ingles escrito a mao e venciam a traducao. Por isso
//=    elas estao comentadas em npc/re/mobs/towns.txt e as
//=    nossas, no fim deste arquivo, spawnam com `--ja--` - que
//=    manda usar o JapaneseName do mob_db, onde ele e "Carrapato
//=    de Tentaculo" (db/guerra/mob_db_guerra.yml). Nome em um
//=    lugar so.
//=
//=    Mesma quantidade (18), mesmas coordenadas e mesmo respawn
//=    (5 segundos, acima do `mob_respawn_time: 1000`, entao nao
//=    e truncado) do vendor - so o nome mudou.
//=
//= 2. A MASCARA BRANCA SAIU de Criar Fantasias. Ela e um dos
//=    cinco materiais opcionais do bRO e nao existe com esse
//=    nome no nosso item_db nem no cliente. Como o jogador
//=    escolhe DOIS dos cinco, sobraram quatro opcoes e a tarefa
//=    continua inteira - inventar um item seria pior.
//===========================================================
//= OS SPRITES foram escolhidos pelo dono e CONFERIDOS um a um
//= antes de entrar: Costureira 4_F_BRZ_WOMAN, Pai Desesperado
//= 4_M_BARYO_MAN, Guarda Salva-Vidas 4_M_LAZY, Tereza
//= 4_F_HUWOMAN, Chef 4_M_BRZ_MAN1, Crianca 4_M_KID1 e a
//= Montoeira 4_SOIL.
//=
//= CONFERIR NAO E CAPRICHO: view id de NPC resolve para
//= `data\sprite\npc\<constante em minusculas>.spr`, e ter o
//= sprite de MONSTRO de mesmo nome nao adianta - e outra pasta
//= e outro nome. Faltando o arquivo, o cliente abre caixa de
//= erro ao desenhar. Foi o que aconteceu com os quatro Agentes
//= em 2026-09-06 (CLAUDE.md secao 5). Os sete daqui estao todos
//= no nosso GRF, entao nenhum precisou de patch.
//=
//= Do lado do servidor os sete existem como `JT_*` em
//= src/map/npc.hpp e sao exportados por script_constants.hpp -
//= e por isso da para escrever o NOME no script em vez do
//= numero. Constante que nao esteja la nao compila no parser e
//= o arquivo INTEIRO morre (CLAUDE.md secao 5).
//===========================================================
//= O DESCANSO DE 3 HORAS mora em `festival_tarefa[0..4]`, uma
//= variavel permanente de personagem, e o que se guarda ali e o
//= INSTANTE em que a tarefa libera (`gettimetick(2)` + 10800),
//= nao o instante em que foi feita. Assim a conta que o NPC faz
//= na tela e uma subtracao so.
//=
//= CUIDADO AO MEXER: zero numa posicao de array APAGA a entrada
//= (CLAUDE.md secao 5). Aqui isso e inofensivo porque so se
//= grava tempo futuro, que nunca e zero - mas nao transformar
//= isto num contador que decresce.
//===========================================================
//= AS DUAS FUNCOES sao o que impede as cinco tarefas de
//= divergirem. `FestivalCheca` recusa quem nao tem Liga, quem
//= esta no descanso e quem esta sem espaco na bolsa - e recusa
//= ANTES de qualquer `delitem`, que e o que garante que nenhuma
//= tarefa cobra sem pagar. `FestivalPaga` da a insignia da Liga
//= do jogador e liga o descanso.
//=
//= As duas fecham o dialogo por conta propria quando recusam
//= (`close` dentro de `callfunc` encerra o script inteiro),
//= entao quem chama nao precisa testar retorno nenhum.
//===========================================================
//= O CONTADOR DE PIRANHAS e um `OnNPCKillEvent` SEPARADO do que
//= ja existe no festival_de_brasilis.txt, e os dois rodam: o
//= `npc_script_event` percorre TODOS os NPCs registrados com
//= aquele label (src/map/npc.cpp), nao so o primeiro. Um cuida
//= da insignia por morte, o outro da tarefa - misturar os dois
//= faria a tarefa depender da chance de drop.
//===========================================================
//= Recarregar: @reloadscript.
//===========================================================

// ---------------------------------------------------------------
// As duas funcoes compartilhadas.
// ---------------------------------------------------------------
function	script	FestivalCheca	{
	.@i = getarg(0);

	if (festival_liga < 1) {
		mes "Isso é coisa do Festival, e o Festival é das Ligas. Jure a uma delas com um dos Agentes, lá na beirada norte da praça, e depois volte aqui.";
		close;
	}

	.@falta = festival_tarefa[.@i] - gettimetick(2);
	if (.@falta > 0) {
		.@h = .@falta / 3600;
		.@m = (.@falta % 3600) / 60;
		if (.@h > 0)
			mes "Você já me ajudou faz pouco. Volte daqui a " + .@h + "h" + .@m + "min.";
		else
			mes "Você já me ajudou faz pouco. Volte daqui a " + .@m + " minutos.";
		close;
	}

	// O espaco e conferido AQUI, antes de qualquer delitem: tarefa
	// que cobra o material e nao consegue entregar a insignia seria
	// roubo. Dois e o maximo que uma tarefa paga.
	.@ins = getvariableofnpc(.Insignia[festival_liga - 1], "FestivalMae");
	if (!checkweight(.@ins, 2)) {
		mes "Sua bolsa está tão cheia que não cabe nem uma insígnia. Faça espaço e volte.";
		close;
	}
	return;
}

function	script	FestivalPaga	{
	.@i = getarg(0);
	.@n = getarg(1);
	.@ins = getvariableofnpc(.Insignia[festival_liga - 1], "FestivalMae");
	getitem .@ins, .@n;
	festival_tarefa[.@i] = gettimetick(2) + 10800;   // 3 horas
	return;
}

// ---------------------------------------------------------------
// 1. CRIAR FANTASIAS - Costureira
//
// Dois materiais de uma lista de quatro, dez de cada. O menu sai
// do MESMO array de ids que o `delitem` usa, e o nome de cada
// linha vem de `getitemname()` - assim nao existe segunda lista
// de nomes para divergir da primeira (CLAUDE.md 4.11).
// ---------------------------------------------------------------
brasilis,142,258,4	script	Costureira::FestivalCostureira	4_F_BRZ_WOMAN,{
	mes "[Costureira]";
	callfunc "FestivalCheca", 0;

	mes "As fantasias do Festival não se costuram sozinhas, e o pano acaba antes da festa. Me traga ^0000FFdez unidades de dois materiais^000000 quaisquer da minha lista e eu te pago com uma insígnia da sua Liga.";
	next;

	setarray .@id[0], 7030, 7196, 7219, 7267;
	.@quantos = getarraysize(.@id);

	mes "[Costureira]";
	mes "Escolha o primeiro.";
	next;
	.@lista$ = "";
	for (.@i = 0; .@i < .@quantos; .@i++) {
		if (.@i > 0)
			.@lista$ = .@lista$ + ":";
		.@lista$ = .@lista$ + getitemname(.@id[.@i]) + " (" + countitem(.@id[.@i]) + "/10)";
	}
	.@a = select(.@lista$) - 1;

	// A segunda lista pula o que ja foi escolhido. O indice
	// original vai guardado +1 de proposito: zero numa posicao de
	// array apaga a entrada (CLAUDE.md secao 5).
	.@n2 = 0;
	.@lista2$ = "";
	for (.@i = 0; .@i < .@quantos; .@i++) {
		if (.@i == .@a)
			continue;
		if (.@n2 > 0)
			.@lista2$ = .@lista2$ + ":";
		.@lista2$ = .@lista2$ + getitemname(.@id[.@i]) + " (" + countitem(.@id[.@i]) + "/10)";
		.@orig[.@n2] = .@i + 1;
		.@n2++;
	}
	mes "[Costureira]";
	mes "E o segundo.";
	next;
	.@b = .@orig[select(.@lista2$) - 1] - 1;

	if (countitem(.@id[.@a]) < 10 || countitem(.@id[.@b]) < 10) {
		mes "[Costureira]";
		mes "Não está tudo aí, meu bem. Preciso de dez de " + getitemname(.@id[.@a]) + " e dez de " + getitemname(.@id[.@b]) + " — você tem " + countitem(.@id[.@a]) + " e " + countitem(.@id[.@b]) + ".";
		close;
	}

	delitem .@id[.@a], 10;
	delitem .@id[.@b], 10;
	callfunc "FestivalPaga", 0, 1;

	mes "[Costureira]";
	mes "Perfeito. Com isso eu visto meia dúzia de foliões. Tome a sua insígnia — e volte daqui a uma três horas, que pano nunca é demais.";
	close;
}

// 5. ESTOQUE DE COMIDA - Chef
//
// Aceita QUALQUER prato de culinaria: a faixa 12041..12070 e a
// dos `Str_Dish01..05` e irmaos, os seis atributos vezes cinco
// niveis, que e o que o sistema de culinaria do RO produz.

brasilis,343,243,4	script	Chef::FestivalChef	4_M_BRZ_MAN1,{
	mes "[Chef]";
	callfunc "FestivalCheca", 4;

	mes "A barraca de comida do Festival não para, e a minha despensa não acompanha. Me traga ^0000FFum prato de culinária^000000, qualquer um, e a insígnia é sua.";

	.@achou = 0;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {

	if (.@achou == 0) {
		mes "[Chef]";
		mes "Você não está com prato nenhum. Qualquer coisa que tenha saído de uma panela de verdade serve — dessas que dão força, inteligência, o que for.";

	mes "[Chef]";
	mes "Tome a insígnia. E se sobrar panela em casa, volte daqui a três horas.";
	close;

// ---------------------------------------------------------------
// O contador de Piranhas da tarefa do Guarda.
//
// E um OnNPCKillEvent SEPARADO do que da a insignia por morte, e
// os dois rodam: o `npc_script_event` percorre todos os NPCs
// registrados com o label. Misturar os dois faria a tarefa
// depender da chance de drop.
// ---------------------------------------------------------------
-	script	#festival_tarefas::FestivalTarefas	-1,{
	end;

OnNPCKillEvent:
	// A linha mais barata primeiro: o servidor inteiro passa por
	// aqui a cada monstro morto.
	if (festival_tubarao != 1) end;
	if (killedrid != 2081) end;          // Carrapato de Tentaculo
	if (festival_piranhas >= 3) end;

	festival_piranhas = festival_piranhas + 1;
	message strcharinfo(0), "Carrapatos de Tentáculo removidos: " + festival_piranhas + "/3.";
	if (festival_piranhas >= 3)
		message strcharinfo(0), "A praia está limpa. Volte ao Guarda Salva-Vidas.";
	end;
}
// ---------------------------------------------------------------
// OS CARRAPATOS DE TENTACULO DA PRAIA
//
// Sao os quatro spawns do vendor (npc/re/mobs/towns.txt), copiados
// para ca com UMA diferenca: o nome e `--ja--` em vez de "Strange
// Hydra".
//
// `--ja--` manda o servidor usar o JapaneseName do mob_db
// (mob.cpp:458) em vez do texto desta linha. E o que faz o nome do
// bicho ter UM lugar so - db/guerra/mob_db_guerra.yml, onde ele e
// "Carrapato de Tentaculo". Com o nome escrito aqui, mexer na
// traducao nao mudaria nada em jogo, e foi exatamente isso que
// escondeu a traducao ate 2026-09-07.
