//===== Por que dialogo e nao loja de troca ==================
//= O PENDENCIAS.md secao 4b dizia "quando for escrito, e `barter`

//= A armadilha do CLAUDE.md secao 4.10 (o `itemshop` recusa moeda

//= `mes` o texto e nosso, e ele esta aqui.

//= "Sim" leva TODAS as caveiras da bolsa - nao uma, nem um numero
//= digitado. E a mesma decisao da Alleria, e pelo mesmo motivo:
//= poupa a caixa de `input`, que precisaria de validacao a mao
//= (negativo, zero, mais do que se tem). O pedido diz "1 pra 1",
//= que e a TAXA, e a taxa e respeitada seja qual for a

//=
//= A ORDEM DAS DUAS LINHAS IMPORTA, e NAO HA checkweight:
//=
//=   delitem primeiro   esvazia o slot inteiro da caveira, entao
//=                      o getitem seguinte tem sempre onde entrar
//=   peso nao aperta    a Caveira Humana pesa 0 e a Moeda Nova

//= Um checkweight aqui seria pergunta sem resposta possivel: os
//= dois itens pesam zero, entao ele nunca recusaria nada. Ver o
//= cabecalho do flores_da_ordem.txt, que tem a mesma secao por um

auction_01,192,72,2	script	Egebreu	404,{
	mes "[Egebreu]";
	mes "Compro suas caveiras, pagando com";
	mes "^0000FFMoedas Novas^000000! Mas precisam ser caveiras";
	mes "humanas, daquelas obtidas na Arena de";
	mes "Prontera. 1 pra 1. Tem interesse?";
	next;

	if (select("Sim:Não") == 2) {
		mes "[Egebreu]";
		mes "Imaginei...";
		close;
	}

	.@qtd = countitem(.caveira);
	if (.@qtd == 0) {
		mes "[Egebreu]";
		mes "Você tem vontade só te falta ação, e uma";
		mes "^0000FFCaveira Humana^000000! Chama alguem pra Arena de";
		mes "Combate, e não morra! hahahaha";
		close;
	}

	// A ORDEM IMPORTA, e o cabecalho explica por que nao ha
	// checkweight: os dois itens pesam 0, e o delitem esvazia o
	// slot da caveira antes de o getitem precisar de lugar.
	delitem .caveira, .@qtd;
	getitem .moeda, .@qtd * .preco;

	mes "[Egebreu]";
	mes "Feito! Estou aqui 24/7 só falar comigo";
	close;

OnInit:
	.caveira = 30995;   // Caveira Humana, db/guerra/item_db.yml
	.moeda   = 30998;   // Moeda Nova, db/guerra/item_db.yml
	.preco   = 1;       // Moedas Novas por caveira - o "1 pra 1"
	end;
}
