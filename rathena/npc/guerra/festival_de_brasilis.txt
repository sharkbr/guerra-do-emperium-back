// ---------------------------------------------------------------
// A INSCRICAO, uma vez so para os cinco NPCs que inscrevem.
//
//   getarg(0)  indice do elemento na tabela (0..3)
//
// `callfunc` abre escopo `.@` novo e vazio (CLAUDE.md secao 5),
// entao nada daqui depende do chamador - o indice chega por
// argumento, e a tabela vem do FestivalMae por getvariableofnpc.
// Quem chama ja deixou a janela aberta com o cabecalho do nome.
// ---------------------------------------------------------------
function	script	FestivalInscreve	{
	.@e = getarg(0);
	.@elemento$ = getvariableofnpc(.Elemento$[.@e], "FestivalMae");
	.@orixa$ = getvariableofnpc(.Orixa$[.@e], "FestivalMae");
	.@cartao = getvariableofnpc(.Cartao[.@e], "FestivalMae");

	// Ja tem Liga: nao se troca, e o texto muda conforme seja
	// ESTA Liga ou outra.
	if (festival_liga > 0) {
		.@dele = festival_liga - 1;
		mes "Você já jurou a " + getvariableofnpc(.Orixa$[.@dele], "FestivalMae") + ", da Liga do " + getvariableofnpc(.Elemento$[.@dele], "FestivalMae") + ".";
		if (.@dele == .@e)
			mes "É a minha Liga, então você é dos nossos. Vá cumprir as tarefas do Festival e traga as insígnias para a Máquina.";
		else
			mes "E não se troca de Liga no meio do Festival — nem eu nem ninguém desfaz um juramento desses. Procure o Agente da sua Liga.";
		close;
	}

	mes "Jurar à Liga do " + .@elemento$ + " é ficar sob a proteção de " + .@orixa$ + " pelo Festival inteiro. Não custa nada, e não tem volta: depois de escolher, não dá para mudar de Liga.";
	next;
	mes "A partir daí, tudo que você fizer pelo Festival te paga em Insígnia do " + .@elemento$ + " — e é com ela que a Máquina da Diversidade entrega o conjunto do nosso elemento.";
	next;
	if (select("Quero jurar à Liga do " + .@elemento$ + ":Deixa eu pensar melhor") == 2) {
		mes "Pense com calma. O Festival não vai embora hoje.";
		close;
	}

	// checkweight ANTES de gravar: `getitem` com a mochila cheia
	// LARGA O ITEM NO CHAO (CLAUDE.md secao 5), e cartao de Liga no
	// chao e cartao perdido.
	if (!checkweight(.@cartao, 1)) {
		mes "Sua bolsa está cheia demais até para um cartão. Faça espaço e volte aqui.";
		close;
	}
	festival_liga = .@e + 1;
	getitem .@cartao, 1;

	mes "Está feito. Você é da Liga do " + .@elemento$ + ", e " + .@orixa$ + " olha por você daqui em diante.";
	next;
	mes "Agora é trabalhar pelo Festival. As tarefas estão espalhadas pela cidade e podem ser refeitas a cada três horas — cada uma paga em insígnia. Caçar pelos arredores também rende, de vez em quando.";
	next;
	mes "Quando tiver insígnias na mão, desça para o meio da cidade e procure a Máquina da Diversidade — uma máquina de refrigerante que não vende refrigerante nenhum. É ela que troca insígnia por equipamento.";
	close;
}

		mes "Vá com os orixás, filho.";
		close;
	}
	end;

OnInit:
	// A TABELA. Seis colunas paralelas, e tudo neste arquivo sai
	// daqui: os Agentes, a Maquina, o drop e a bencao.
	setarray .Elemento$[0], "Fogo", "Água", "Vento", "Terra";
	setarray .Orixa$[0],    "Xangô", "Oxum", "Iansã", "Oxóssi";
	setarray .Cartao[0],    25483, 25484, 25485, 25482;
	setarray .Insignia[0],  25486, 25487, 25488, 25489;
	setarray .Loja$[0],     "FestivalFogo#loja", "FestivalAgua#loja", "FestivalVento#loja", "FestivalTerra#loja";

	.Ligas = getarraysize(.Elemento$);

	// A trava da secao 4.11: comentario nao segura ordem, contagem
	// segura. Se uma coluna crescer sozinha, isto grita no console.
	if (getarraysize(.Orixa$) != .Ligas || getarraysize(.Cartao) != .Ligas ||
	    getarraysize(.Insignia) != .Ligas || getarraysize(.Loja$) != .Ligas)
		debugmes "FESTIVAL DE BRASILIS: as seis colunas da tabela nao tem o mesmo tamanho - menu e destino vao divergir.";
	end;

// ---------------------------------------------------------------
// O motor da cacada. Sem mapa, sem sprite: so o evento global de
// morte. NPC proprio, e nao um label dentro de outro: evento que
// chega a um NPC com dialogo aberto entra na fila em vez de rodar.
// ---------------------------------------------------------------
-	script	#festival_cacada::FestivalCacada	-1,{
	end;

OnNPCKillEvent:
	// A linha mais barata primeiro: o servidor inteiro passa por
	// aqui a cada monstro morto.
	if (festival_liga < 1) end;

	.@m$ = strcharinfo(3);
	if (.@m$ != "bra_fild01" && .@m$ != "bra_dun01" && .@m$ != "bra_dun02") end;

	.@liga = festival_liga - 1;
	.@ins = getvariableofnpc(.Insignia[.@liga], "FestivalMae");

	.@quantas = 0;
	if (killedrid == .Chefe_id)
		.@quantas = .Chefe;
	if (.@quantas == 0) {
		// rand(100) e 0..99. Nunca rand(1): com um argumento so ele
		// faz `maximum -= 1` e MATA o script (CLAUDE.md secao 5).
		if (rand(100) >= .Chance) end;
		.@quantas = 1;
	}

	// `getitem` com a mochila cheia LARGA O ITEM NO CHAO (CLAUDE.md
	// secao 5), e insignia no chao de masmorra e insignia perdida.
	if (!checkweight(.@ins, .@quantas)) {
		message strcharinfo(0), "Sua bolsa está cheia — a insígnia não coube.";
		end;
	}

	getitem .@ins, .@quantas;

	// O jogador precisa VER que funcionou (CLAUDE.md 4.21): o total
	// na tela toda vez, porque insignia guardada nao aparece em
	// lugar nenhum ate ele abrir a bolsa.
	if (killedrid == .Chefe_id)
		mapannounce .@m$, strcharinfo(0) + " derrubou o Boitatá e levou " + .@quantas + " " + getitemname(.@ins) + "!", bc_map;
	else
		message strcharinfo(0), getitemname(.@ins) + ": " + countitem(.@ins) + " no total.";
	end;

OnInit:
	.Chance = 5;        // % de insignia por morte comum
	.Chefe = 25;        // insignias que o chefe da de uma vez
	.Chefe_id = 2068;   // Boitata, o chefe de bra_dun02
	end;
}
