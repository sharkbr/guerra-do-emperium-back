//===== Por que checkweight ANTES de cobrar ==================
//= `getitem` com a mochila cheia LARGA O ITEM NO CHAO: o
//= buildin_getitem cai num `map_addflooritem` quando o
//= pc_additem falha (CLAUDE.md secao 5). Quem segura a queda e o
//= `NoDrop` do item, e AQUI SO UM DOS QUATRO O TEM:
//=
//=   14675  Caixa de Sombrios   NoDrop/NoTrade/NoSell - segura
//=   23436  Martelo de Refino   sem trava nenhuma - CAI
//=   23247  Combinador          sem trava nenhuma - CAI
//=   23926  Martelo Sombrio +9  sem trava nenhuma - CAI
//=
//= Tres dos quatro premios cairiam no chao do salao, ao alcance
//= de quem passasse - e o jogador teria pago as duas Moedas.
//= Por isso o `checkweight` na frente do `delitem`.
//=
//= E POR ISSO O SORTEIO VEM ANTES DA COBRANCA. O checkweight
//= precisa saber QUAL item vai entregar E QUANTOS: os quatro
//= pesam 10 cada e sao empilhaveis, mas empilhavel so dispensa
//= slot livre se o jogador JA tem aquele item, e o Martelo de
//= Refino sai de cinco em cinco (50 de peso). Checar um e
//= entregar outro - ou checar um e entregar cinco - passa no
//= teste e larga no chao assim mesmo. Entao a ordem e:
//= sorteia -> confere o premio sorteado -> cobra -> entrega.
//= Nada e revelado antes de cobrar, entao nao ha o que refazer.
//===== Cinco martelos por giro ==============================
//= O Martelo de Refino Sombrio (23436) sai 5 por giro desde
//= 2026-08-18. Ele e o consumivel de refinar equipamento Sombrio:
//= um por tentativa, e a tentativa falha - a 15/34 de chance de
//= sair, um martelo por giro fazia o jogador gastar duas Moedas
//= Novas para uma tentativa de refino.
//=
//= A QUANTIDADE E A TERCEIRA COLUNA DA TABELA, `.qtd`, e nao um
//= numero escrito no `getitem`. E a regra 11 do CLAUDE.md pela
//= terceira vez neste arquivo: quem le a lista na tela, quem
//= confere o peso da mochila e quem entrega leem TODOS o mesmo
//= array. Numero escrito no `getitem` sairia de sincronia com o
//= texto da tela no dia em que alguem mudasse um dos dois, e a
//= divergencia nao daria erro - a maquina anunciaria cinco e
//= entregaria um.
//=
//= O `getarraysize` confere as TRES colunas no OnInit, e com isso
//= a guarda ficou mais forte do que era: coluna a menos em
//= qualquer uma delas deixa `.total` em 0 e a maquina recusa
//= abrir. Nenhuma quantidade pode ser 0 - alem de nao fazer
//= sentido, zero em variavel de `.` APAGA a entrada do array
//= (CLAUDE.md secao 5) e a coluna encolheria calada.
//=
//= E o `checkweight` passou a receber a quantidade. Cinco
//= martelos pesam 50, nao 10, e conferir 1 para entregar 5 e
//= exatamente o buraco que a secao "Por que checkweight ANTES de
//= cobrar" existe para fechar.

//===== ESPACO NO INICIO DO `mes` COLA A LINHA NA ANTERIOR ===
//= A lista de premios nasceu com dois espacos de indentacao em
//= cada linha ("  Caixa de Sombrios..."), e em jogo o Martelo
//= Sombrio +9 apareceu GRUDADO no fim da linha do Combinador de

//= O `clif_scriptmes` (src/map/clif.cpp:2472) manda a string
//= CRUA, sem `\n` nenhum - quem decide onde quebrar e o cliente.
//= E o criterio dele, medido nas seis linhas desta janela: um
//= `mes` que comeca com ESPACO e tratado como continuacao da
//= linha corrente; um que comeca com caractere visivel abre
//= linha nova. As quatro linhas da lista foram todas coladas
//= umas nas outras; tres pareceram certas porque estouraram a
//= largura da caixa e quebraram sozinhas, e a quarta coube.

//= Ou seja: TRES DAS QUATRO ESTAVAM ERRADAS E PARECIAM CERTAS.
//= Mexer no texto de qualquer uma - nome de item mais curto,
//= porcentagem com menos digitos - muda onde a quebra cai e a
//= janela se reorganiza sozinha, sem erro nenhum.
//=
//= Por isso a linha de premio NAO TEM indentacao. Para recuar
//= visualmente, usar caractere visivel (`- `, `. `), nunca
//= espaco. Vale para todo `mes` do projeto.
//===== Notas ================================================
//= A conta de saldo e informativa E e trava, ao contrario da
//= maquina.txt - la a loja de troca recusava sozinha, aqui nao

auction_01,189,58,4	script	Sombrios Totais	10375,{
	// Guarda da tabela: com as colunas fora de sincronia o OnInit
	// deixa .total em 0, e sortear com ela seria sortear fora da
	// tabela. Ver a secao "A tabela de premios sai de UM lugar so".
	if (!.total) {
		mes "[Sombrios Totais]";
		mes "O mostrador pisca em vermelho.";
		mes "A máquina está fora de serviço.";
		close;
	}

	mes "[Sombrios Totais]";
	mes "Um cilindro de vidro sobre uma coluna,";
	mes "cheio de cápsulas escuras que giram devagar.";
	mes "A fenda do painel pede ^0000FF" + .preco + " Moedas Novas^000000.";
	next;

	// A lista, a porcentagem e a quantidade saem dos MESMOS tres
	// arrays do sorteio, por laco - nenhum numero destes esta
	// escrito no arquivo. CLAUDE.md secao 4.11.
	mes "[Sombrios Totais]";
	mes "O mostrador lista o que a gaveta solta:";
	for (.@i = 0; .@i < getarraysize(.premio); .@i++) {
		// Duas casas decimais com aritmetica inteira: o +.total/2
		// arredonda em vez de truncar (15/34 da 44,12% e nao
		// 44,11%), e o zero a esquerda entra a mao porque nao ha
		// formatacao de casa decimal no script do rAthena.
		.@pct = (.peso[.@i] * 10000 + .total / 2) / .total;
		.@dec = .@pct % 100;
		// A quantidade so entra na linha quando passa de 1: premio
		// de um continua escrito como sempre foi.
		mes "^0000FF" + getitemname(.premio[.@i]) +
		    ((.qtd[.@i] > 1) ? " x" + .qtd[.@i] : "") + "^000000  " +
		    (.@pct / 100) + "," +
		    ((.@dec < 10) ? "0" + .@dec : "" + .@dec) + "%";
	}
	next;

	.@saldo = countitem(.moeda);
	mes "[Sombrios Totais]";
	mes "Você tem ^0000FF" + .@saldo + "^000000 " +
	    ((.@saldo == 1) ? "Moeda Nova" : "Moedas Novas") + ".";
	if (.@saldo < .preco) {
		mes "Não dá para um giro.";
		mes "A fenda continua aberta, esperando.";
		close;
	}
	next;

	if (select("Girar por " + .preco + " Moedas Novas",
	           "Deixa para depois") == 2) {
		mes "[Sombrios Totais]";
		mes "As cápsulas param de girar.";
		close;
	}

	// SORTEIO ANTES DA COBRANCA, e de proposito - o checkweight
	// abaixo precisa saber qual item vai entregar. Ver a secao
	// "Por que checkweight ANTES de cobrar".
	//
	// rand(.total) devolve 0..total-1; o laco anda pelos pesos
	// acumulados ate passar do sorteado. Com 15/15/3/1 as faixas
	// sao 0-14, 15-29, 30-32 e 33.
	.@sorte = rand(.total);
	.@i = 0;
	.@ate = .peso[0];
	while (.@sorte >= .@ate) {
		.@i++;
		.@ate += .peso[.@i];
	}
	.@premio = .premio[.@i];
	.@qtd = .qtd[.@i];

	// checkweight com a QUANTIDADE sorteada, nao com 1: cinco
	// martelos pesam cinco vezes mais, e o que tem de caber e o que
	// vai sair da gaveta.
	if (!checkweight(.@premio, .@qtd)) {
		mes "[Sombrios Totais]";
		mes "A gaveta tenta abrir e trava.";
		mes "Não há espaço na sua mochila,";
		mes "e a máquina não cobrou nada.";
		close;
	}

	// Segunda conferencia de saldo, colada no delitem: `delitem`
	// sem saldo nao falha limpo, aborta o script com erro no
	// console. Ver a secao "Notas".
	if (countitem(.moeda) < .preco) {
		mes "[Sombrios Totais]";
		mes "A fenda cospe de volta o que sobrou.";
		mes "Faltou ^0000FFMoeda Nova^000000.";
		close;
	}

	delitem .moeda, .preco;
	getitem .@premio, .@qtd;

	mes "[Sombrios Totais]";
	mes "A máquina range, e a gaveta abre com um estalo.";
	mes "Você recebeu ^0000FF" + getitemname(.@premio) +
	    ((.@qtd > 1) ? " x" + .@qtd : "") + "^000000.";
	close;

OnInit:
	.moeda = 30998;   // Moeda Nova, db/guerra/item_db.yml
	.preco = 2;

	// A TABELA. As tres colunas andam juntas - id em cima, peso no
	// meio, quantidade embaixo -, e sao a UNICA fonte da lista na
	// tela, da porcentagem, do sorteio e do que sai da gaveta.
	// Premio novo e uma coluna em cada uma das tres.
	//
	// Quantidade nunca e 0: zero em variavel de `.` apaga a entrada
	// e a coluna encolheria calada (CLAUDE.md secao 5).
	//
	//                       14675  23436  23247  23926
	setarray .premio[0],     14675, 23436, 23247, 23926;
	setarray .peso[0],          15,    15,     3,     1;
	setarray .qtd[0],            1,     5,     1,     1;

	.@n = getarraysize(.premio);
	.@np = getarraysize(.peso);
	.@nq = getarraysize(.qtd);

	// Dois `if` e nao um `||`: os operadores logicos do rAthena nao
	// fazem curto-circuito (CLAUDE.md secao 5), e assim a mensagem
	// ainda diz qual coluna destoou.
	.@desalinhado = 0;
	if (.@n != .@np) .@desalinhado = 1;
	if (.@n != .@nq) .@desalinhado = 1;
	if (.@desalinhado) {
		// .total fica 0 e o corpo recusa a abrir. Melhor a maquina
		// calada que a maquina sorteando fora da tabela.
		debugmes "Maquina de Sombrios Totais: .premio (" + .@n +
		         "), .peso (" + .@np + ") e .qtd (" + .@nq +
		         ") com tamanhos diferentes - SORTEIO DESALINHADO.";
		end;
	}

	.total = 0;
	for (.@i = 0; .@i < .@n; .@i++)
		.total += .peso[.@i];
	end;
}
