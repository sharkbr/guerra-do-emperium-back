//===== Guerra do Emperium ===================================
//= Sprite da Teletransportadora
//===== Onde =================================================
//= Em lugar nenhum. E um NPC flutuante, sem corpo e sem mapa:
//= so existe para rodar uma vez no OnInit.
//===== Descricao ============================================
//= Troca o sprite das 45 duplicatas do Warper
//= (npc/guerra/teletransportadora.txt) de 811 (4_F_KAFRA1, a
//= moca de uniforme da Kafra) para 10007 (JT_PORTAL, o portal).
//===== Por que de fora, e nao na propria tabela =============
//= O sprite esta escrito 45 vezes, uma por duplicata, no fim
//= daquele arquivo. Ate 2026-08-08 o motivo de nao mexer la era
//= que ele era o npc/custom/warper.txt do rAthena, e editar 45
//= linhas de codigo de terceiros sujaria o vendor.
//=
//= HOJE AQUELE ARQUIVO E UMA COPIA NOSSA, e o motivo mudou de
//= forma sem mudar de conclusao: a copia vale exatamente pelo
//= diff curto contra o upstream, e 45 linhas trocadas la
//= tornariam caro trazer qualquer correcao futura do rAthena.
//= setnpcdisplay faz o mesmo de fora e nao custa linha nenhuma.
//=
//= ARMADILHA AO DEPURAR: quem abrir a teletransportadora vai ler
//= 811 e nao vai bater com o portal que aparece no jogo. A troca
//= e aqui. Se um dia quiser a Kafra de volta, basta comentar a
//= linha deste arquivo no scripts_guerra.conf - aquele arquivo
//= sozinho ja nasce Kafra.
//===== Notas ================================================
//= 10007 = JT_PORTAL, de src/map/npc.hpp:944, contado a partir
//= do NPC_RANGE3_START = 10000 da linha 937. Esta dentro da
//= faixa que o npcdb_checkid aceita (npc.hpp:1595) e tem
//= viewdata montada em npc.cpp:6332.
//=
//= A lista de nomes saiu do proprio arquivo da
//= teletransportadora. Ela detecta nome que SUMIU (avisa no
//= console), mas nao detecta nome ACRESCENTADO: cidade nova
//= nasceria Kafra e ficaria fora da lista, calada. Ao trazer
//= mudanca do rAthena para a copia, reconferir contra o fim
//= dela.
//============================================================

-	script	SpriteTeletransportadora	-1,{
	end;

OnInit:
	// Ordem igual a do fim da teletransportadora: primeiro o bloco comum,
	// depois o bloco Renewal.
	setarray .@nome$[0],
		"Warper#tre",   "Warper#alb",   "Warper#alde",  "Warper#ama",
		"Warper#ayo",   "Warper#com",   "Warper#einbe", "Warper#einbr",
		"Warper#orc",   "Warper#gef",   "Warper#glh",   "Warper#gon",
		"Warper#hug",   "Warper#izd",   "Warper#izl",   "Warper#jaw",
		"Warper#lhz",   "Warper#lou",   "Warper#man",   "Warper#mid",
		"Warper#mjo",   "Warper#moc",   "Warper#mor",   "Warper#mos",
		"Warper#nam",   "Warper#nif",   "Warper#arc",   "Warper#pay",
		"Warper#prt",   "Warper#cul",   "Warper#rac",   "Warper#spl",
		"Warper#thor",  "Warper#umb",   "Warper#ve",    "Warper#xmas",
		"Warper#yuno",
		// Renewal:
		"Warper#bra",   "Warper#dew",   "Warper#dic",   "Warper#ecl",
		"Warper#las",   "Warper#mal",   "Warper#ma",    "Warper#mora";

	.@sumiram = 0;
	for (.@i = 0; .@i < getarraysize(.@nome$); .@i++) {
		// setnpcdisplay devolve 1 quando o NPC nao existe.
		if (setnpcdisplay(.@nome$[.@i], 10007))
			.@sumiram++;
	}

	if (.@sumiram)
		debugmes "SpriteTeletransportadora: "+.@sumiram+" de "+getarraysize(.@nome$)+" duplicatas do Warper nao foram encontradas. A lista deste arquivo saiu de sincronia com o fim de npc/guerra/teletransportadora.txt.";
	end;
}
