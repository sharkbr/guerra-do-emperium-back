//===== Onde o encanto MORA na peca ==========================
//= Encanto nao e campo do item: e uma CARTA gravada numa cova,
//= de tras para frente. E o arranjo oficial, e o rAthena o usa
//= em todos os encantadores dele (ver `enchan_verus.txt:229` -
//= `if (.@card[3] < 1 && .@max_slot < 4)`).
//=
//=   cova 0   a cova de CARTA de verdade, e nunca se toca nela
//=   cova 1   3o encanto (so calcado e capa)
//=   cova 2   2o encanto
//=   cova 3   1o encanto

//= O piso e calculado, nao escrito: `4 - <teto do grupo>`, mas
//= nunca abaixo do numero de covas REAIS da peca
//= (`getitemslots`). As onze pecas de hoje tem 1 cova (as Botas

//= CUIDADO AO LER ISTO NO CLIENTE: a peca continua dizendo
//= "[1]" no nome, porque o `[n]` vem do `slotCount` do
//= itemInfo.lua e nao do pacote. Os encantos aparecem na
//= DESCRICAO, que le as quatro covas.
//===== Como a peca volta inteira ============================
//= Nao existe comando que escreva numa cova: o caminho e
//= desmontar e remontar, `delequip` mais `getitem4`.
//=
//= Remontar PERDE O QUE NAO FOR PASSADO DE VOLTA:
//=
//=   refino          `getequiprefinerycnt`, lido antes
//=   as quatro covas `getequipcardid`, lidas antes
//=   grau de encanto `@inventorylist_enchantgrade[]`
//=   vinculo (bound) `@inventorylist_bound[]`
//=   bonus aleatorio `@inventorylist_option_*`, os cinco
//=
//= OS TRES PRIMEIROS NAO TEM `getequip*` NENHUM: quem os tem e
//= o `getinventorylist`, e por isso ele e chamado e a linha da
//= peca vestida e procurada pelo bit de posicao
//= (`@inventorylist_equip[] & EQP_*`, que e EQP e nao EQI). Sem
//= isso, encantar um item VINCULADO o devolveria solto.
//=
//= OS BONUS ALEATORIOS SAO PRESERVADOS. Nada no bRO manda
//= perde-los, e um encanto que apagasse a opcao aleatoria da
//= peca seria uma perda calada. O `getitem4` le a quantidade
//= pela chave mais alta do array de IDs
//= (`script_array_highest_key`, script.cpp:7104), entao passar
//= os cinco arrays cobre de nenhuma opcao a cinco sem contar
//= nada.
//=
//= ITEM ALUGADO E RECUSADO: `@inventorylist_expire[]` diz que a
//= peca tem prazo, e remontar com `getitem4` devolveria uma
//= peca eterna.
//=
//= A PECA VOLTA DESVESTIDA. E consequencia do `delequip` e nao
//= ha o que fazer - `equip` de volta escolheria sozinho entre
//= duas copias iguais na mochila. O NPC avisa por escrito.
//===== O BUG QUE COMEU UM ANEL, e a trava que ficou =========
//= EM 2026-08-28, NA PRIMEIRA IDA A JOGO, ESTE NPC APAGOU UM
//= Anel de Jasper (490113) COM UMA Carta Ahat (27322) DENTRO,
//= e travou a janela no "Deixa comigo". A causa esta registrada
//= aqui porque a crenca que a produziu estava escrita no
//= CLAUDE.md, e era falsa.
//=
//= A remontagem morava num `callsub S_Remonta`, escrito sobre a
//= afirmacao de que "`callsub` nao abre escopo novo: ele
//= enxerga as `.@` de quem chamou". O `buildin_callsub`
//= (src/map/script.cpp:5508) faz o OPOSTO, e e a mesma coisa
//= que o `callfunc` faz:
//=
//=     st->stack->scope.vars   = i64db_alloc(...);
//=     st->stack->scope.arrays = idb_alloc(...);
//=
//= Escopo NOVO e VAZIO. O que `callsub` tem de diferente e so
//= passar `.@` como ARGUMENTO por referencia, para o `getarg`.
//=
//= Dentro do `S_Remonta`, entao, `.@part` valia 0 - que e
//= `EQI_ACC_L` - e `.@equip_id` valia 0. O `delequip 0` tirou e
//= APAGOU o que estivesse no acessorio esquerdo, e o
//= `getitem4 0` falhou com "Nonexistant item 0 requested",
//= matando o script com a janela aberta. Os dois sintomas eram
//= a mesma linha.
//=
//= DUAS COISAS FICARAM, e as duas valem mais que o conserto:
//=
//= 1) A REMONTAGEM E INLINE, e nao ha `callsub` em nenhum
//=    caminho que apague ou crie item. Sub-rotina que precise
//=    de valor do chamador so serve com `getarg` - e array nao
//=    passa por `getarg`, o que torna `callsub` errado para
//=    este trabalho por natureza, e nao por descuido.
//=
//= 2) A TRAVA DE REMONTAGEM, inline, logo antes do `delequip`. Ela
//=    recusa
//=    seguir se a peca vestida agora nao for exatamente a que o
//=    NPC leu. Ela e redundante hoje - e e de proposito: e ela
//=    que teria transformado o bug acima em "o Lider recusou
//=    atender", em vez de um item apagado. Comando que APAGA
//=    coisa do jogador nao roda sobre valor que nao foi
//=    conferido na linha de cima.
//===== Armadilhas que este arquivo respeita =================
//= `rand(1)` NAO devolve 0: ele MATA o script (buildin_rand,
//= script.cpp:5604). Toda faixa de sorteio aqui tem 2 ou mais
//= linhas hoje, mas o sorteio esta guardado com
//= `if (.@qtde > 1)` mesmo assim.
//=
//= `||` e `&&` NAO fazem curto-circuito (C_LOR/C_LAND,
//= script.cpp:3839): os dois lados sao avaliados sempre. Por
//= isso toda guarda que combina "achei" com "olha o array" esta
//= escrita como `if` aninhado, nunca com o operador.
//=
//= AS TRES `F_IsEquip*Hack` SO VALEM COM VALOR VELHO. Elas
//= comparam o que esta vestido AGORA com o que se passa a elas,
//= e o reflexo natural - reler a peca logo antes de chamar - as
//= transforma em `x == x`: passam sempre, sem erro e sem log.
//= Por isso o retrato da peca e tirado UMA VEZ, antes do
//= primeiro `next`, e nao e refeito depois.
//=
//= UM `mes` POR PARAGRAFO, e nunca um `mes` por linha de tela.
//= Quem quebra a linha e o cliente, pela largura da janela; um
//= texto ja quebrado no script aparece quebrado DUAS vezes, com
//= as sobras curtas no meio da frase. Ver CLAUDE.md secao 4.22.
//===== Notas ================================================
//= Os 46 encantos e as 11 pecas foram conferidos um a um nas
//= duas travas, em 2026-08-28: existem no item_db do servidor
//= (os 46 com `SubType: Enchant`, que e o CARD_ENCHANT do

//=          chamados "ATQ +1%" (4819 e 4882) na vizinhanca do
//=          "ATQM +3%", e nenhum deles e o par dele.
//=
//= NENHUM ARQUIVO DO rAthena FOI TOCADO. A celula 43,65 ja
//= esta livre porque o `#ordem_montagem` do
//= ordem_dos_exploradores.txt desliga o `Auction Broker#lhz3`
//= que ficava nela - este arquivo depende daquele.

auction_02,43,65,4	script	Líder#ordem_lider	898,{
	mes "[Líder]";
	mes "A Ordem guardou muita coisa dos tempos antigos, mas o que eu guardei foram as ^0000FFpedras^000000.";
	next;
	mes "[Líder]";
	mes "Encanto ^0000FFcapa^000000, ^0000FFcalçado^000000 e ^0000FFacessório^000000 - e reseto, quando o encanto sai ruim. Um de cada vez, e por um preço.";
	next;

	// ----- que pecas o jogador esta vestindo que eu atendo ------------
	// O menu SAI da tabela do OnInit, num laco. Ver "As tabelas, e por
	// que sao UMA SO" no cabecalho: nao ha uma segunda lista de nomes
	// para sair de sincronia com os ids.
	.@ng = getarraysize(.rotulo$) - 1;   // as tabelas sao 1-based
	.@np = getarraysize(.peca) - 1;
	.@menu$ = "";
	.@qtd = 0;
	deletearray .@gr[0];

	for (.@g = 1; .@g <= .@ng; .@g++) {
		if (getequipisequiped(.slot[.@g]) == 0)
			continue;
		.@id = getequipid(.slot[.@g]);

		// `&&` avalia os dois lados sempre - por isso o `if` aninhado.
		.@atende = 0;
		for (.@j = 1; .@j <= .@np; .@j++) {
			if (.peca[.@j] == .@id) {
				if (.grupo[.@j] == .@g)
					.@atende = 1;
			}
		}
		if (.@atende == 0)
			continue;

		.@qtd++;
		.@gr[.@qtd] = .@g;
		.@menu$ = .@menu$ + .rotulo$[.@g] + " - " +
		          getequipname(.slot[.@g]) + ":";
	}

	if (.@qtd == 0) {
		mes "[Líder]";
		mes "Você não está usando nada que eu saiba encantar. E acessório tem que estar na ^FF0000mão direita^000000 - a pedra não assenta do outro lado.";
		next;
		callsub S_Lista;
		mes "[Líder]";
		mes "Vista uma dessas e volte.";
		close;
	}

	.@esc = select("Deixa para depois:" + .@menu$);
	if (.@esc == 1) {
		mes "[Líder]";
		mes "As pedras não têm pressa.";
		close;
	}
	.@g = .@gr[.@esc - 1];

	// A escolha vira posicao de equipamento AQUI, e a guarda existe
	// porque foi um `.@g` fora da faixa que fez o `delequip` cair no
	// acessorio esquerdo em 2026-08-28. Ver "O BUG QUE COMEU UM ANEL".
	if (.@g < 1 || .@g > .@ng) {
		debugmes "Lider#ordem_lider: escolha " + .@esc + " virou grupo " +
		         .@g + ", fora de 1.." + .@ng + ".";
		mes "[Líder]";
		mes "Me perdi no que você escolheu. Fale comigo de novo.";
		close;
	}
	.@part = .slot[.@g];

	// ----- o retrato da peca ------------------------------------------
	// ESTE E O RETRATO QUE VALE. Entre este bloco e a cobranca ha `next`
	// de sobra para o jogador trocar de equipamento, e o que fecha essa
	// porta e comparar o que foi lido AQUI com o que estiver vestido LA -
	// e o que as tres `F_IsEquip*Hack` do rAthena fazem, e por isso elas
	// so servem se receberem estes valores, nao valores relidos na hora.
	if (getequipisequiped(.@part) == 0) {
		mes "[Líder]";
		mes "Não há nada vestido aí.";
		close;
	}
	.@equip_id = getequipid(.@part);
	.@refino = getequiprefinerycnt(.@part);
	setarray .@cova[0],
		getequipcardid(.@part,0), getequipcardid(.@part,1),
		getequipcardid(.@part,2), getequipcardid(.@part,3);

	// O piso: de que cova para cima o encanto pode morar. Ver "Onde o
	// encanto MORA na peca" - a conta e que impede o encanto de comer
	// uma cova de carta de verdade numa peca de duas covas.
	.@piso = 4 - .maxenc[.@g];
	.@covas_reais = getitemslots(.@equip_id);
	if (.@piso < .@covas_reais)
		.@piso = .@covas_reais;
	.@teto = 4 - .@piso;

	.@usados = 0;
	for (.@i = .@piso; .@i <= 3; .@i++) {
		if (.@cova[.@i] > 0)
			.@usados++;
	}

	mes "[Líder]";
	mes "^0000FF" + getequipname(.@part) + "^000000";
	mes "Refino ^0000FF+" + .@refino + "^000000, com ^0000FF" + .@usados +
	    "^000000 de " + .@teto + " encantos.";
	if (.@usados > 0) {
		for (.@i = 3; .@i >= .@piso; .@i--) {
			if (.@cova[.@i] > 0)
				mes "- " + getitemname(.@cova[.@i]);
		}
	}
	mes "Toda pedra que eu tenho para esse tipo de peça tem a ^0000FFmesma chance^000000 de assentar - são ^0000FF" +
	    (.fim[.@g] - .ini[.@g] + 1) + "^000000, e nenhuma é mais rara que as outras.";
	next;

	.@acao = select("Encantar mais um:Resetar os encantos:Deixa para depois");
	if (.@acao == 3) {
		mes "[Líder]";
		mes "As pedras não têm pressa.";
		close;
	}

	if (.@acao == 1) {
		if (.@usados >= .@teto) {
			mes "[Líder]";
			mes "Essa peça já tem os ^0000FF" + .@teto + "^000000 encantos que aguenta. Para trocar, tem que resetar - e reseta ^FF0000tudo^000000.";
			close;
		}
	} else {
		if (.@usados == 0) {
			mes "[Líder]";
			mes "Não há encanto nenhum nessa peça para eu tirar.";
			close;
		}
	}

	mes "[Líder]";
	mes "Custa ^0000FF" + .custo_moeda + " " +
	    ((.custo_moeda == 1) ? "Moeda do Explorador" : "Moedas do Explorador") +
	    "^000000 e ^0000FF" + .custo_zeny + "^000000 zeny.";
	if (.@acao == 1) {
		mes "A pedra assenta em ^0000FF" + .chance + "^000000 de cada 100 tentativas. Se não assentar, ^FF0000você perde só o pagamento^000000 - a peça, o refino e os encantos que já estão nela ficam como estão.";
	} else {
		mes "^FF0000Sai tudo^000000: os " + .@usados + " encantos se perdem de uma vez. O refino e a carta da peça ficam.";
	}
	mes "E a peça volta para a mochila ^FF0000desvestida^000000 - é assim que eu trabalho. Vista de novo depois.";
	next;

	if (select("Sim, pode fazer:Deixa para depois") == 2) {
		mes "[Líder]";
		mes "As pedras não têm pressa.";
		close;
	}

	// ===== daqui para baixo nao ha mais menu ==========================
	// A PECA NAO E RELIDA AQUI, e isso e o ponto: as tres funcoes abaixo
	// comparam o que esta vestido AGORA com o retrato tirado la em cima,
	// antes do primeiro `next`, e registram no `logmes` quando difere.
	// Reler e passar o valor recem-lido faria as tres compararem uma
	// coisa com ela mesma - passariam sempre, e a porta ficaria aberta.
	if (getequipisequiped(.@part) == 0) {
		mes "[Líder]";
		mes "Você tirou a peça. Vista de novo e me chame.";
		close;
	}
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id)) {
		mes "[Líder]";
		mes "Essa não é mais a peça que você me mostrou. Nada foi cobrado.";
		close;
	}
	if (callfunc("F_IsEquipRefineHack", .@part, .@refino)) {
		mes "[Líder]";
		mes "O refino dessa peça mudou enquanto conversávamos. Me mostre de novo - nada foi cobrado.";
		close;
	}
	if (callfunc("F_IsEquipCardHack", .@part, .@cova[0], .@cova[1],
	             .@cova[2], .@cova[3])) {
		mes "[Líder]";
		mes "As covas dessa peça mudaram enquanto conversávamos. Me mostre de novo - nada foi cobrado.";
		close;
	}

	// Peca forjada ou assinada usa as covas para outra coisa: a cova 0 e
	// um marcador e as covas 2 e 3 guardam o id do dono partido em dois.
	// 254 = CARD0_CREATE, 255 = CARD0_FORGE, 256 = CARD0_PET.
	if (.@cova[0] == 254 || .@cova[0] == 255 || .@cova[0] == 256) {
		mes "[Líder]";
		mes "Essa peça tem assinatura de quem a fez. Não mexo nesse tipo de trabalho.";
		close;
	}

	// O vinculo, o prazo, o grau de encanto e os bonus aleatorios so
	// existem aqui - nao ha `getequip*` para nenhum deles. Ver "Como a
	// peca volta inteira", no cabecalho.
	.@achou = 0;
	.@vinculo = 0;
	.@prazo = 0;
	.@grau = 0;
	// Os tres arrays nascem EXISTINDO, e nao por zelo: o `getitem4` os
	// exige como array de verdade (`script_array_src`, script.cpp:7055) e
	// recusa a chamada inteira se um deles nunca tiver sido declarado.
	// Zerados eles ficam vazios - o que e o certo para peca sem bonus
	// aleatorio nenhum, porque a quantidade que o buildin le e a chave
	// mais alta do array de IDs.
	setarray .@op_id[0],  0, 0, 0, 0, 0;
	setarray .@op_val[0], 0, 0, 0, 0, 0;
	setarray .@op_par[0], 0, 0, 0, 0, 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_equip[.@i] & .bit[.@g]) {
			if (@inventorylist_id[.@i] == .@equip_id) {
				.@vinculo = @inventorylist_bound[.@i];
				.@prazo   = @inventorylist_expire[.@i];
				.@grau    = @inventorylist_enchantgrade[.@i];
				setarray .@op_id[0],
					@inventorylist_option_id1[.@i],
					@inventorylist_option_id2[.@i],
					@inventorylist_option_id3[.@i],
					@inventorylist_option_id4[.@i],
					@inventorylist_option_id5[.@i];
				setarray .@op_val[0],
					@inventorylist_option_value1[.@i],
					@inventorylist_option_value2[.@i],
					@inventorylist_option_value3[.@i],
					@inventorylist_option_value4[.@i],
					@inventorylist_option_value5[.@i];
				setarray .@op_par[0],
					@inventorylist_option_parameter1[.@i],
					@inventorylist_option_parameter2[.@i],
					@inventorylist_option_parameter3[.@i],
					@inventorylist_option_parameter4[.@i],
					@inventorylist_option_parameter5[.@i];
				.@achou = 1;
				break;
			}
		}
	}
	if (.@achou == 0) {
		debugmes "Lider#ordem_lider: peca vestida nao achada no " +
		         "getinventorylist (id " + .@equip_id + ").";
		mes "[Líder]";
		mes "Não consegui pegar essa peça direito. Vista de novo e me chame.";
		close;
	}
	if (.@prazo != 0) {
		mes "[Líder]";
		mes "Essa peça é ^FF0000alugada^000000. Se eu a desmontar, ela volta sem prazo - e isso eu não faço.";
		close;
	}

	// `.@piso`, `.@teto`, `.@usados` e `.@cova[]` continuam valendo: `.@`
	// pertence a ESTA execucao do script e atravessa `next` inteiro. E as
	// tres funcoes acima acabaram de provar que a peca vestida ainda e a
	// mesma que gerou esses numeros.
	if (countitem(.moeda) < .custo_moeda) {
		mes "[Líder]";
		mes "Faltam ^0000FFMoedas do Explorador^000000. São " + .custo_moeda +
		    ", e você tem " + countitem(.moeda) + ".";
		close;
	}
	if (Zeny < .custo_zeny) {
		mes "[Líder]";
		mes "Falta zeny. São ^0000FF" + .custo_zeny + "^000000, e você tem ^0000FF" + Zeny + "^000000.";
		close;
	}

	// ----- daqui em diante o pagamento ja foi ------------------------
	delitem .moeda, .custo_moeda;
	Zeny -= .custo_zeny;

	mes "[Líder]";
	mes "Deixa comigo.";
	next;

	.@tirados = 0;
	.@novo = 0;

	if (.@acao == 2) {
		// ----- resetar ------------------------------------------------
		// So as covas de ENCANTO, e so as da janela: a cova de carta de
		// verdade nao entra na conta, e a Essencia de Morroc, que mora
		// nela, fica onde esta.
		for (.@i = .@piso; .@i <= 3; .@i++) {
			if (.@cova[.@i] > 0) {
				if (getiteminfo(.@cova[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT) {
					.@cova[.@i] = 0;
					.@tirados++;
				}
			}
		}
		if (.@tirados == 0) {
			debugmes "Lider#ordem_lider: reset nao achou encanto na " +
			         "janela (id " + .@equip_id + ", piso " + .@piso + ").";
			getitem .moeda, .custo_moeda;
			Zeny += .custo_zeny;
			mes "[Líder]";
			mes "Não achei encanto nenhum para tirar. Tome o seu de volta.";
			close;
		}
	} else {
		// ----- encantar -----------------------------------------------
		if (rand(100) >= .chance) {
			specialeffect2 EF_SUI_EXPLOSION;
			mes "-- Resultado --";
			mes "A pedra ^FF0000não assentou^000000. ^0000FFNada se perdeu^000000: a peça, o refino e os encantos que já estavam nela estão como estavam.";
			next;
			mes "[Líder]";
			mes "Foi só o material. Quiser tentar de novo, é só falar.";
			close;
		}

		// UMA FAIXA SO POR GRUPO, e o refino NAO entra nesta conta.
		// Ver "A CHANCE, e por que a regra 7 nao existe aqui" no
		// cabecalho: a coluna inteira do browiki sorteia com peso
		// igual, e nao ha segunda faixa nem tabela de peso.
		.@ini = .ini[.@g];
		.@fim = .fim[.@g];
		.@qtde = .@fim - .@ini + 1;

		// `rand(1)` NAO devolve 0 - ele MATA o script (script.cpp:5604).
		// Nenhuma faixa de hoje tem uma linha so, e esta guarda e o que
		// garante que continue funcionando se um dia tiver.
		.@k = 0;
		if (.@qtde > 1)
			.@k = rand(.@qtde);
		.@novo = .enc[.@ini + .@k];

		// A cova livre mais ALTA, que e a ordem oficial de preenchimento:
		// 3, depois 2, depois 1.
		.@onde = -1;
		for (.@i = 3; .@i >= .@piso; .@i--) {
			if (.@onde < 0) {
				if (.@cova[.@i] == 0)
					.@onde = .@i;
			}
		}
		if (.@onde < 0 || .@novo <= 0) {
			debugmes "Lider#ordem_lider: encanto impossivel (id " +
			         .@equip_id + ", piso " + .@piso + ", cova " + .@onde +
			         ", pedra " + .@novo + ").";
			getitem .moeda, .custo_moeda;
			Zeny += .custo_zeny;
			mes "[Líder]";
			mes "Errei a conta. Tome o seu de volta.";
			close;
		}
		.@cova[.@onde] = .@novo;
	}

	// ===== A REMONTAGEM =================================================
	// INLINE, e nao em `callsub`. Ver "O BUG QUE COMEU UM ANEL" no
	// cabecalho: `callsub` abre escopo `.@` NOVO E VAZIO
	// (script.cpp:5508), e dentro dele `.@part` valia 0 - o acessorio
	// esquerdo - e o `delequip` apagou o anel de quem estava testando.
	//
	// A TRAVA ABAIXO E REDUNDANTE DE PROPOSITO. Nenhum comando que apague
	// coisa do jogador roda sobre valor que nao foi conferido na linha de
	// cima; se um dia algo voltar a zerar essas variaveis, o Lider recusa
	// atender em vez de comer equipamento.
	.@ok = 1;
	if (.@equip_id <= 0) .@ok = 0;
	if (.@part < 0) .@ok = 0;
	if (.@part > 3) .@ok = 0;
	if (.@part != .slot[.@g]) .@ok = 0;
	if (.@ok == 1) {
		if (getequipisequiped(.@part) == 0) .@ok = 0;
	}
	if (.@ok == 1) {
		if (getequipid(.@part) != .@equip_id) .@ok = 0;
	}
	if (.@ok == 0) {
		debugmes "Lider#ordem_lider: a trava de remontagem recusou " +
		         "(part " + .@part + ", grupo " + .@g + ", id " +
		         .@equip_id + ").";
		getitem .moeda, .custo_moeda;
		Zeny += .custo_zeny;
		mes "[Líder]";
		mes "Alguma coisa saiu do lugar na minha bancada e eu não vou arriscar sua peça. Tome o seu de volta e me chame de novo.";
		close;
	}

	delequip .@part;
	if (.@vinculo > 0)
		getitembound4 .@equip_id, 1, 1, .@refino, 0, .@cova[0], .@cova[1], .@cova[2], .@cova[3], .@vinculo, .@grau, .@op_id, .@op_val, .@op_par;
	else
		getitem4 .@equip_id, 1, 1, .@refino, 0, .@cova[0], .@cova[1], .@cova[2], .@cova[3], .@grau, .@op_id, .@op_val, .@op_par;

	if (.@acao == 2) {
		specialeffect2 EF_DETOXICATION;
		mes "-- Resultado --";
		mes "^0000FF" + .@tirados + "^000000 " +
		    ((.@tirados == 1) ? "encanto saiu" : "encantos saíram") +
		    " da peça. O refino, a carta e os bônus aleatórios ficaram.";
		next;
		mes "[Líder]";
		mes "Está limpa. A peça voltou ^FF0000desvestida^000000 para a mochila.";
		close;
	}

	specialeffect2 EF_MAXPOWER;
	mes "-- Resultado --";
	mes "A pedra assentou: ^0000FF" + getitemname(.@novo) + "^000000.";
	mes "Encantos agora: ^0000FF" + (.@usados + 1) + "^000000 de " + .@teto + ".";
	next;
	mes "[Líder]";
	mes "A peça voltou ^FF0000desvestida^000000 para a mochila. Vista de novo e veja na descrição.";
	close;

S_Lista:
	.@ls_ng = getarraysize(.rotulo$) - 1;
	.@ls_np = getarraysize(.peca) - 1;
	for (.@ls_g = 1; .@ls_g <= .@ls_ng; .@ls_g++) {

		    .maxenc[.@ls_g] + " encantos):";
		for (.@ls_j = 1; .@ls_j <= .@ls_np; .@ls_j++) {
			if (.grupo[.@ls_j] == .@ls_g)
				mes "- " + getitemname(.peca[.@ls_j]);
		}
	}
	next;
	return;

		end;
	}
