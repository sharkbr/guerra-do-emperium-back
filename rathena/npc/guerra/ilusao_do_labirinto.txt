//===== Guerra do Emperium ===================================
//= Ilusão do Labirinto - a missão de acesso, a Fenda e o MVP
//===========================================================
//= Mapa prt_mz03_i, nível mínimo 170. A porta é a Fenda
//= Retorcida, em prt_maze01 99,23 - a mesma posição do bRO -,
//= e ela SÓ EXISTE para quem fez a missão de acesso.
//===========================================================
//
//= O QUE ESTE ARQUIVO É, E O QUE ELE NÃO É
//
//= Aqui mora a MISSÃO DE ACESSO e a MECÂNICA: os NPCs da
//= missão, os quatro Restos, a porta, os quatro Noviços com
//= renascimento de uma hora, o Bafomé Caótico que só nasce
//= quando os quatro caem, e a invencibilidade dele.
//
//= Os portais internos do labirinto e o povoamento das 25 salas
//= NÃO estão aqui - são gerados em
//= npc/guerra/ilusao_do_labirinto_mapa.txt por
//= ferramentas/monta_ilusao_do_labirinto.py, junto com os
//= catorze monstros. Aquele arquivo não se edita à mão.
//
//= POR QUE O MAPA PRECISA DOS DOIS ARQUIVOS
//
//= O chão andável do prt_mz03_i está partido em 26 pedaços que
//= não se tocam: 25 salas e um punhado de ruído de borda. Quem
//= liga uma sala na outra são os 61 portais do arquivo gerado.
//= Sem eles o jogador entra e fica preso numa sala de 500
//= células, sem erro nenhum no servidor para denunciar isso.
//
//= A MISSÃO DE ACESSO - E POR QUE ELA VEIO DEPOIS
//
//= A primeira versão desta caverna (2026-09-11, de manhã) abria
//= a Fenda para qualquer um com nível 170. O dono entrou de
//= primeira, sem missão nenhuma, e pediu o contrário: a caverna
//= fechada atrás dos passos do bRO, e a Fenda NEM VISÍVEL para
//= quem não começou. É o que está escrito abaixo.
//
//= Os sete passos são os da página Ilusão_do_Labirinto do
//= arquivo.browiki.org, que é a fonte do bRO (regra 3 do
//= CLAUDE.md), e as coordenadas são as de lá - com UMA correção
//= do dono, em 2026-09-12:
//
//=   SÃO TRÊS NPCs, e cada um no seu lugar (nomes e sprites
//=   conferidos em jogo pelo dono em 2026-09-12):
//=
//=     Irene                    prt_fild01 136,370  4_F_ACOLYTE
//=     Membro do Clã Irene      prt_maze01 99,20    4_F_ERENE
//=     Líder do Clã Esmeralda   prontera 212,320    4_F_08
//=
//=   A da PORTA é só "Irene", e é a que começa a missão. Quem
//=   tem o título de clã é a de dentro do labirinto, que fica
//=   na frente da greta e atende os passos 3 e 5. A Líder é
//=   uma só e não sai de Prontera.
//
//=   1. falar com a Irene, na porta do Labirinto da Floresta
//=      (prt_fild01 136,370), e aceitar ajudar;
//=   2. encontrar a Líder do Clã Esmeralda em Prontera
//=      (212,320);
//=   3. entrar no labirinto e falar com o Membro do Clã Irene
//=      (prt_maze01 99,20), que revela a Fenda Retorcida
//=      (prt_maze01 99,23), logo abaixo dela;
//=   4. entrar e recolher os quatro Restos, cada um com uma
//=      barra de conjuração que o dano interrompe:
//=        Andrea  107,104   Agnes    10,18
//=        Silvano 135,68    Cecília 183,26
//=   5. ao pegar o último, o jogador é puxado para fora e fala
//=      com o Membro do Clã Irene de novo;
//=   6. os quatro espíritos se reúnem na entrada;
//=   7. falar com Andrea encerra a missão - sem recompensa de
//=      item, como no bRO. A recompensa É o acesso.
//
//= O ESTADO MORA EM VARIÁVEL DE PERSONAGEM, E NÃO EM QUEST
//
//= `ilusao_labirinto` vai de 0 a 6, e `ilusao_labirinto_restos`
//= é a máscara dos quatro Restos (1, 2, 4, 8).
//
//= Não é `setquest` de propósito: quest que o CLIENTE não
//= conhece DERRUBA o cliente (CLAUDE.md §5), e a janela de
//= missões exigiria gerar a outra metade com o
//= monta_missoes_da_ordem.py e mandar patch. Variável de
//= personagem é também o que o próprio vendor usa nas outras
//= ilusionais (`illusion_moonlight` e irmãs, em
//= npc/re/quests/quests_illusion_dungeons.txt).
//
//= COMO A FENDA FICA INVISÍVEL
//
//= `cloakonnpc` sem char_id esconde para todo mundo, e é o que
//= o OnInit faz com a Fenda, com os quatro recrutas e com os
//= quatro Restos. O `cloakoffnpc "<npc>",<char_id>` mostra para
//= UM jogador - e dura só até ele trocar de mapa ou sair
//= (doc/script_commands.txt, *cloakonnpc). Por isso existe o
//= vigia lá embaixo, no OnPCLoadMapEvent, com `loadevent` nos
//= dois mapas: é ele que redecide, a cada entrada de mapa, quem
//= vê o quê.
//
//= Cloak não é `disablenpc`: o NPC continua de pé, os OnTimer
//= continuam correndo e o clique ainda funciona - o que some é
//= o desenho e a área de toque. Por isso a Fenda ainda responde
//= com uma recusa a quem clicar nela sem ter a missão, que foi
//= o segundo pedido do dono.
//
//= DE ONDE VEM A MECÂNICA DO CALABOUÇO
//
//= Da mesma página. O que ela diz, e está implementado abaixo:
//
//=   - para invocar o Bafomé Caótico é preciso eliminar os 4
//=     Noviços - Andrea, Agnes, Silvano e Cecília;
//=   - cada Noviço renasce uma hora depois de morto;
//=   - uma hora após a derrota do MVP, os Noviços reaparecem;
//=   - enquanto vivo, a cada 10 segundos o Bafomé tem 5,5% de
//=     chance de ativar uma invencibilidade;
//=   - se os Noviços renascerem e o MVP ainda estiver vivo, cada
//=     morte de Noviço aumenta essa chance, até 17%;
//=   - a invencibilidade dura 10 segundos, e ao fim deles há
//=     chance de continuar ou de cair. Quanto maior a chance de
//=     ativação, menor a de cair.
//
//= O QUE A PÁGINA NÃO DIZ, E FOI DECIDIDO AQUI
//
//= Três coisas, e é honesto que estejam separadas do resto:
//
//=   1. o PASSO da chance. A página dá as duas pontas (5,5% e
//=      17%) e o gatilho (morte de Noviço), mas não o degrau.
//=      São quatro Noviços, então o degrau é (170-55)/4 = 28,75
//=      décimos de por cento, arredondado para 29 - com os
//=      quatro mortos dá 171, que o teto corta em 170;
//=   2. a CHANCE DE CAIR ao fim dos 10 segundos. A página só diz
//=      que ela é menor quanto maior a de ativar. Aqui ela é o
//=      complemento: cai com 100% menos a chance de ativação, ou
//=      seja 94,5% no começo e 83% com os quatro mortos;
//=   3. ONDE o MVP nasce. A página não diz. Nasce em 99,103, o
//=      centro do labirinto - que é a mesma sala onde Andrea
//=      caiu -, e o mapa inteiro é avisado quando isso acontece.
//
//= (Havia uma quarta: o SPRITE dos quatro Restos, que era
//= `4_GHOST_STAND` por escolha nossa, pelo texto da página -
//= "ajude seus espíritos voltarem para casa". O dono passou os
//= `_D` oficiais em 2026-09-12 e ela saiu daqui: hoje TODOS os
//= sprites desta missão vêm dele, dos prints do bRO.)
//
//= OS SPRITES, E O QUE ELES CUSTAM DO LADO DO CLIENTE
//
//= A Irene é `4_F_ERENE` (view 10337) e as duas Esmeraldas são
//= `4_F_08` (view 10095) - os dois com constante no npc.hpp, no
//= npcidentity.lub deste cliente E com `.spr`/`.act` já em
//= `data\sprite\npc\` do nosso GRF. Não custam nada.
//
//= Os quatro recrutas e os quatro Restos usam os SPRITES DE NPC
//= oficiais da missão, e não mais os view ids do mob_db. Os
//= oito pares vieram do dono em 2026-09-12:
//
//=     Andrea   10310  4_M_ANDREA    Restos  10311  ..._D
//=     Agnes    10312  4_F_ANES      Restos  10313  ..._D
//=     Silvano  10314  4_M_SILVANO   Restos  10315  ..._D
//=     Cecília  10316  4_F_CECILIA   Restos  10317  ..._D
//
//= Os `_D` são a versão CAÍDA de cada um - é o que os Restos
//= desenham agora, no lugar do `4_GHOST_STAND` genérico que
//= estava aqui antes.
//
//= E isso ZEROU o custo de cliente desta caverna: os oito
//= `.spr`/`.act` já estão em `data\sprite\npc\` do nosso
//= data.grf, sem cifra - conferido em 2026-09-12 com
//= `grf.py ... find`. Enquanto os recrutas usavam o view id do
//= MONSTRO (20521..20524) a arte só existia na pasta de
//= monstro, e os oito `ill_*` copiados à mão para
//= `cliente\data\sprite\npc\` precisavam de PATCH para chegar
//= ao jogador - era a armadilha dos Agentes da Liga
//= (ARMADILHAS-CLIENTE.md). Com o sprite de NPC oficial a
//= dependência acabou, e os `ill_*` soltos ficaram órfãos.
//
//= O nome na tela perdeu o "Recruta " no mesmo dia, pedido do
//= dono: são "Andrea", "Agnes", "Silvano" e "Cecília". O nome
//= ÚNICO - o que vem depois do `::` - não mudou: é ele que o
//= `cloakonnpc` usa, e mexer nele quebraria os quatro
//= `cloakoffnpc` do passo 6.
//
//= POR QUE A INVENCIBILIDADE É `sc_start` E NÃO A HABILIDADE
//
//= O NPC_INVINCIBLE do rAthena existe, mas o skill_db lhe dá
//= `Duration1: 300000` - cinco minutos, e o que se quer são dez
//= segundos. O `sc_start` aceita GID (doc/script_commands.txt,
//= *sc_start) e permite dizer a duração exata.
//
//= E ela funciona: quem tem SC_INVINCIBLE cai no
//= `is_infinite_defense` (src/map/battle.cpp:2940), que é o
//= mesmo caminho das plantas - todo golpe entrega 1 de dano.
//= Não depende do `invincible.nodamage` do battle conf, que
//= está `no` e é outra coisa (aquele trata o caminho do
//= status_damage).
//
//= POR QUE A INVENCIBILIDADE FALA
//
//= Porque o cliente de RO não desenha barra de vida de monstro,
//= e 1 de dano por golpe lê-se como defeito e não como
//= mecânica. É a regra 4.21 do CLAUDE.md: número que o jogador
//= não enxerga precisa de retorno visual como PARTE da
//= mecânica. Por isso o Bafomé anuncia quando fica intocável e
//= quando deixa de estar, e o mapa inteiro é avisado a cada
//= Noviço que cai com ele vivo - que é o que explica ao jogador
//= por que ele está ficando mais duro.
//
//= O RELÓGIO É UM SÓ
//
//= Um `OnTimer10000` cuida de tudo: renascimento dos quatro,
//= reposição depois do MVP e a rolagem da invencibilidade. O
//= alternativo seria um temporizador por Noviço, e são quatro
//= prazos independentes de uma hora - com um relógio só, o
//= prazo vira comparação de `gettimetick(2)`, que sobrevive a
//= `@reloadscript` mal e a leitura de código bem.
//
//===========================================================

function	script	F_IlusaoLabResto	{
	.@bit = getarg(0);
	.@nome$ = getarg(1);

	if (ilusao_labirinto != 3) {
		mes "^4D4DFFUma luz parada, do tamanho de alguém ajoelhado. Ela não responde a você.^000000";
		close;
	}
	if (ilusao_labirinto_restos & .@bit) {
		mes "^4D4DFFVocê já recolheu o que havia aqui.^000000";
		close;
	}
	mes "^4D4DFFA luz tem a forma de alguém ajoelhado, de costas. Quando você chega perto, ela não foge - vira o que seria o rosto.^000000";
	next;
	mes "[" + .@nome$ + "]";
	mes "...a senhora veio?";
	next;
	mes "^4D4DFFRecolher isto leva alguns segundos, e é preciso não sofrer dano nenhum enquanto durar.^000000";
	next;
	if (select("Recolher", "Agora não") == 2)
		close;
	close2;
	progressbar "000000", 5;

	// se o jogador levar dano ou andar, o progressbar aborta e o script morre
	// aqui mesmo - o bit não é marcado, e ele recomeça
	ilusao_labirinto_restos = ilusao_labirinto_restos | .@bit;
	specialeffect EF_EXIT;

	while (1) {
		switch (select("Atravessar a fenda", "O que há do outro lado?", "Ficar aqui")) {

			mes "[Fenda Retorcida]";

			next;
			mes "[Fenda Retorcida]";
			mes "^4D4DFFNeste momento:^000000";
			.@vivos = 0;
			for (.@i = 0; .@i < 4; .@i++)
				if (.NovicoGid[.@i] > 0) .@vivos++;
			if (.MvpGid > 0)
				mes "- o ^FF0000Bafomé Caótico^000000 está no centro do labirinto.";

OnInit:
	.Mapa$ = "prt_mz03_i";

	setarray .NovicoNome$[0], "Andrea", "Agnes", "Silvano", "Cecília";

	.MvpId = 20520;
	.MvpX = 99;
	.MvpY = 103;

	.Renasce = 3600;        // uma hora, em segundos
	.Duracao = 10000;       // a invencibilidade, em ms
	.ChanceBase = 55;       // 5,5% em décimos de por cento
	.ChancePasso = 29;      // cada Noviço morto com o MVP vivo
	.ChanceTeto = 170;      // 17%

	// A Fenda, os quatro recrutas e os quatro Restos nascem escondidos de
	// TODO MUNDO. Quem os revela, por jogador, é o vigia lá embaixo.
	cloakonnpc "FendaRetorcida";

	cloakonnpc "Restos de Silvano#ilusaolab";
	cloakonnpc "RestosCeciliaIlusaoLab";

	for (.@i = 0; .@i < 4; .@i++)
		callsub S_NasceNovico, .@i;
	initnpctimer;
	end;

// O relógio único. Dez segundos é o passo da invencibilidade, e o
// renascimento de uma hora só precisa de um prazo comparado.
OnTimer10000:
	// A sonda do lado do servidor, uma vez por subida do servidor. O `mobcount`
	// com rótulo VAZIO conta quem nasceu por linha de spawn (os quatro Noviços
	// têm rótulo e ficam de fora), e é a única medição que responde sem cliente
	// - CLAUDE.md seção 4.21. Ela fica no primeiro tique e não no OnInit porque
	// no OnInit o mapa ainda está nascendo, e lá o número sai pela metade.
	// E `debugmes` e não `ShowInfo`: só o primeiro chega ao log, e mesmo assim
	// só com o bit 4 do `console_msg_log` (nesta máquina ele é 3, então o lugar
	// de ler isto é a JANELA do map-server).
	if (.Sondou == 0) {
		.Sondou = 1;
		debugmes "Ilusao do Labirinto: " + mobcount(.Mapa$, "") + " monstros comuns no mapa, Novicos nos gids " + .NovicoGid[0] + "/" + .NovicoGid[1] + "/" + .NovicoGid[2] + "/" + .NovicoGid[3] + ".";
	}
	.@agora = gettimetick(2);
	for (.@i = 0; .@i < 4; .@i++) {
		if (.NovicoGid[.@i] > 0) continue;
		if (.NovicoRenasce[.@i] == 0) continue;
		if (.@agora < .NovicoRenasce[.@i]) continue;
		callsub S_NasceNovico, .@i;
	}
	// uma hora depois da queda do MVP, os quatro voltam de uma vez
	if (.MvpGid == 0 && .MvpMorreu > 0 && .@agora >= .MvpMorreu + .Renasce) {
		.MvpMorreu = 0;
		for (.@i = 0; .@i < 4; .@i++)
			if (.NovicoGid[.@i] == 0) callsub S_NasceNovico, .@i;
	}
	if (.MvpGid > 0) callsub S_Invencibilidade;
	setnpctimer 0;
	end;

OnNovicoMorre:
	// `killedrid` é a CLASSE do monstro, que é o que identifica qual dos
	// quatro caiu - o `killedgid` seria o da instância que acabou de morrer
	.@qual = -1;
	for (.@i = 0; .@i < 4; .@i++)
		if (.NovicoId[.@i] == killedrid) .@qual = .@i;
	if (.@qual < 0) end;

	.NovicoGid[.@qual] = 0;
	.NovicoRenasce[.@qual] = gettimetick(2) + .Renasce;

	if (.MvpGid > 0) {
		// morto com o MVP no mapa: é isto que endurece o Bafomé, e o
		// jogador precisa ouvir, senão a mudança fica invisível
		if (.Mortos < 4) .Mortos++;
		mapannounce .Mapa$, "O que restava de " + .NovicoNome$[.@qual] + " se desfaz - e o Bafomé Caótico fica mais difícil de ferir.", bc_map;
		end;
	}

	.@caidos = 0;
	for (.@i = 0; .@i < 4; .@i++)
		if (.NovicoGid[.@i] == 0) .@caidos++;
	if (.@caidos == 4) callsub S_NasceMvp;
	end;

OnMvpMorre:
	.MvpGid = 0;
	.Invencivel = 0;
	.Mortos = 0;
	.MvpMorreu = gettimetick(2);
	mapannounce .Mapa$, "O Bafomé Caótico foi derrotado. Em uma hora os quatro recrutas voltarão a vagar pelo labirinto.", bc_map;
	debugmes "Ilusao do Labirinto: Bafome Caotico derrotado.";
	end;

S_NasceNovico:
	// `callsub` abre escopo `.@` novo e VAZIO - as variáveis de quem chamou
	// não chegam aqui, e por isso o índice vem por `getarg` (CLAUDE.md §5)
	.@i = getarg(0);
	monster .Mapa$, .NovicoX[.@i], .NovicoY[.@i], "--ja--", .NovicoId[.@i], 1, "FendaRetorcida::OnNovicoMorre";
	.NovicoGid[.@i] = $@mobid[0];
	.NovicoRenasce[.@i] = 0;
	return;

S_NasceMvp:
	monster .Mapa$, .MvpX, .MvpY, "--ja--", .MvpId, 1, "FendaRetorcida::OnMvpMorre";
	.MvpGid = $@mobid[0];
	.Mortos = 0;
	.Invencivel = 0;
	mapannounce .Mapa$, "Os quatro recrutas caíram. O Bafomé Caótico surge no centro do labirinto, em " + .MvpX + "," + .MvpY + ".", bc_map;
	debugmes "Ilusao do Labirinto: Bafome Caotico invocado.";
	return;

S_Invencibilidade:
	.@chance = .ChanceBase + .ChancePasso * .Mortos;
	if (.@chance > .ChanceTeto) .@chance = .ChanceTeto;

	if (.Invencivel == 0) {
		if (rand(1000) < .@chance) {
			sc_start SC_INVINCIBLE, .Duracao, 1, 10000, 0, .MvpGid;
			.Invencivel = 1;
			unittalk .MvpGid, "Nada do que vocês têm me alcança agora.";
			specialeffect EF_ENERGYCOAT, AREA, .MvpGid;
		}
		return;
	}

	// já está intocável: quanto maior a chance de ativar, menor a de cair
	if (rand(1000) < 1000 - .@chance) {
		sc_end SC_INVINCIBLE, .MvpGid;
		.Invencivel = 0;
		unittalk .MvpGid, "A casca cede... por enquanto.";
		specialeffect EF_SUI_EXPLOSION, AREA, .MvpGid;
		return;
	}
	// continua: renova por mais dez segundos, senão o status vence sozinho
	sc_start SC_INVINCIBLE, .Duracao, 1, 10000, 0, .MvpGid;
	return;
}
