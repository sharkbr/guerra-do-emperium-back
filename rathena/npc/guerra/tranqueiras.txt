//= Sprite 5 - `JOB_MERCHANT`, o Mercador -, pedido assim em

//= FUNCIONA, e o caminho e explicito no codigo: o
//= `status_set_viewdata` (src/map/status.cpp, case BL_NPC) tenta o

//= que existe nos dois sexos no nosso data.grf (conferido).

//= `look[LOOK_HAIR] = cap_value(0, MIN_HAIR_STYLE, MAX_HAIR_STYLE)`, e

//= Gravado em Latin-1 (ISO-8859-1), como todo texto que o jogo le.

prontera,151,131,4	script	Tranqueiras	JOB_MERCHANT,{
	callshop "Tranqueiras#loja",0;
	end;

OnInit:
	// Aparencia. As duas linhas existem porque o sprite e uma CLASSE de
	// jogador e nao um view id de NPC - ver a secao "O sprite e uma
	// CLASSE DE JOGADOR" no cabecalho. O ramo do rAthena que monta essa
	// aparencia deixa o penteado em ZERO, e este cliente so tem de 1 a
	// 42: sem a segunda linha o NPC pede um sprite de cabeca que nao
	// existe. Trocando o sprite por um view id de NPC, as duas saem.
	setunitdata(getnpcid(0), UNPC_SEX, SEX_MALE);
	setunitdata(getnpcid(0), UNPC_HAIRSTYLE, 1);

	// O assunto, e nao a contagem como nas lojas de carta: o nome
	// "Tranqueiras" nao diz o que esta a venda.
	placadevenda "Materiais de runa e veneno";
	end;
}
