//= O teste que dispara esse aviso (src/map/npc.cpp:4153) e
//= `value*0.75 < value_sell*1.24` - a compra mais barata possivel,
//= com Descontar nivel 10, contra a venda mais cara possivel, com

//= `Sell` exatamente igual a `Buy/2`, da `0,75*Buy` contra

//===== O sprite e uma CLASSE DE JOGADOR ======================
//= Sprite 5 - `JOB_MERCHANT`, o Mercador -, pedido assim em
//= 2026-08-12. Isso e incomum e merece as tres linhas abaixo, porque
//= NENHUM NPC do rAthena inteiro faz isso: os 26 mil `script` do
//= vendor usam view id de NPC (>= 44), nunca id de classe.
//=
//= FUNCIONA, e o caminho e explicito no codigo: o
//= `status_set_viewdata` (src/map/status.cpp, case BL_NPC) tenta o
//= `npc_get_viewdata` primeiro, que devolve nulo para 5 - o
//= `npcdb_checkid` so aceita de 44 para cima -, e cai num ramo
//= `else if (pcdb_checkid(class_))` que monta a aparencia a mao:
//= `look[LOOK_BASE] = 5`. Dai o cliente desenha o corpo de Mercador
//= do jogador, `data\sprite\<coreano>\<corpo>\<sexo>\<mercador>`,
//= que existe nos dois sexos no nosso data.grf (conferido).
//=
//= MAS AQUELE RAMO DEIXA A CABECA EM ZERO, e zero nao existe. Ele faz
//= `look[LOOK_HAIR] = cap_value(0, MIN_HAIR_STYLE, MAX_HAIR_STYLE)`, e
//= o nosso `min_hair_style` (conf/battle/client.conf) e 0 - entao o
//= NPC nasceria pedindo o penteado 0. Os penteados deste cliente vao
//= de 1 a 42, nos dois sexos: nao ha `0_<sexo>.spr`. Por isso as duas
//= linhas de `setunitdata` no OnInit, e elas NAO sao enfeite.
//=
//= As duas gravam no `nd->vd` do proprio NPC (clif_changelook, case
//= LOOK_HAIR, faz `vd->look[type] = val`), entao valem para quem logar
//= depois - nao sao um pacote solto para quem estava por perto.
//=
//= SE MESMO ASSIM ELE NASCER INVISIVEL ou sem cabeca, a saida ja esta
//= escolhida: trocar o `JOB_MERCHANT` da linha do NPC por 776
//= (`4_M_TWMIDMAN`), que e sprite de NPC de verdade - conferido no
//= `npcidentity.lub` DESTE cliente e com `.spr` e `.act` em
//= `data\sprite\npc\`. As duas linhas de `setunitdata` saem junto,
//= porque view id de NPC traz a aparencia pronta do `npc_viewdb`.
//===== Notas ================================================
//= A loja e FLUTUANTE (`-` no lugar do mapa) e o NPC a chama com
//= `callshop`. E o idioma dos tres mercados, e o motivo e o mesmo: um

//= placa. Ver npc/guerra/mercado_contemporaneo.txt.

//= A PLACA sobre a cabeca e a mesma dos 21 mercados
//= (src/custom/placa_de_venda). Aqui ela ganha o assunto e nao a
//= contagem, porque "Tranqueiras" nao diz o que se vende.

//= OS ITENS FORAM VALIDADOS UM A UM antes de entrar (CLAUDE.md 4.4),
//= em duas rodadas, as duas em 2026-08-12:
//=
//=   - a primeira, dos 25 de runa e veneno, conferiu 26 - o Ouro
//=     passou por ela, saiu horas depois por economia e voltou na

//= UMA RESSALVA de nome, e ela nao quebra nada: o 7030 se chama
//= "Garra de Lobo do Deserto" no servidor e "Garra de Lobo do Deserto "

//=
//= Gravado em Latin-1 (ISO-8859-1), como todo texto que o jogo le.
//= Ver CLAUDE.md secao 4.1. Editar com script que grave em cp1252 - a
//= ferramenta de edicao do assistente troca TODO byte acentuado do
//= arquivo por U+FFFD, e sem avisar.

//= Mudanca neste arquivo pega com @reloadscript.
//============================================================

prontera,151,131,4	script	Tranqueiras	JOB_MERCHANT,{
	callshop "Tranqueiras#loja",0;
	end;

OnInit:
	// Aparencia. As duas linhas existem porque o sprite e uma CLASSE de
	// jogador e nao um view id de NPC - ver a secao "O sprite e uma
	// CLASSE DE JOGADOR" no cabecalho. O ramo do rAthena que monta essa
	// aparencia deixa o penteado em ZERO, e este cliente so tem de 1 a
	// 42: sem a segunda linha o NPC pede um sprite de cabeca que nao
	// existe. Trocando o sprite por um view id de NPC, as duas saem.
	setunitdata(getnpcid(0), UNPC_SEX, SEX_MALE);
	setunitdata(getnpcid(0), UNPC_HAIRSTYLE, 1);

	// O assunto, e nao a contagem como nas lojas de carta: o nome
	// "Tranqueiras" nao diz o que esta a venda. Cresceu junto com a
	// vitrine em 2026-08-12, quando a alquimia entrou. Teto de 79
	// caracteres (MESSAGE_SIZE, src/map/map.hpp:273); esta em 36.
	placadevenda "Materiais de runa, veneno e alquimia";
	end;
}
