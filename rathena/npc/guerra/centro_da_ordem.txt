// ---------------------------------------------------------------
// A montagem: desliga o leilao que o rAthena deixou no mapa

// NPC flutuante, sem corpo e sem mapa - existe so pelo OnInit.
// Ver a secao "O que este arquivo desliga do rAthena", no
// cabecalho, para o porque de cada um dos onze.
-	script	#centro_ordem_montagem	-1,{
	end;

OnInit:
	setarray .@alvo$[0],
		// Os dois warps do leilao. O primeiro e substituido pelo
		// `centro_ordem_saida` acima, na mesma celula; o segundo sela o
		// bloco oeste.
		"auction_entrance_moc",  "auction_entrance_prt",
		// As duas portas de fora que ainda apontavam para ca.
		"Auction Hall Guide#moc", "Information Post#moc",
		"Auction Hall Guide#prt", "Information Post#prt",
		// Os sete corretores. O leilao esta desligado no servidor
		// (`feature.auction: off`), entao nao fazem nada - so
		// ocupariam a casa.
		"Auction Broker#moc1", "Auction Broker#moc2",
		"Auction Broker#moc3", "Auction Broker#moc4",
		"Auction Broker#prt1", "Auction Broker#prt2",
		"Auction Broker#prt3";

	.@sumiram = 0;
	for (.@i = 0; .@i < getarraysize(.@alvo$); .@i++) {
		// getnpcid devolve 0 quando o NPC nao existe. Sem este
		// teste, um nome fora de sincronia abortaria o OnInit e
		// deixaria os alvos seguintes de pe.
		if (!getnpcid(0, .@alvo$[.@i])) {
			.@sumiram++;
			continue;
		}
		disablenpc .@alvo$[.@i];
	}

	if (.@sumiram)
		debugmes "CentroDaOrdem: "+.@sumiram+" de "+getarraysize(.@alvo$)+" alvos nao foram encontrados. A lista deste arquivo saiu de sincronia com npc/other/auction.txt - PODE HAVER NPC DE LEILAO DE PE NO SALAO.";

	// -----------------------------------------------------------
	// As celulas da escrivaninha
	// -----------------------------------------------------------
	// O modelo 3D da mesa esta no .rsw (ver o cabecalho), e MODELO
	// NAO BLOQUEIA PASSAGEM: o override de .rsw nao toca o .gat, e
	// o jogador atravessa a mesa. Quem fecha as celulas e este
	// bloco, no servidor.
	//
	// As cinco celulas foram DADAS pelo dono, andando em cima da
	// mesa (2026-08-11). Elas nao sao a pegada geometrica do
	// modelo, que mede 4,0 x 2,8 celulas e cobre x191-192 em
	// y70-73 (oito celulas): faltam 192,70, 192,71 e 192,72. Fica
	// registrado porque a diferenca e visivel em jogo - da para
	// entrar na metade leste do tampo. Acrescentar e por os tres
	// pares no array abaixo.
	//
	// E SETWALL, NAO SETCELL, e a diferenca importa. O `setcell`
	// muda a celula so do lado do servidor: o cliente continua
	// achando que da para andar ali, e o proprio doc do rAthena
	// avisa que isso "may cause movement problems". O `setwall`
	// faz a mesma coisa e ainda manda `clif_changemapcell`
	// (`map.cpp:3509`) - e o `map_iwall_get` reenvia para quem
	// entrar no mapa depois (`clif.cpp:11098`). As duas metades
	// ficam de acordo.
	//
	// UMA CELULA POR PAREDE, de proposito. O `setwall` aceita
	// tamanho, mas o `map_iwall_set` PARA na primeira celula ja
	// bloqueada e grava `iwall->size = i` sem reclamar
	// (`map.cpp:3503`) - parede mais curta que a pedida, calada.
	// Com tamanho 1 nao ha o que truncar, e o array vira a unica
	// fonte tanto do que se fecha quanto do que se confere.
	//
	// `shootable` = true: fecha o andar e nao a linha de tiro, que
	// e o que uma mesa faz. E nao ha combate no salao.
	setarray .@mesa[0],
		191,70,  191,71,  191,72,  191,73,  192,73;

	.@abertas = 0;
	for (.@i = 0; .@i < getarraysize(.@mesa); .@i += 2) {
		.@nome$ = "centro_ordem_mesa"+(.@i / 2);

		// O `setwall` RECUSA nome que ja existe e devolve falha
		// calada, entao no @reloadscript o segundo OnInit nao
		// faria nada. O `delwall` antes torna o bloco idempotente.
		// Ele devolve a celula para andavel+atiravel sem consultar
		// o mapa original (`map.cpp:3553`) - aqui isso e inofensivo
		// porque as cinco sao chao andavel de verdade, mas nao
		// vale para celula que ja nascesse bloqueada.
		if (checkwall(.@nome$))
			delwall .@nome$;

		setwall "auction_01",.@mesa[.@i],.@mesa[.@i+1],1,DIR_NORTH,true,.@nome$;

		// A prova. Sem ela, parede que nao pegou some no silencio.
		if (!checkcell("auction_01",.@mesa[.@i],.@mesa[.@i+1],CELL_CHKNOPASS))
			.@abertas++;
	}

	if (.@abertas)
		debugmes "CentroDaOrdem: "+.@abertas+" de "+(getarraysize(.@mesa)/2)+" celulas da escrivaninha continuam andaveis depois do setwall - A MESA ESTA ATRAVESSAVEL.";
	end;
}
