auction_02,59,34,4	script	Missões A#ordem_a	837,{
	// Qual placa e esta: sai do pedaco depois do `#` do nome. As
	// tres dividem este script e as tabelas do OnInit, entao este
	// e o unico jeito de uma duplicata saber quem e.
	.@g$ = strnpcinfo(2);
	if (.@g$ == "ordem_a")      .@g = 0;
	else if (.@g$ == "ordem_b") .@g = 1;
	else                        .@g = 2;

	.@ini = .ini[.@g];
	.@fim = .fim[.@g];
	.@esp = .espera[.@g];

	mes "[" + strnpcinfo(1) + "]";
	mes "Um quadro de cortica tomado de recortes,";
	mes "cada um com o carimbo da ^0000FFOrdem dos Exploradores^000000.";
	mes "Sao pedidos de caçada - todos dentro de memórias,";
	mes "e todos pagos em ^0000FFMoeda do Explorador^000000.";
	next;

	// O estado de cada missao do lote. checkquest(...,HUNTING):
	//   -1 nunca pegou   0 em andamento   2 pronta
	.@nova = 0; .@pronta = 0; .@andando = 0;
	mes "[" + strnpcinfo(1) + "]";
	for (.@i = .@ini; .@i <= .@fim; .@i++) {
		.@e = checkquest(.quest[.@i], HUNTING);
		if (.@e == -1) {
			mes " ^777777" + .nome$[.@i] + "^000000";
			.@nova++;
		} else if (.@e == 2) {
			mes " ^0000FF" + .nome$[.@i] + "^000000  ^0000FF(pronta)^000000";
			.@pronta++;
		} else {
			mes " " + .nome$[.@i] + "  ^777777(em andamento)^000000";
			.@andando++;
		}
	}
	next;

	mes "[" + strnpcinfo(1) + "]";
	mes "Você tem ^0000FF" + countitem(.moeda) + "^000000 " +
	    ((countitem(.moeda) == 1) ? "Moeda do Explorador" :
	                                "Moedas do Explorador") + ".";
	next;

	switch (select("Pegar os pedidos que faltam:" +
	               "Prestar contas das caçadas prontas:" +
	               "Só estava olhando")) {

	// ----- pegar --------------------------------------------------
	case 1:
		if (.@nova == 0) {
			mes "[" + strnpcinfo(1) + "]";
			mes "Você já está com todos os pedidos deste quadro.";
			mes "Não há recorte novo para levar.";
			close;
		}
		// So o que ainda nao foi pego: `setquest` de quest repetida
		// deixa aviso no log e nao acrescenta nada.
		for (.@i = .@ini; .@i <= .@fim; .@i++)
			if (checkquest(.quest[.@i], HAVEQUEST) == -1)
				setquest .quest[.@i];

		mes "[" + strnpcinfo(1) + "]";
		mes "Você copia ^0000FF" + .@nova + "^000000 " +
		    ((.@nova == 1) ? "pedido" : "pedidos") + " para o seu diário.";
		mes "A Ordem não cobra prazo: vá quando puder,";
		mes "e volte quando tiver o que contar.";
		close;

	// ----- entregar -----------------------------------------------
	case 2:
		if (.@pronta == 0) {
			mes "[" + strnpcinfo(1) + "]";
			if (.@andando > 0) {
				mes "Nenhuma caçada fechada ainda.";
				mes "A Ordem paga por serviço terminado.";
			} else {
				mes "Você não tem o que prestar contas.";
				mes "Leve os pedidos primeiro.";
			}
			close;
		}

		// A trava do dia. PLAYTIME devolve 0 enquanto nao deu 6h;
		// -1 (nunca pegou) e 2 (ja passou) liberam.
		if (checkquest(.@esp, PLAYTIME) == 0) {
			mes "[" + strnpcinfo(1) + "]";
			mes "O quadro já foi fechado hoje.";
			mes "A Ordem só reabre a escrita às ^0000FFseis da manhã^000000 -";
			mes "é quando o escrivão chega.";
			close;
		}

		.@moedas = 0;
		.@ganho = 0;
		for (.@i = .@ini; .@i <= .@fim; .@i++)
			if (checkquest(.quest[.@i], HUNTING) == 2) {
				.@moedas += .paga[.@i];
				.@ganho  += .exp[.@i];
			}

		// bolsa cheia o getitem a PERDERIA em vez de larga-la no
		// chao, e a caçada teria sido de graca.

			mes "[" + strnpcinfo(1) + "]";
			mes "O escrivão olha a sua bolsa e balança a cabeça.";

			    ((.@moedas == 1) ? "moeda" : "moedas") + ".";
			mes "Abra espaço e volte - o pedido continua seu.";
			close;
		}

		getitem .moeda, .@moedas;
		getexp .@ganho, .@ganho;

		// Apaga e repoe na mesma passada: e o que faz o jogador nao
		// precisar voltar amanha so para pegar de novo.
		for (.@i = .@ini; .@i <= .@fim; .@i++)
			if (checkquest(.quest[.@i], HUNTING) == 2) {
				erasequest .quest[.@i];
				setquest .quest[.@i];
			}

		// A espera. Apagar antes de por de novo e o que reinicia o
		// prazo para o proximo 06:00.
		if (checkquest(.@esp, HAVEQUEST) != -1)
			erasequest .@esp;
		setquest .@esp;

		mes "[" + strnpcinfo(1) + "]";
		mes "^0000FF" + .@pronta + "^000000 " +
		    ((.@pronta == 1) ? "caçada conferida" : "caçadas conferidas") +
		    " e assinada" + ((.@pronta == 1) ? "" : "s") + ".";
		mes "A Ordem paga ^0000FF" + .@moedas + "^000000 " +
		    ((.@moedas == 1) ? "Moeda do Explorador" :
		                       "Moedas do Explorador") + ".";
		next;
		mes "[" + strnpcinfo(1) + "]";
		mes "Os recortes já foram trocados pelos de amanhã.";
		mes "Pode ir caçando; a escrita reabre às seis.";
		close;

	case 3:
		mes "[" + strnpcinfo(1) + "]";
		mes "O quadro fica. A Ordem também.";
		close;
	}
	end;
