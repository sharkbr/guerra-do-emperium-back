//===== Guerra do Emperium ===================================
//= Honra de Combate - a reputação de PvP da Arena de Combate
//===== Onde =================================================
//= prontera 142,180 - a placa do placar, CINCO células a oeste
//= da porta da arena (147,180) e na mesma fileira que ela.
//=
//= ELA ANDOU DUAS VEZES:
//=
//=   até 2026-08-08   prontera 163,187
//=   2026-08-08       prontera 152,187, a ponta oeste da fileira
//=                    do porteiro do campo de treino, no mesmo
//=                    pedido que trocou a arena e o campo de
//=                    treino de lugar
//=   2026-08-13       prontera 142,180, a pedido, no mesmo dia
//=                    em que a pontuação perdeu o anti-conluio
//=
//= A célula de hoje foi conferida no .gat DESTE cliente: tipo 0,
//= andável, e sem NPC ligado a menos de quatro células. O mais
//= perto é o Clan Helper do rAthena (npc/re/other/clans.txt),
//= em 138,183. O Lottery e o Stock Market de npc/custom/, que
//= ficariam em 141,182 e 140,181, estão os dois comentados no
//= scripts_custom.conf e não nascem.
//===== Descrição ============================================
//= Matar na arena dá um ponto, morrer tira um. A conta aparece
//= no modal de reputação do cliente (Honra de Combate) e o
//= ranking sai nesta placa.
//=
//= Pedido em 2026-08-07.
//===== Por que o ranking NÃO está no modal ==================
//= Porque o modal não consegue, e não adianta insistir. O
//= pacote ZC_REPUTE_INFO (src/map/clif.cpp) carrega só `type` e
//= `points` - dois inteiros por linha, nenhum campo de texto. O
//= nome de cada linha vem de um arquivo do CLIENTE
//= (data\contentdata\repute\reputeinfodata.bson), igual para
//= todo mundo e lido só na inicialização.
//=
//= Ou seja: o modal responde "quanto EU tenho em cada
//= categoria". Um ranking é a matriz transposta - "quem tem
//= quanto" -, e para o servidor mandar o nome "Fulano" não há
//= por onde. Por isso a placa existe.
//===== Onde a pontuação mora ================================
//= EM DOIS LUGARES, de propósito, e a tabela é quem manda:
//=
//=   guerra_pvp_placar  (SQL)  - a verdade. Escrita no instante
//=       da morte, então a placa nunca mostra número velho.
//=   #RepPointsPvP      (conta) - a cópia que o modal lê.
//=
//= A duplicação não é descuido, é o preço de ter as duas
//= coisas. A variável de conta só chega ao banco quando o
//= rAthena salva o personagem - autosave de 300s ou logout -, e
//= `chrif_save` NÃO é chamado ao trocar de mapa dentro do mesmo
//= map-server: sair da arena andando ou de Asa de Borboleta não
//= salva. Um placar lido de acc_reg_num mostraria até 5 minutos
//= de atraso.
//=
//= O OnPCLoginEvent resincroniza a variável a partir da tabela.
//= É isso que impede as duas de desencontrarem se o map-server
//= cair entre um autosave e outro.
//=
//= O `#` do Variable (db/guerra/reputation.yml) é o que faz a
//= pontuação valer PARA A CONTA: pc_setreg2 despacha pelo
//= prefixo, e `#` cai em pc_setaccountreg.
//===== Por que set_reputation_points e não `#RepPointsPvP = ` =
//= Porque o buildin chama `clif_reputation_type` no fim
//= (src/map/script.cpp), que empurra a linha para o cliente na
//= hora. Atribuir a variável na mão atualizaria o banco e
//= deixaria a tela velha até o jogador trocar de mapa - a lista
//= completa só é enviada no clif_parse_LoadEndAck.
//===== As regras ============================================
//= SÃO DUAS, e desde 2026-08-30 é tudo o que existe:
//= 1) Só na arena (pvp_n_1-5). Fora dela o evento sai na
//=    primeira linha.
//= 2) Matador e morto precisam ser nível .Nivel. Fecha a porta
//=    para personagem descartável criado só para dar pontos, e
//=    vale para o PONTO e para a CAVEIRA de uma vez.
//=
//= Passada a (2): quem mata ganha um ponto e recolhe a Caveira
//= Humana, quem morre perde um ponto. Não há mais nenhuma
//= condição sobre a pontuação de quem morreu.
//=
//= O QUE SAIU EM 2026-08-30, a pedido do dono ("vamos derrubar
//= a maioria das regras"), e as duas eram de 2026-08-08:
//=   - o .Piso: o matador só pontuava se o morto tivesse zero
//=     ou mais. Derrubar quem já estava negativo não rendia
//=     nada a ninguém; hoje rende o ponto e a caveira.
//=   - o .TrofeuPiso: a Caveira Humana só saía se o morto
//=     tivesse 1 ponto ou mais antes de morrer. Hoje sai em
//=     toda morte válida.
//= As duas variáveis foram APAGADAS do OnInit, e não zeradas:
//= regra que não existe mais não deve continuar configurável.
//=
//= O QUE FICOU do lado do morto: ele perde um ponto em toda
//= morte válida, tenha ele quanto tiver - até .PisoQueda (-10),
//= de onde não desce mais. Este piso foi mantido de propósito
//= no mesmo pedido; é o único número que ainda limita alguma
//= coisa aqui.
//=
//= O PISO DE QUEDA NÃO É ESCOLHA DESTE ARQUIVO SOZINHO: o
//= `Minimum: -10` de db/guerra/reputation.yml já trava o modal
//= no mesmo número (o set_reputation_points passa por um
//= cap_value, src/map/script.cpp). Sem o piso aqui, a TABELA
//= desceria abaixo de -10 e o modal ficaria parado em -10 - as
//= duas metades da pontuação discordando, caladas, e a placa
//= mostrando um número que o jogador nunca vê. Mexer num dos
//= dois exige mexer no outro.
//===== NÃO HÁ MAIS ANTI-CONLUIO ==============================
//= Retirado em 2026-08-13, a pedido. Até ali havia um saldo por
//= PAR DE CONTAS por DIA (tabela guerra_pvp_confronto, preso em
//= mais ou menos .Limite = 2): revezamento honesto pontuava o
//= dia inteiro, fazenda de um lado só travava no terceiro
//= golpe.
//=
//= Saiu INTEIRO: não há teto por par, não há janela de 24
//= horas, não há espera entre uma morte e a seguinte. Toda
//= morte válida vale, quantas vezes acontecer.
//=
//= ISSO ABRE A PORTA PARA CONTA DESCARTÁVEL, e o dono sabe: o
//= pedido veio com "por enquanto as pessoas podem ficar criando
//= char ou até conta pra isso, mas vamos observar primeiro". O
//= nível .Nivel dos dois lados é o único custo que sobrou, e
//= não é pequeno - mas não é trava.
//=
//= E DESDE 2026-08-30 O NÍVEL É O ÚNICO CUSTO QUE EXISTE. Até
//= aquele dia a economia tinha um segundo freio, que era a
//= regra do .Piso: um alvo só valia ponto enquanto estivesse em
//= zero ou acima, e cada morte o empurrava para baixo, então
//= uma conta descartável rendia UMA caveira e UM ponto e depois
//= parava. Com o .Piso fora, a mesma conta rende uma caveira
//= POR MORTE, sem fim. Foi pedido assim, sabendo disso; o piso
//= de queda (-10) segue o morto para baixo mas não impede mais
//= nada.
//=
//= A tabela guerra_pvp_confronto continua NO BANCO, sem ninguém
//= escrever nela. Foi tirada do sql-files/guerra_arena_pvp.sql;
//= quem quiser limpar:
//=   DROP TABLE guerra_pvp_confronto;
//= O caminho de volta é o git, que tem as duas metades.
//=
//= O ANÚNCIO SAI SEMPRE, mesmo quando a morte não pontua. Foi
//= pedido assim desde o começo: o broadcast conta o que
//= aconteceu, não o que valeu ponto.
//===== O troféu da arena ====================================
//= A Caveira Humana (30995) vai para o INVENTÁRIO do matador
//= em toda morte válida. Ela não cai no chão em
//= caso nenhum, e isso não depende deste script: o item é
//= `NoDrop` no db/guerra/item_db.yml, o que fecha o único
//= caminho pelo qual o `getitem` largaria item no chão - com a
//= mochila cheia o buildin tenta `map_addflooritem`, mas passa
//= antes por `pc_candrop` (src/map/script.cpp), que recusa item
//= `NoDrop`. Mochila cheia perde a caveira, e o cliente avisa o
//= jogador com a mensagem de inventário lotado.
//=
//= A CONDIÇÃO É UMA SÓ, desde 2026-08-30: matador e morto em
//= nível .Nivel - exatamente a mesma que libera o ponto. Onde
//= sai um, sai o outro, e por isso os dois moram no mesmo `if`.
//=
//= ELA JÁ TEVE TRÊS. As outras duas eram "a morte precisa
//= PONTUAR para o matador" (o .Piso) e "o morto precisa ter 1
//= ponto ou mais ANTES de morrer" (o .TrofeuPiso), as duas de
//= 2026-08-08 e as duas fora em 2026-08-30. O que elas
//= seguravam está na seção acima: a Caveira é MOEDA - a
//= descrição da Moeda Nova (30998) promete trocá-la - e hoje
//= nada limita quantas saem do mesmo par de contas.
//=
//= COM ELAS SAIU O `.@pontos_morto`: este script não lê mais a
//= reputação do morto em lugar nenhum, porque não há mais
//= decisão que dependa dela. O `attachrid` continua, e é pelo
//= nível e pelo nome.
//===== Notas ================================================
//= `killedrid` é ACCOUNT ID: para jogador, o id de bloco é o
//= account_id (src/map/pc.cpp, `sd->id = account_id`).
//=
//= OnPCKillEvent só dispara em morte causada por JOGADOR
//= (src/map/pc.cpp confere `src->type == BL_PC` e resolve
//= pet/homúnculo/mercenário para o dono). Por isso não há teste
//= de "morreu para monstro" aqui - o OnPCDieEvent precisaria
//= dele, este não.
//=
//= O piso de queda é aplicado em SQL, com GREATEST(), e não no
//= script: o valor novo tem de sair da MESMA expressão que soma
//= o delta, senão duas mortes simultâneas leriam o mesmo total
//= antigo e uma sobrescreveria a outra. É o mesmo motivo pelo
//= qual o total é relido do banco logo depois, e não calculado
//= aqui.
//=
//= O sprite 20269 (JT_GUILD_SKILL_FLAG) existe neste cliente:
//= está no npcidentity.lub do GRF e a arte
//= (guild_skill_flag.spr/.act) também. A faixa 20xxx vai até
//= 21587 - o teto de 10508 da minha nota antiga é só da faixa
//= 10xxx.
//=
//= As sub-rotinas usam `.@sp_` nos nomes porque `callsub`
//= COMPARTILHA o escopo `.@` com quem chamou (ao contrário de
//= `callfunc`). Sem o prefixo, a primeira chamada apagaria as
//= variáveis que a segunda precisa.
//=
//= Os textos vão COM ACENTO, gravados em Latin-1 (ISO-8859-1).
//= Ver PENDENCIAS.md, seção "Acentuação no diálogo".
//=
//= ATENÇÃO ao editar: por causa do acento, ferramenta que
//= assume UTF-8 troca cada letra acentuada por '?' sem avisar.
//= Editar com script que grave em cp1252, como o resto de
//= npc/guerra/.
//=
//= Mudança neste arquivo pega com @reloadscript. A tabela
//= precisa existir antes: ver sql-files/guerra_arena_pvp.sql.
//============================================================

prontera,142,180,4	script	Placar da Arena	20269,{
	mes "[Placar da Arena]";
	mes "^777777TOP " + .Top + " Vencedores:^000000";

	// Só pontuação POSITIVA entra no quadro, e o mesmo critério vale
	// no "Ranking completo" - foi pedido assim. Quem está em zero ou
	// negativo não aparece em lugar nenhum da placa.
	.@n = query_sql("SELECT nome, pontos FROM guerra_pvp_placar WHERE pontos > 0 ORDER BY pontos DESC, nome ASC LIMIT " + .Top, .@nome$, .@pontos);

	// As .Top linhas saem SEMPRE, mesmo sem gente para preenchê-las: um
	// pódio com buraco diz "cabe mais um" melhor que uma lista curta.
	for (.@i = 0; .@i < .Top; .@i++) {
		if (.@i < .@n)
			mes (.@i + 1) + " - " + .@nome$[.@i] + "  ^0000FF" + .@pontos[.@i] + "^000000";
		else
			mes (.@i + 1) + " - ^777777vago^000000";
	}
	mes " ";

	// As regras que o jogador precisa saber ANTES de brigar. Os números
	// vêm do OnInit, e não do texto: mexer no piso conserta a regra e a
	// explicação de uma vez.
	mes "A regra é uma só: valem as lutas entre dois de nível ^0000FF" + .Nivel + "^000000 - os DOIS lados.";
	mes " ";
	mes "Quem mata ganha ^0000FF1^000000 ponto e ainda recolhe a ^0000FFCaveira Humana^000000 do perdedor.";
	mes " ";
	mes "Quem morre perde ^0000FF1^000000 ponto, até o piso de ^0000FF" + .PisoQueda + "^000000. Daí não desce mais.";
	mes " ";
	mes "^777777(Não há limite por dia nem espera entre uma luta e outra. Toda morte na arena vira anúncio, valendo ponto ou não.)^000000";
	next;

	switch (select("Ranking completo:Minha pontuação:Sair")) {
	case 1:
		.@q = query_sql("SELECT nome, pontos FROM guerra_pvp_placar WHERE pontos > 0 ORDER BY pontos DESC, nome ASC LIMIT " + .Completo, .@rk$, .@rp);
		mes "[Placar da Arena]";
		if (!.@q) {
			mes "A placa está limpa. Ninguém pontuou ainda.";
			mes " ";
			mes "A arena fica cinco células a leste daqui.";
			close;
		}
		mes "^777777Ranking completo:^000000";
		for (.@i = 0; .@i < .@q; .@i++)
			mes (.@i + 1) + " - " + .@rk$[.@i] + "  ^0000FF" + .@rp[.@i] + "^000000";
		close;
	case 2:
		mes "[Placar da Arena]";
		mes "A sua conta: ^0000FF" + #RepPointsPvP + "^000000 ponto(s).";
		next;
		if (select("Abrir o meu status de reputação:Sair") == 2) {
			mes "[Placar da Arena]";
			mes "A placa continua aqui.";
			close;
		}
		close2;
		reputationui .Grupo;
		end;
	case 3:
		mes "[Placar da Arena]";
		mes "A placa continua aqui.";
		close;
	}
	end;

OnInit:
	.Mapa$ = "pvp_n_1-5";   // a arena, e só ela
	.Rep = 5;               // o Id: de db/guerra/reputation.yml
	.Grupo = 4;             // o Id: de db/guerra/reputation_group.yml
	.Nivel = 200;           // nível de base exigido dos DOIS lados
	.PisoQueda = -10;       // o fundo do poço de quem morre, e o único limite que
	                        // sobrou depois de 2026-08-30. Tem de ser
	                        // igual ao `Minimum:` do Id 5 em
	                        // db/guerra/reputation.yml, senão a tabela e o modal
	                        // param em números diferentes. Ver o cabeçalho.
	.Trofeu = 30995;        // Caveira Humana - o troféu que o matador leva, e
	                        // ele sai em toda morte válida desde 2026-08-30
	.Top = 5;               // linhas do quadro de cima
	.Completo = 15;         // linhas do "Ranking completo"
	end;

OnPCKillEvent:
	if (strcharinfo(3) != .Mapa$)
		end;

	.@matador = getcharid(3);
	.@morto = killedrid;
	if (.@morto == 0 || .@morto == .@matador)
		end;

	.@cid_matador = getcharid(0);
	.@nome_matador$ = strcharinfo(0);
	.@nivel_matador = BaseLevel;

	// Um pulo até o morto para ler o que só se lê estando anexado: nível
	// e nome. Se ele já saiu do ar entre a morte e este evento, não há o
	// que pontuar nem o que anunciar.
	//
	// A reputação dele NÃO é lida mais desde 2026-08-30: nenhuma decisão
	// depende dela. Ver "As regras" no cabeçalho.
	if (!attachrid(.@morto))
		end;
	.@cid_morto = getcharid(0);
	.@nome_morto$ = strcharinfo(0);
	.@nivel_morto = BaseLevel;
	attachrid(.@matador);

	// Sai antes das travas: anúncio é relato, não prêmio.
	announce .@nome_matador$ + " matou " + .@nome_morto$ + " na Arena de Combate de Prontera", bc_all;

	// A ÚNICA TRAVA QUE SOBROU, e ela vale para as três coisas que vêm
	// depois: o ponto do matador, a caveira e o ponto perdido pelo morto.
	// Ver "As regras" no cabeçalho antes de acrescentar qualquer outra.
	if (.@nivel_matador < .Nivel || .@nivel_morto < .Nivel)
		end;

	// O troféu sai ANTES do S_Pontua de propósito: aqui o jogador anexado
	// ainda é com certeza o matador.
	getitem .Trofeu, 1;
	dispbottom "Você recolheu a Caveira Humana de " + .@nome_morto$ + ".";

	callsub S_Pontua, .@matador, .@cid_matador, .@nome_matador$, 1;
	callsub S_Pontua, .@morto, .@cid_morto, .@nome_morto$, -1;
	end;

// (conta, char id, nome, quanto somar) -> grava na tabela e devolve o total
// para o modal. A tabela é escrita PRIMEIRO e relida depois de propósito: é
// ela que decide o número, e não a variável de conta que estava em memória.
//
// O GREATEST() é o piso de queda, e ele fica aqui e não num `if` do script
// porque precisa sair da mesma expressão que soma o delta - ver "Notas".
S_Pontua:
	.@sp_conta = getarg(0);
	.@sp_cid = getarg(1);
	.@sp_nome$ = getarg(2);
	.@sp_delta = getarg(3);

	query_sql("INSERT INTO guerra_pvp_placar (account_id, nome, pontos, atualizado) VALUES (" + .@sp_conta + ", '" + escape_sql(.@sp_nome$) + "', GREATEST(" + .@sp_delta + ", " + .PisoQueda + "), NOW()) ON DUPLICATE KEY UPDATE pontos = GREATEST(pontos + " + .@sp_delta + ", " + .PisoQueda + "), nome = VALUES(nome), atualizado = NOW()");
	if (query_sql("SELECT pontos FROM guerra_pvp_placar WHERE account_id = " + .@sp_conta, .@sp_pontos))
		set_reputation_points .Rep, .@sp_pontos, .@sp_cid;
	return;

// Rede contra queda do map-server: se ele cair entre dois autosaves, a
// variável de conta volta ao valor antigo e a tabela fica certa. Aqui a
// tabela reimpõe o número.
//
// Quem nunca pontuou não paga pacote nenhum: sem linha na tabela e com a
// variável zerada, o evento sai antes do set_reputation_points.
OnPCLoginEvent:
	.@pontos = 0;
	.@achou = query_sql("SELECT pontos FROM guerra_pvp_placar WHERE account_id = " + getcharid(3), .@pontos);
	if (!.@achou && #RepPointsPvP == 0)
		end;
	set_reputation_points .Rep, .@pontos, getcharid(0);
	end;
}
