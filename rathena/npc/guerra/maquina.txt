prontera,167,199,6	script	Máquina	564,{
	mes "[Máquina]";
	mes "A máquina acorda com um estalo.";
	mes "Uma fenda estreita se abre no painel,";
	mes "do tamanho exato de uma ^0000FFMoeda Nova^000000.";
	next;

	.@saldo = countitem(.moeda);
	mes "[Máquina]";
	if (.@saldo == 0) {
		mes "Você não tem nenhuma ^0000FFMoeda Nova^000000.";
		mes "A fenda continua aberta, esperando.";
		next;
		mes "[Máquina]";
		mes "Ainda assim o mostrador acende,";
		mes "e lista o que a gaveta guarda.";
	} else {
		mes "Você tem ^0000FF" + .@saldo + "^000000 " +
		    ((.@saldo == 1) ? "Moeda Nova" : "Moedas Novas") + ".";
		mes "O mostrador acende.";
	}

	// `close2` e nao `close`: fecha a caixa de dialogo mas mantem o
	// jogador preso ao NPC, que e o que permite a janela de troca
	// abrir em seguida. Com `close` a loja nao abriria.
	close2;
	callshop "Maquina#loja";
	end;

OnInit:
	// O ID mora aqui e nao espalhado pelo corpo pelo mesmo motivo de
	// sempre: se a Moeda um dia mudar de ID, muda-se numa linha.
	// A LISTA de mercadoria NAO esta mais neste arquivo - ver o
	// npc/guerra/barters_guerra.yml.
	.moeda = 30998;   // Moeda Nova, db/guerra/item_db.yml
	end;
}
