// 2026-08-14: deixou de ser `duplicate` do de 165,87 - e a porta
// da Sala Secreta agora. Ver o AVISO no cabecalho.
auction_01,194,87,4	script	Guarda#co_can_l	966,{
	if (SalaSecretaOrdem < 11) {
		mes "[Guarda]";
		mes "Sem acesso.";
		close;
	}

	if (SalaSecretaOrdem == 11) {
		mes "[Guarda]";
		mes "Você tem uma senha pra mim?";
		next;
		input .@senha$;
		if (.@senha$ == "Audaces Fortuna Juvat") {
			mes "[Guarda]";
			mes "As palavras são ditas!";
			next;
			mes "[Guarda]";
			mes "Bem-vindo, milord!";
			SalaSecretaOrdem = 12;
			close2;
			warp "prt_in",129,108;
			end;
		}
		mes "[Guarda]";
		mes "Sem acesso.";
		close;
	}

	// SalaSecretaOrdem >= 12: acesso ja liberado, sem pedir a
	// senha de novo.
	mes "[Guarda]";
	mes "Entrar na Sala Secreta?";
	next;
	if (select("Sim:Não") == 1) {
		close2;
		warp "prt_in",129,108;
		end;
	}
	close;
}
