//= zera_revenda_das_lojas.py: ele e para as vitrines de
//= Prontera (CLAUDE.md secao 4.16), e barter nao e vitrine.
//===== Como o item volta inteiro ============================
//= Nao existe comando de script que tire UMA carta de UMA cova.
//= O `successremovecards` do rAthena tira TODAS as cartas da
//= peca de uma vez - e tira a Essencia de Morroc junto, porque
//= ela e `Type: Card` - o que quebraria as regras 2 e 8 ao
//= mesmo tempo. Entao o caminho e desmontar e remontar:
//= `delequip` mais `getitem4`, com as covas que ficaram.
//=
//= Remontar assim PERDE O QUE NAO FOR PASSADO DE VOLTA, e isso
//= e o que exige cuidado:
//=
//=   refino          `getequiprefinerycnt`, lido antes
//=   covas restantes `getequipcardid` das quatro, lidas antes
//=   grau de encanto `@inventorylist_enchantgrade[]`
//=   vinculo (bound) `@inventorylist_bound[]`
//=   bonus aleatorio NAO se devolve - regra 5, e e o unico
//=                   campo que se perde de proposito
//=
//= O grau e o vinculo NAO tem funcao de leitura por slot de
//= equipamento: quem os tem e o `getinventorylist`, e por isso
//= ele e chamado e a linha do item vestido e procurada pelo
//= bit de posicao (`@inventorylist_equip[] & EQP_*`). Sem isso,
//= separar carta de um item VINCULADO o devolveria solto - um
//= item de conta virando mercadoria, calado.
//=
//= ITEM ALUGADO E RECUSADO por isso mesmo: o
//= `@inventorylist_expire[]` diz que ele tem prazo, e remontar
//= com `getitem4` devolveria uma peca eterna. E a unica recusa
//= que este NPC faz por natureza do item.
//=
//= E o `getitem4` exige TRES arrays de opcao aleatoria, mesmo
//= para nao por nenhuma: eles vao zerados (.@op_id/.@op_val/
//= .@op_par), e sao o que cumpre a regra 5.
//===== O que o codigo protege, e nao se ve ==================
//= 1) A LISTA DE CARTAS DE MVP e GERADA, nao escrita a mao. Os
//=    147 ids sairam de `ferramentas/varre_cartas.chefes()`, a
//=    mesma funcao que decide quem entra na loja "Carta de MVP"
//=    do npc/guerra/mercado_de_cartas.txt - ou seja as duas
//=    listas concordam por construcao. A do rAthena, no
//=    card_separation.txt, tem 45 e esta parada em 2015.
//=
//=    Carta de MVP nao se reconhece pelo item_db: nao ha marca
//=    la. Ela se deriva do `mob_db`, por quem a dropa, e o
//=    rAthena marca MVP de duas formas (`MvpExp:` e `Mvp:`).
//=    Ao rodar o varre_cartas.py de novo, reconferir esta lista.
//=
//=    Nenhuma carta SELADA cai nesta lista - conferido: as
//=    seladas nao sao drop de MVP. E o que faz a regra 7 valer
//=    sozinha, sem uma segunda lista.
//=
//= 2) O MENU E OS ARRAYS SAEM DA MESMA FONTE, num laco - nao ha
//=    string de menu escrita a mao ao lado de um `setarray`
//=    escrito a parte. E a regra 4.11 do CLAUDE.md, e ela custou
//=    caro no Teleportador da Ordem: menu e destino divergiram e
//=    catorze viagens foram parar no lugar errado, sem erro
//=    nenhum. Aqui as tres colunas (rotulo, EQI, EQP) sao
//=    conferidas por `getarraysize` no OnInit, com `debugmes`.
//=
//= 3) NENHUM `||` DENTRO DE `if` COM INDICE. O `||` do script do
//=    rAthena NAO faz curto-circuito (CLAUDE.md secao 5): os
//=    dois lados sao sempre avaliados. Por isso toda guarda do
//=    tipo "se ha carta E ela e comum" esta escrita como `if`
//=    aninhado.
//=
//= 4) O ITEM E RECONFERIDO DEPOIS DO ULTIMO MENU. Entre a
//=    escolha da cova e o pagamento ha `next`, e nele o jogador
//=    pode trocar de equipamento. Antes de cobrar, o NPC
//=    confere que a peca continua vestida, que e a mesma
//=    (`getequipid`) e que as quatro covas nao mudaram
//=    (`F_IsEquipCardHack`, que ainda escreve no log).
//=
//= 5) EQUIPAMENTO FORJADO OU ASSINADO E RECUSADO. Em peca
//=    forjada a cova 0 guarda um marcador (254, 255 ou 256) e as
//=    covas 2 e 3 guardam o id do personagem partido em dois -
//=    numeros grandes que passam por id de item. Ler isso como
//=    carta e o caminho para o NPC "devolver" uma carta que nao
//=    existe.
//=
//= 6) `rand` COM 100, NUNCA COM VARIAVEL. O `rand(n)` mata o
//=    script quando n vale 1 (CLAUDE.md secao 5); aqui o
//=    argumento e sempre a constante 100.
//============================================================

malangdo,208,166,6	script	Richard#guerra	559,{
	disable_items;

	if (checkweight(1201,3) == 0) {
		mes "[Richard]";
		mes "Você está carregando tipos de item demais. Guarde alguma coisa e volte.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "[Richard]";
		mes "Você está pesado demais. Guarde alguma coisa e volte.";
		close;
	}

	mes "[Richard]";
	mes "Eu separo carta de equipamento. Uma carta por vez, e a peça precisa estar ^0000FFvestida^000000.";
	next;
	mes "[Richard]";
	mes "Se der errado, ^0000FFvocê não perde nada^000000: nem a peça, nem a carta, nem o refino, nem o encantamento. Perde só o que me pagou.";
	next;
	mes "[Richard]";
	mes "Mas repare: quando dá ^0000FFcerto^000000, os ^FF0000bônus aleatórios^000000 da peça se perdem. E Essência de Morroc não é carta - dessas eu não cuido.";
	next;

	// As TRES colunas da tabela de slots. Ver a nota (2) do cabecalho:
	// o menu abaixo e montado num laco a partir daqui, e nao escrito a
	// mao ao lado. A terceira coluna e o bit de posicao, e ela existe
	// porque o `getinventorylist` responde por EQP_*, nao por EQI_*.
	setarray .@rotulo$[0],
		"Chapéu (topo)","Chapéu (meio)","Chapéu (baixo)","Armadura",
		"Mão direita","Mão esquerda","Capa","Sapato",
		"Acessório direito","Acessório esquerdo";
	setarray .@slot[0],
		EQI_HEAD_TOP,EQI_HEAD_MID,EQI_HEAD_LOW,EQI_ARMOR,
		EQI_HAND_R,EQI_HAND_L,EQI_GARMENT,EQI_SHOES,
		EQI_ACC_R,EQI_ACC_L;
	setarray .@bit[0],
		EQP_HEAD_TOP,EQP_HEAD_MID,EQP_HEAD_LOW,EQP_ARMOR,
		EQP_HAND_R,EQP_HAND_L,EQP_GARMENT,EQP_SHOES,
		EQP_ACC_R,EQP_ACC_L;

	.@n = getarraysize(.@slot);
	if (getarraysize(.@rotulo$) != .@n) {
		debugmes "Richard#guerra: rotulo e slot com tamanhos diferentes.";
		mes "[Richard]";
		mes "Minha bancada está desarrumada hoje. Volte depois.";
		close;
	}
	if (getarraysize(.@bit) != .@n) {
		debugmes "Richard#guerra: bit e slot com tamanhos diferentes.";
		mes "[Richard]";
		mes "Minha bancada está desarrumada hoje. Volte depois.";
		close;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < .@n; .@i++) {
		if (getequipisequiped(.@slot[.@i]))
			.@menu$ = .@menu$ + .@rotulo$[.@i] + " - " + getequipname(.@slot[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^999999" + .@rotulo$[.@i] + " (vazio)^000000:";
	}

	.@esc = select("Deixa para depois:" + .@menu$);
	if (.@esc == 1) {
		mes "[Richard]";
		mes "Quando precisar, estou aqui.";
		close;
	}
	.@col = .@esc - 2;
	.@part = .@slot[.@col];

	if (getequipisequiped(.@part) == 0) {
		mes "[Richard]";
		mes "Não há nada vestido aí.";
		close;
	}

	.@equip_id = getequipid(.@part);
	.@refino = getequiprefinerycnt(.@part);
	setarray .@cova[0],
		getequipcardid(.@part,0), getequipcardid(.@part,1),
		getequipcardid(.@part,2), getequipcardid(.@part,3);

	// Ver a nota (5) do cabecalho: em peca forjada ou assinada a cova 0
	// e um marcador e as covas 2 e 3 sao o id do dono partido em dois.
	// 254 = CARD0_CREATE, 255 = CARD0_FORGE, 256 = CARD0_PET.
	if (.@cova[0] == 254 || .@cova[0] == 255 || .@cova[0] == 256) {
		mes "[Richard]";
		mes "Essa peça tem assinatura de quem a fez. Não mexo nesse tipo de trabalho.";
		close;
	}

	// A cova so entra no menu se tiver carta COMUM. `CARD_ENCHANT` e o
	// que a Essencia de Morroc e todo encantamento sao (regra 8), e
	// `getiteminfo` devolve -1 para id que nao existe.
	.@menu$ = "";
	.@tem = 0;
	for (.@i = 0; .@i < 4; .@i++) {
		.@comum = 0;
		if (.@cova[.@i] > 0) {
			if (getiteminfo(.@cova[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL)
				.@comum = 1;
		}
		if (.@comum == 1) {
			.@menu$ = .@menu$ + "Cova " + (.@i + 1) + " - " + getitemname(.@cova[.@i]) + ":";
			.@tem = 1;
		} else {
			.@menu$ = .@menu$ + "^777777Cova " + (.@i + 1) + " - sem carta^000000:";
		}
	}

	if (.@tem == 0) {
		mes "[Richard]";
		mes "Não vejo carta nenhuma nessa peça. Encantamento e Essência de Morroc não contam.";
		close;
	}

	mes "[Richard]";
	mes "De qual cova? Da esquerda para a direita, 1, 2, 3 e 4.";
	next;

	.@esc = select("Deixa para depois:" + .@menu$);
	if (.@esc == 1) {
		mes "[Richard]";
		mes "Quando precisar, estou aqui.";
		close;
	}
	.@qual = .@esc - 2;

	.@carta = .@cova[.@qual];
	.@ok = 0;
	if (.@carta > 0) {
		if (getiteminfo(.@carta, ITEMINFO_SUBTYPE) == CARD_NORMAL)
			.@ok = 1;
	}
	if (.@ok == 0) {
		mes "[Richard]";
		mes "Essa cova não tem carta que eu possa separar.";
		close;
	}

	// Regra 6 e 7: quem decide e a lista gerada do mob_db (nota 1 do
	// cabecalho). Carta selada nao esta la, entao ela cai no caminho
	// comum sozinha.
	.@mvp = 0;
	if (compare(.mvp$, "|" + .@carta + "|"))
		.@mvp = 1;

	if (.@mvp == 1) {
		mes "[Richard]";
		mes "^0000FF" + getitemname(.@carta) + "^000000 é carta de MVP. Lubrificante não dá conta dela: só o ^0000FFÓleo Removedor Especial^000000, e esse não falha nunca.";
		next;
		if (countitem(.oleo) < 1) {
			mes "[Richard]";
			mes "E você não tem nenhum. A Máquina Especial ali do lado vende, por " + .preco_oleo + " Frutas dos Gatos.";
			close;
		}
		if (select("Deixa para depois:Usar o Óleo Removedor Especial") == 1) {
			mes "[Richard]";
			mes "Quando precisar, estou aqui.";
			close;
		}
		.@paga = .oleo;
		.@chance = .chance_oleo;
	} else {
		mes "[Richard]";
		mes "Carta comum. Escolha como quer pagar - quanto melhor o lubrificante, maior a chance.";
		next;

		.@menu$ = "Deixa para depois:";
		if (Zeny >= .preco_zeny)
			.@menu$ = .@menu$ + "Pagar " + callfunc("F_InsertComma", .preco_zeny) + " zeny - " + .chance_zeny + "% de chance:";
		else
			.@menu$ = .@menu$ + "^999999Pagar " + callfunc("F_InsertComma", .preco_zeny) + " zeny (falta zeny)^000000:";
		if (countitem(.basico) > 0)
			.@menu$ = .@menu$ + "Usar Lubrificante Básico - " + .chance_basico + "% de chance:";
		else
			.@menu$ = .@menu$ + "^999999Lubrificante Básico (não tem)^000000:";
		if (countitem(.refinado) > 0)
			.@menu$ = .@menu$ + "Usar Lubrificante Refinado - " + .chance_refinado + "% de chance";
		else
			.@menu$ = .@menu$ + "^999999Lubrificante Refinado (não tem)^000000";

		switch (select(.@menu$)) {
		case 1:
			mes "[Richard]";
			mes "Quando precisar, estou aqui.";
			close;
		case 2:
			if (Zeny < .preco_zeny) {
				mes "[Richard]";
				mes "Você não tem esse dinheiro todo.";
				close;
			}
			.@paga = 0;
			.@chance = .chance_zeny;
			break;
		case 3:
			if (countitem(.basico) < 1) {
				mes "[Richard]";
				mes "Você não tem Lubrificante Básico.";
				close;
			}
			.@paga = .basico;
			.@chance = .chance_basico;
			break;
		case 4:
			if (countitem(.refinado) < 1) {
				mes "[Richard]";
				mes "Você não tem Lubrificante Refinado.";
				close;
			}
			.@paga = .refinado;
			.@chance = .chance_refinado;
			break;
		}
	}

	// ----- daqui para baixo nao ha mais menu -------------------------
	// Ver a nota (4) do cabecalho: entre o primeiro menu e esta linha o
	// jogador teve `next` de sobra para trocar de equipamento.
	if (getequipisequiped(.@part) == 0) {
		mes "[Richard]";
		mes "Você tirou a peça. Vista de novo e me chame.";
		close;
	}
	if (getequipid(.@part) != .@equip_id) {
		mes "[Richard]";
		mes "Essa não é mais a peça que você me mostrou.";
		close;
	}
	if (callfunc("F_IsEquipCardHack", .@part, .@cova[0], .@cova[1], .@cova[2], .@cova[3]))
		close;

	// O vinculo, o prazo e o grau de encanto so existem aqui - nao ha
	// `getequip*` para nenhum dos tres. Ver "Como o item volta inteiro".
	.@achou = 0;
	.@vinculo = 0;
	.@prazo = 0;
	.@grau = 0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_equip[.@i] & .@bit[.@col]) {
			if (@inventorylist_id[.@i] == .@equip_id) {
				.@vinculo = @inventorylist_bound[.@i];
				.@prazo   = @inventorylist_expire[.@i];
				.@grau    = @inventorylist_enchantgrade[.@i];
				.@achou = 1;
				break;
			}
		}
	}
	if (.@achou == 0) {
		debugmes "Richard#guerra: item vestido nao achado no getinventorylist (id " + .@equip_id + ").";
		mes "[Richard]";

		close;
	}
	if (.@prazo != 0) {
		mes "[Richard]";
		mes "Essa peça é ^FF0000alugada^000000. Se eu a desmontar, ela volta sem prazo - e isso eu não faço.";
		close;
	}

	// Cobranca. Daqui em diante o pagamento ja foi, de certo ou errado -
	// e a regra 3: na falha e SO ele que se perde.
	if (.@paga == 0)
		Zeny = Zeny - .preco_zeny;
	else
		delitem .@paga, 1;

	mes "[Richard]";
	mes "Vamos lá...";
	next;

	if (rand(100) >= .@chance) {
		specialeffect2 EF_SUI_EXPLOSION;
		mes "-- Resultado da separação --";
		mes "A carta ^FF0000não saiu^000000 da cova.";
		mes "^0000FFNada se perdeu^000000: a peça, a carta, o refino e o encantamento estão como estavam.";
		next;
		mes "[Richard]";
		mes "Perdemos só o material. Quiser tentar de novo, é só falar.";
		close;
	}

	.@saiu = .@cova[.@qual];
	.@cova[.@qual] = 0;

	// Os tres arrays zerados sao o que APAGA o bonus aleatorio - regra 5.
	// O `getitem4` os exige mesmo quando nao ha opcao nenhuma a por.
	setarray .@op_id[0], 0,0,0,0,0;
	setarray .@op_val[0], 0,0,0,0,0;
	setarray .@op_par[0], 0,0,0,0,0;

	delequip .@part;
	if (.@vinculo > 0)
		getitembound4 .@equip_id,1,1,.@refino,0,.@cova[0],.@cova[1],.@cova[2],.@cova[3],.@vinculo,.@grau,.@op_id,.@op_val,.@op_par;
	else
		getitem4 .@equip_id,1,1,.@refino,0,.@cova[0],.@cova[1],.@cova[2],.@cova[3],.@grau,.@op_id,.@op_val,.@op_par;
	getitem .@saiu, 1;

	specialeffect2 EF_MAXPOWER;
	mes "-- Resultado da separação --";
	mes "^0000FFA carta saiu inteira^000000, e a peça também.";
	mes "Refino e encantamento ficaram como estavam.";
	mes "Os ^FF0000bônus aleatórios^000000 se perderam, como eu avisei.";
	next;
	mes "[Richard]";
	mes "Bom proveito. Volte quando quiser.";
	close;

//============================================================
// A MAQUINA ESPECIAL de Malangdo.
//
// E a `Special Vending Machine` do rAthena, reescrita em
// portugues e com a mercadoria do modelo novo. Ver a nota (3)
// do cabecalho: ela NAO existe so por causa de lubrificante -
// as seis primeiras linhas sao a entrada das instancias do
// Esgoto (Protecao do Deus do Mar) e da Caverna do Polvo
// (Caixa de Arpoes), e desligar o original sem trazer as seis
// junto fecharia as duas, calado.
//
// O QUE MUDOU DA LISTA VELHA:
//
//   saiu   Lubrificante Comum        (6440)  36 Frutas
//   saiu   Lubrificante Sofisticado  (6441)  56 Frutas
//   entra  Lubrificante Basico      (25238)   4 Frutas
//   entra  Lubrificante Refinado    (25239)   7 Frutas
//   ficou  Oleo Removedor Especial   (6443) 192 Frutas
//
// A TROCA DOS VELHOS e a regra 10 da imagem, e o preco dela e
// o que os dois CUSTAVAM na propria maquina - 36 e 56 Frutas.
// Devolver menos seria confiscar; devolver mais abriria um
// ganho para quem tivesse estoque. Nao ha numero oficial para
// isso, entao ficou o unico defensavel.
//
// O nome unico e `Maquina Especial#malangdo`, e o `#` nao e
// enfeite: ja existe uma "Maquina Especial" no projeto, a da
// Sala Secreta da Ordem (prt_in 137,108). Nome unico repetido
// derruba o segundo NPC na carga.
//
// Nao e loja de barter, e por dois motivos: ela precisa
// COMPRAR de volta (a janela de troca so vende), e o nome
// `MaquinaEspecial#loja` ja e da Sala Secreta.
//============================================================

malangdo,218,165,5	script	Máquina Especial#malangdo	562,{
	if (checkweight(1201,1) == 0) {
		mes "[Máquina Especial]";
		mes "Você está carregando tipos de item demais.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "[Máquina Especial]";
		mes "Você está pesado demais.";
		close;
	}

	mes "[Máquina Especial]";
	mes "Bem-vindo à CX-82. Aqui se compra com ^0000FF" + getitemname(.fruta) + "^000000.";
	mes "Você tem: ^0000FF" + countitem(.fruta) + "^000000.";
	next;

	switch (select("Comprar:Trocar lubrificante antigo:Explicação:Nada, obrigado")) {
	case 1:
		break;
	case 2:
		// `goto` e nao `callsub` porque o bloco de troca nunca volta:
		// todo caminho dele termina em `close`.
		goto S_TROCA;
	case 3:
		mes "[Máquina Especial]";
		mes "A ^0000FFProteção do Deus do Mar^000000 e o ^0000FFArpão de Polvo^000000 dão entrada nas masmorras do Esgoto e da Caverna do Polvo, cada um pelo prazo que o nome diz.";
		next;
		mes "[Máquina Especial]";
		mes "Os ^0000FFlubrificantes^000000 aumentam a chance do Richard separar uma carta: o Básico leva a " + .chance_basico + "%, o Refinado a " + .chance_refinado + "%.";
		next;
		mes "[Máquina Especial]";
		mes "O ^0000FFÓleo Removedor Especial^000000 é outra coisa: ele tira ^0000FFcarta de MVP^000000, e com " + .chance_oleo + "% de certeza. Não serve para carta comum nem para carta selada.";
		next;
		mes "[Máquina Especial]";
		mes "O Lubrificante Comum e o Sofisticado ^FF0000não servem mais^000000. Troque-os aqui pela fruta que custaram.";
		close;
	case 4:
		mes "[Máquina Especial]";
		mes "Volte sempre.";
		close;
	}

	// A vitrine. As TRES colunas saem do mesmo `setarray` e o menu e
	// montado num laco a partir delas - CLAUDE.md secao 4.11.
	//
	// O PRECO DOS TRES LUBRIFICANTES APARECE EM DOIS NPCs: aqui, que e
	// quem cobra, e no `.preco_*` do OnInit do Richard, que so o CITA
	// numa fala ("a Maquina Especial vende, por 192 Frutas"). Mudar um
	// sem o outro nao quebra nada e nao da erro - so faz o Richard
	// mentir o preco. Os dois lugares estao marcados com este mesmo
	// aviso.
	setarray .@item[0],
		16740, 16763, 16764,
		16765, 16766, 16767,
		25238, 25239, 6443;
	setarray .@custo[0],
		4, 12, 24,
		2, 5, 10,
		4, 7, 192;
	setarray .@prazo$[0],
		" [23 horas]", " [7 dias]", " [15 dias]",
		" [23 horas]", " [3 dias]", " [7 dias]",
		"-", "-", "-";

	.@n = getarraysize(.@item);
	if (getarraysize(.@custo) != .@n) {
		debugmes "Maquina Especial#malangdo: item e custo com tamanhos diferentes.";
		mes "[Máquina Especial]";
		mes "Erro interno. Chame um administrador.";
		close;
	}
	if (getarraysize(.@prazo$) != .@n) {
		debugmes "Maquina Especial#malangdo: item e prazo com tamanhos diferentes.";
		mes "[Máquina Especial]";
		mes "Erro interno. Chame um administrador.";
		close;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < .@n; .@i++) {
		.@rot$ = getitemname(.@item[.@i]);
		if (.@prazo$[.@i] != "-")
			.@rot$ = .@rot$ + .@prazo$[.@i];
		.@menu$ = .@menu$ + .@rot$ + " - " + .@custo[.@i] + " fruta(s):";
	}

	.@esc = select("Deixa para depois:" + .@menu$);
	if (.@esc == 1) {
		mes "[Máquina Especial]";
		mes "Volte sempre.";
		close;
	}
	.@i = .@esc - 2;

	if (countitem(.fruta) < .@custo[.@i]) {
		mes "[Máquina Especial]";
		mes "Faltam ^FF0000" + (.@custo[.@i] - countitem(.fruta)) + "^000000 " + getitemname(.fruta) + ".";
		close;
	}

	delitem .fruta, .@custo[.@i];
	getitem .@item[.@i], 1;
	mes "[Máquina Especial]";
	mes "Negócio fechado. Volte sempre.";
	close;

// A regra 10 da imagem: os dois lubrificantes velhos voltam a ser a
// fruta que custaram nesta mesma maquina.
S_TROCA:
	if (countitem(.velho_comum) < 1) {
		if (countitem(.velho_sofis) < 1) {
			mes "[Máquina Especial]";
			mes "Você não tem nenhum dos dois lubrificantes antigos.";
			close;
		}
	}

	mes "[Máquina Especial]";
	mes "O Lubrificante Comum e o Sofisticado não servem mais para nada. Devolvo a fruta que custaram.";
	next;

	.@menu$ = "Deixa para depois:";
	if (countitem(.velho_comum) > 0)
		.@menu$ = .@menu$ + getitemname(.velho_comum) + " (" + countitem(.velho_comum) + ") - " + .volta_comum + " cada:";
	else
		.@menu$ = .@menu$ + "^999999" + getitemname(.velho_comum) + " (não tem)^000000:";
	if (countitem(.velho_sofis) > 0)
		.@menu$ = .@menu$ + getitemname(.velho_sofis) + " (" + countitem(.velho_sofis) + ") - " + .volta_sofis + " cada";
	else
		.@menu$ = .@menu$ + "^999999" + getitemname(.velho_sofis) + " (não tem)^000000";

	switch (select(.@menu$)) {
	case 1:
		mes "[Máquina Especial]";
		mes "Volte sempre.";
		close;
	case 2:
		.@qual = .velho_comum;
		.@vale = .volta_comum;
		break;
	case 3:
		.@qual = .velho_sofis;
		.@vale = .volta_sofis;
		break;
	}

	.@quantos = countitem(.@qual);
	if (.@quantos < 1) {
		mes "[Máquina Especial]";
		mes "Você não tem esse.";
		close;
	}

	delitem .@qual, .@quantos;
	getitem .fruta, .@quantos * .@vale;
	mes "[Máquina Especial]";
	mes "Troquei ^0000FF" + .@quantos + "^000000 por ^0000FF" + (.@quantos * .@vale) + "^000000 " + getitemname(.fruta) + ".";
	close;

OnInit:
	disablenpc "Special Vending Machine";

	.fruta = 6417;            // Fruta dos Gatos

	.chance_basico = 10;
	.chance_refinado = 20;
	.chance_oleo = 100;

	// A troca da regra 10. O valor e o que os dois CUSTAVAM aqui.
	.velho_comum = 6440;      // Lubrificante Comum
	.volta_comum = 36;
	.velho_sofis = 6441;      // Lubrificante Sofisticado
	.volta_sofis = 56;
	end;
}
