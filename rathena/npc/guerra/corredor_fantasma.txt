// O cerebro: guarda a configuracao e recebe a morte dos chefes.
// Flutuante (sem mapa) de proposito - nao ha nada para clicar aqui,
// e o nome curto e o que as 65 linhas de spawn referenciam.
-	script	corredor_fantasma	-1,{
	end;

OnInit:
	.Mapa$ = "vis_h01";     // a sala
	.EntraX = 36;           // onde o jogador desembarca
	.EntraY = 34;
	.Volta$ = "comodo";     // e para onde ele volta
	.VoltaX = 208;
	.VoltaY = 185;

	.Flor = 25503;          // Flor Visionaria
	.FlorQtd = 1;           // quantas por chefe, quando cai
	.FlorChance = 30;       // em quantas mortes de cada 100 ela cai
	end;

OnChefeMorto:
	// Disparado pelas 65 linhas de spawn do fim do arquivo, e so por
	// elas. O jogador anexado e quem deu o ultimo golpe.
	if (!playerattached())
		end;

	// Rede de seguranca: se um dia alguem copiar uma linha de spawn
	// deste arquivo para outro mapa e esquecer de trocar o evento, a
	// flor nao vaza para fora da sala.
	if (strcharinfo(3) != .Mapa$)
		end;

	// rand(100) devolve 0..99, entao `< 30` e 30% cravado.
	if (rand(100) >= .FlorChance)
		end;

	getitem .Flor, .FlorQtd;
	end;
}
