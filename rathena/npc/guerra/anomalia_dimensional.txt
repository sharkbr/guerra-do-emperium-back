//===== A PECA MAIS IMPROVAVEL: CURAR MONSTRO FUNCIONA ======================
//
//= A mecanica central do evento parece precisar de C++ e nao precisa. Em
//= RO a intuicao diz que Curar so alcanca jogador, e que em monstro ou
//= falha ou vira dano. Metade disso e verdade - a metade dos mortos-vivos.
//
//= Quem decide e o SkillHeal::castendNoDamageId
//= (src/map/skills/acolyte/heal.cpp, neste rAthena refatorado em que cada
//= habilidade tem classe propria). A funcao calcula o valor, e a UNICA
//= coisa que ela faz com alvo monstro e zerar a cura em tres casos:
//
//=     if (status_isimmune(bl) || (dstmd && (status_get_class(bl) ==
//=         MOBID_EMPERIUM || status_get_class_(bl) == CLASS_BATTLEFIELD)))
//=             heal = 0;
//
//= e logo abaixo faz `status_heal(bl, heal, 0, 0)` com o `bl` que veio -
//= monstro inclusive. Ou seja: cura monstro de verdade, desde que ele nao
//= seja o Emperium, nao seja `Class: Battlefield` e nao seja imune. O
//= `status_isimmune` (src/map/status.cpp:9306) so olha jogador (a Garra do
//= Golden Thief Bug) e SC_HERMODE, entao para monstro devolve 0 sempre.
//
//= O Santuario segue o mesmo caminho, num lugar diferente: o
//= `case UNT_SANCTUARY` do skill_unit_onplace_timer (skill.cpp:6923) tem os
//= MESMOS tres testes e o mesmo `status_heal(bl, heal, 0, 0)`. As duas
//= metades do pedido do bRO - "Curar, Curatio e Santuario" - funcionam
//= nativas, e nenhuma linha de C++ nosso entrou neste evento.
//
//= O que TEM de ser evitado, e por isso a pedra e o 1907 e nao outro mob:
//=   - morto-vivo NAO SERVE. Em alvo undead o Curar vira dano ofensivo
//=     (skill.cpp:4417, o desvio para skill_castend_damage_id) e a pedra
//=     perderia HP em vez de ganhar. O 1907 e `Element: Neutral` e
//=     `Race: Formless`.
//=   - `Class: Battlefield` NAO SERVE, pelo `if` de cima. O 1907 e
//=     `Class: Boss`, que passa. (O vizinho 1906 e Battlefield - se um dia
//=     alguem trocar o Id "porque da no mesmo", a cura zera calada.)
//=   - MDEF 100 nao serve. O 1907 tem 50.

//===== A ECONOMIA, COM OS NUMEROS NA MAO ===================================
//
//= A missao paga 7 Moedas de Estimacao por dia. Um sorteio na Maquina
//= custa 10, e a propria tabela devolve moeda em cinco das 58 linhas -
//= media de 4,14 por sorteio. Custo liquido: 5,86 moedas, ou seja ~1,2
//= sorteios por dia para quem so faz a missao diaria.
//
//= As chances sao as do bRO ESCALADAS, por decisao do dono na mesma data:
//= o Ovo de Freeoni sai de 0,006664% para 0,3332%, que e x50. Multiplicar
//= a tabela inteira por 50 e IMPOSSIVEL - somaria 276% -, entao o fator
//= decresce conforme o premio fica comum (x50 no Freeoni, x10 no Gato de
//= Nove Caudas, x5 nos nove raros, x4 nos doze comuns) e o espaco sai das
//= comidas de pet, que continuam sendo a maior parte da tabela. A soma
//= fecha em 1.000.000 milionesimos exatos, conferida por
//= ferramentas/ ... nao: pelo script de apoio que gerou os setarray.
//
//= O que isso da na pratica, e o dono ja sabe: Freeoni e 1 em 300
//= sorteios, ou uns 250 dias de missao diaria pura. E o premio maximo e
//= tem dois atalhos - o reset pago (Moeda Nova, abaixo) e os 30 Amagos do
//= Manouro. Se doer, o lugar de calibrar e o array .Peso do OnInit da
//= Maquina, num arquivo so, e nada mais precisa mudar junto.
//
//===== O RESET DIARIO E AS 4 DA MANHA ======================================
//
//= No bRO a missao repete "apos as 4 horas da manha". Isso NAO foi feito
//= com quest de TimeLimit, que seria o caminho obvio (CLAUDE.md secao 5
//= explica que `4h` sem o `+` e hora exata e devolve o proximo 04:00) - e
//= foi de proposito: quest nova exige entrada na tabela do CLIENTE, e
//= quest que o cliente nao conhece DERRUBA O CLIENTE, uma caixa de erro
//= por missao e por atualizacao da janela. Custou um bug em 2026-08-08 nas
//= placas da Ordem. Um temporizador de evento nao vale esse risco.
//
//= O que se usa aqui e um "dia logico" calculado na mao, guardado na
//= variavel permanente `anomalia_dia` do personagem. A conta esta na
//= funcao AnomaliaDia abaixo e nao depende de constante de fuso: ela
//= descobre o deslocamento comparando a hora LOCAL (gettime(DT_HOUR), que
//= o servidor da em America/Sao_Paulo) com a hora UTC derivada do
//= gettimetick(2), e so entao corta o dia as 4h. Se o horario de verao
//= voltar, ela acompanha sozinha.
//
//===== ARMADILHAS DO PROJETO QUE ESTE ARQUIVO RESPEITA ======================

//= - `||` e `&&` NAO fazem curto-circuito no script do rAthena (secao 5).

//= - `rand(1)` MATA o script. Todo rand daqui tem argumento >= 2, e o
//=   sorteio da Maquina usa rand(1000000), constante.
//= - zero em variavel `.` ou `.@` APAGA a entrada, entao toda coluna de
//=   inteiro termina em SENTINELA -1 e a conferencia de tamanho compara

//=   pagamento aqui passa por checkweight antes.

	// db/guerra/mob_db.yml, gerado do navi_mob_br.lub do proprio bRO.
	setarray .Sombria[0], 1503,1505,1504,1507;
	setarray .Sombria[4], -1;

	// Os MVPs que a fenda pode deixar passar. O bRO diz so "um MVP
	// aleatorio, com chance desconhecida e bem baixa" - a lista e a chance
	// sao ESCOLHA NOSSA, e e o unico lugar deste evento que nao veio de
	// fonte. Sao MVPs classicos que este cliente desenha; o Phreeoni abre a
	// lista de proposito, porque e o dono do premio maximo da Maquina.
	setarray .Mvp[0], 1159,1038,1039,1046,1059,1086,1087,1112,1115,1147,1150,1157,1190,1251,1252;
	setarray .Mvp[15], -1;
	// Em milesimos, por onda. 20 = 2%.
	.MvpChance = 20;

	// Teto de monstros vivos no mapa. Sem ele uma rodada longa com pouca
	// gente vira um mapa entupido de Entidades, e o servidor sente.
	//
	// Subiu de 60 para 180 JUNTO com o triplo de monstros por onda
	// (2026-08-26), e nao por pedido separado: mantido em 60, o teto seria
	// alcancado em dez ondas e as Entidades parariam de nascer: o triplo
	// pedido nao apareceria em jogo. Sao 30 ondas de folga.
	//
	// E ESTE e o botao a mexer se o mapa pesar - nao a quantidade por onda.
	.TetoMobs = 180;

	// As duas colunas de posicao andam juntas, e a sentinela -1 fecha cada
	// uma (zero em variavel `.` apaga a entrada, entao contar sem sentinela
	// mentiria). Coluna com um valor a menos desloca tudo e nao da erro.
	if (getarraysize(.px) - 1 != 4) {
		debugmes "ANOMALIA: .px tem " + (getarraysize(.px) - 1) + " posicoes, e nao 4";
	}
	if (getarraysize(.py) - 1 != 4) {
		debugmes "ANOMALIA: .py tem " + (getarraysize(.py) - 1) + " posicoes, e nao 4";
	}

	.rodada = 0;
	.rodada_num = 0;
	initnpctimer;
	debugmes "ANOMALIA: dimensao pronta - 4 pedras de " + .PedraHpMax + " HP, comecando em " + .PedraHpIni + ".";
	end;

// Um tique por segundo. O corpo so trabalha quando ha gente na dimensao -
// mapa vazio custa uma comparacao e volta a dormir.
OnTimer1000:
	setnpctimer 0;
	if (.rodada == 0) {
		if (getmapusers("pprontera") > 0) {
			donpcevent "VarmuntDim::OnComecaRodada";
		}
		end;
	}
	if (.rodada == 2) {
		.espera -= 1;
		if (.espera <= 0) {
			donpcevent "VarmuntDim::OnLimpaRodada";
		}
		end;
	}

	// Mapa esvaziou no meio da rodada: encerra e limpa. Sem isto as
	// Entidades continuariam nascendo de dez em dez segundos numa Prontera
	// sem ninguem, ate bater no teto - e ficariam la.
	if (getmapusers("pprontera") == 0) {
		donpcevent "VarmuntDim::OnLimpaRodada";
		end;
	}

	// --- rodada em curso: mede as quatro Pedras ---
	.@somaHp = 0;
	for (.@i = 0; .@i < 4; .@i++) {
		if (unitexists(.pedra[.@i]) == 0) {
			// Pedra derrubada por alguem (ou perdida num reload): reergue no
			// mesmo lugar, com o HP que ela tinha. Nao ha aviso: a rodada
			// simplesmente continua.
			callsub S_Planta, .@i, .hpguardado[.@i];
		}
		// getunitdata NAO E FUNCAO: e comando que PREENCHE UM ARRAY, e os
		// indices sao as proprias constantes UMOB_*. Escrito como
		// `.@hp = getunitdata(gid, UMOB_HP)` ele devolve SEMPRE ZERO, com
		// um aviso que so aparece na janela do map-server. Custou a manha
		// de 2026-08-26: o dono curava, a Pedra recebia, e as quatro liam
		// "0 de 15000" - o que parecia cura que nao pega e era leitura
		// errada. Ver CLAUDE.md secao 5.
		getunitdata .pedra[.@i], .@dados;
		.@hp = .@dados[UMOB_HP];

		// O FEEDBACK, e ele e o coracao desta mecanica - nao um enfeite.
		//
		// O cliente de RO NAO DESENHA BARRA DE VIDA DE MONSTRO. Sem isto o
		// jogador lanca cura na Pedra, ve o numero verde subir e... nada
		// mais: nao sabe se falta um lancamento ou cinquenta, nem se o
		// evento esta funcionando. Foi exatamente o relato do dono em
		// 2026-08-26: "curava, com um valor, mas eu nao tive feedback,
		// entao nao sabia se algum dia ia parar".
		//
		// Entao a propria Pedra anuncia o quanto ja voltou, toda vez que o
		// HP dela sobe. Um balao por segundo no maximo (este laco e o
		// tique de 1s), e so quando ha progresso de verdade.
		if (.@hp > .hpguardado[.@i]) {
			.@pc = .@hp * 100 / .PedraHpMax;
			if (.@pc >= 100) {
				unittalk .pedra[.@i], "A luz voltou por inteiro!";
			}
			else {
				unittalk .pedra[.@i], "Pedra Guardiã: " + .@pc + "% restaurada";
			}
			specialeffect EF_HEAL, AREA, .pedra[.@i];
		}

		.hpguardado[.@i] = .@hp;
		.@somaHp += .@hp;
	}
	.@alvo = .PedraHpMax * 4;

	// Aviso do CONJUNTO, a cada dez por cento vencidos. Era de 25 em 25 e
	// passou a 10 em 10 na mesma rodada em que o balao da Pedra entrou
	// (2026-08-26): o balao diz como vai UMA Pedra, este diz como vai a
	// rodada, e sem os dois o jogador nao sabe se esta perto do fim.
	//
	// O `.aviso` guarda a ultima dezena anunciada, entao cada faixa sai uma
	// vez so - e nao a cada segundo.
	.@pct = .@somaHp * 100 / .@alvo;
	.@dez = .@pct / 10;
	if (.@dez > .aviso) {
		if (.@dez < 10) {
			.aviso = .@dez;
			mapannounce "pprontera", "As Pedras Guardiãs: " + (.@dez * 10) + "% da luz restaurada.", bc_map, 0x70A0FF;
		}
	}

	if (.@somaHp >= .@alvo) {
		donpcevent "VarmuntDim::OnFechaRodada";
		end;
	}

	// --- ondas, de dez em dez segundos ---
	.tique += 1;
	if (.tique % 10 == 0) {
		donpcevent "VarmuntDim::OnOnda";
		// Sonda de console, de dez em dez segundos. Existe porque a unica
		// pergunta que importa nesta mecanica - "a cura esta ENTRANDO?" -
		// nao tem resposta na tela: o cliente nao desenha barra de vida de
		// monstro. Com esta linha, quem estiver olhando o map-server ve o
		// HP das quatro andar (ou nao andar) enquanto alguem lanca a magia.
		// `debugmes` e nao `ShowInfo` de proposito: informacao nao tem bit
		// no console_msg_log 3 e nao chegaria ao arquivo (CLAUDE.md sec.5).
		debugmes "ANOMALIA: pedras " + .hpguardado[0] + "/" + .hpguardado[1] + "/" +
			.hpguardado[2] + "/" + .hpguardado[3] + " de " + .PedraHpMax +
			"  (" + (.@somaHp * 100 / .@alvo) + "%)";
	}
	end;

OnComecaRodada:
	if (.rodada != 0) end;
	.rodada = 1;
	.rodada_num += 1;
	.tique = 0;
	.aviso = 0;

	for (.@i = 0; .@i < 4; .@i++) {
		callsub S_Planta, .@i, .PedraHpIni;
		.hpguardado[.@i] = .PedraHpIni;
	}
	// Marca as quatro para TODO MUNDO que ja esta no mapa. E `viewpointmap`
	// e nao `viewpoint` porque este label roda pelo temporizador, sem
	// jogador anexado - o `viewpoint` comum precisa de um RID e aqui nao ha
	// nenhum. Sem esta linha, quem entrou ANTES da rodada comecar ficaria
	// sem marca ate falar com o Varmunt de novo.
	for (.@i = 0; .@i < 4; .@i++) {
		viewpointmap "pprontera", 1, .px[.@i], .py[.@i], .@i + 1, 0xFF3030;
	}
	mapannounce "pprontera", "A fenda se abre. Quatro Pedras Guardiãs pedem socorro.", bc_map, 0xFFCC00;
	end;

// Planta a Pedra <numero 0..3> com <hp>. Chamado no comeco da rodada e de
// novo se alguem derrubar a Pedra no meio.
// Marca as quatro Pedras no mini-mapa de QUEM ESTIVER ANEXADO. Precisa de um
// jogador (o viewpoint e por personagem), entao so e chamado de dentro de um
// dialogo ou do OnPCLoadMapEvent - nunca do OnTimer, que roda sem RID.
//
// Sem isto o Varmunt prometia uma marca que nao existia, e foi assim que a
// dimensao foi ao ar em 2026-08-26: o jogador chegava, ouvia "estao marcadas
// no seu mini-mapa" e nao tinha marca nenhuma. E a secao 4.17 do CLAUDE.md
// pega no ato - o texto descrevendo uma regra que ninguem escreveu.
S_Marca:
	for (.@mi = 0; .@mi < 4; .@mi++) {
		viewpoint 1, .px[.@mi], .py[.@mi], .@mi + 1, 0xFF3030;
	}
	return;

// Apaga as quatro marcas. O `2` e a acao de limpar.
S_Desmarca:
	for (.@mi = 0; .@mi < 4; .@mi++) {
		viewpoint 2, .px[.@mi], .py[.@mi], .@mi + 1, 0xFF3030;
	}
	return;

// Dispara quando alguem PISA na dimensao - depende do mapflag `loadevent`
// la em cima. O teste do mapa nao e paranoia: este label e chamado para
// TODO mapa que tenha aquele mapflag, e o servidor tem outros.
OnPCLoadMapEvent:
	if (strcharinfo(3) != "pprontera") end;
	if (.rodada == 1) {
		callsub S_Marca;
	}
	end;

// TUDO O QUE ESTE BLOCO PRECISA CHEGA POR `getarg`, e nao por herdar as
// `.@` do chamador: `callsub` abre escopo `.@` NOVO E VAZIO, igual ao
// `callfunc` (as duas ultimas linhas do buildin_callsub,
// src/map/script.cpp:5508). Ler `.@hp` do laco do OnTimer1000 aqui
// devolveria ZERO, calado.
//
// Ate 2026-08-28 este comentario afirmava o contrario, e o CLAUDE.md
// tambem. O codigo daqui sempre esteve certo - o porque e que estava
// errado -, mas a frase custou um item de jogador no
// npc/guerra/encantamento_da_ordem.txt, que foi escrito acreditando
// nela. O prefixo .@p continua, e agora por outro motivo: legibilidade.
S_Planta:
	.@pn = getarg(0);
	.@php = getarg(1);
	// Piso: `.hpguardado` de uma Pedra que nunca foi plantada vale 0 (zero
	// em variavel `.` nao guarda - ele APAGA a entrada), e uma Pedra com
	// HP 0 nasceria morta. Neste caso ela recomeca moribunda, como no
	// inicio da rodada.
	if (.@php < 1) .@php = .PedraHpIni;
	.@pk = .@pn;
	monster "pprontera", .px[.@pk], .py[.@pk], "Pedra Guardiã", 1907, 1;
	.pedra[.@pn] = $@mobid[0];
	// SAO TRES CHAMADAS, E A ORDEM E O REMEDIO PARA UM BUG DO rAthena.
	//
	// O 1907 nasce com 120.500 de HP e nos queremos 15.000. Reduzir o
	// maximo com `UMOB_MAXHP` sozinho CORROMPE o HP, porque o
	// status_set_maxhp (src/map/status.cpp:1343) faz:
	//
	//     heal = maxhp - status->max_hp;      // uint32 - uint32
	//
	// com os dois lados `uint32`. Reduzir da underflow: 15000 - 120500 vira
	// um numero gigante POSITIVO, o `if (heal > 0)` acerta, e o servidor
	// CURA em vez de reduzir. Medido em 2026-08-26: o HP foi para
	// 2.147.604.147 e nenhum `UMOB_HP` depois disso conseguia trazer de
	// volta - ficava travado em 120.500.
	//
	// A saida e nunca deixar o HP acima do maximo NOVO na hora da troca:
	//   1) baixa o HP primeiro, ainda dentro do maximo velho;
	//   2) troca o maximo - agora o underflow ainda acontece, mas o
	//      status_heal que ele dispara e limitado ao maximo recem-gravado,
	//      entao o HP so sobe ate 15.000;
	//   3) baixa o HP de novo, agora dentro do maximo certo.
	//
	// Medido passo a passo, nesta ordem exata, antes de entrar aqui.
	setunitdata .pedra[.@pn], UMOB_HP, .@php;
	setunitdata .pedra[.@pn], UMOB_MAXHP, .PedraHpMax;
	setunitdata .pedra[.@pn], UMOB_HP, .@php;
	// 1% do dano recebido. Nao e enfeite: sem isso um jogador entediado
	// derruba a Pedra e a rodada anda para tras. Nao se usa 0 aqui - o
	// campo e porcentagem inteira e 1 e o menor valor util (CLAUDE.md
	// secao 5, a entrada do damagetaken).
	setunitdata .pedra[.@pn], UMOB_DAMAGETAKEN, 1;
	// Le de volta o que o servidor REALMENTE gravou. Nao e paranoia: em
	// 2026-08-26 as Pedras nasceram cheias e a rodada fechava no mesmo
	// segundo, e so esta leitura separa "o setunitdata nao pegou" de "o
	// valor que eu mandei ja estava errado".
	getunitdata .pedra[.@pn], .@pconf;
	debugmes "ANOMALIA: plantou " + .@pn + " gid=" + .pedra[.@pn] +
		" pedi hp=" + .@php + " -> ficou " + .@pconf[UMOB_HP] +
		"/" + .@pconf[UMOB_MAXHP];
	return;

OnOnda:
	if (.rodada != 1) end;
	// Label VAZIO, nao "all": o mobcount conta monstros que tenham aquele
	// label, e o especial de "todos" e a string vazia - que conta os sem
	// label, que e exatamente o caso dos nossos (doc/script_commands.txt).
	// Com "all" ele procuraria um label chamado all, acharia zero, e o teto
	// nunca pegaria - calado, ate o mapa entupir.
	if (mobcount("pprontera", "") >= .TetoMobs) end;

	// A onda nasce em volta de uma das quatro Pedras, sorteada.
	.@n = rand(4);
	.@k = .@n;
	.@x = .px[.@k];
	.@y = .py[.@k];
	.@qual = .Sombria[rand(4)];
	// O <xs>,<ys> do areamonster e um RETANGULO de cantos, nao um raio -
	// aqui sao 20x20 celulas em volta da Pedra.
	//
	// SEIS por onda, e nao dois: o dono pediu o TRIPLO em 2026-08-26, com
	// a instrucao de "manter a cadencia, apenas aumentar a quantidade por
	// spawn". Por isso o numero mudou aqui e o intervalo de dez segundos
	// do OnTimer1000 ficou onde estava - sao dois botoes diferentes, e
	// mexer no errado deixaria as Entidades chegando em rajadas curtas em
	// vez de em maior numero.
	//
	// O `.TetoMobs` continua sendo o limite real: com 6 por onda ele e
	// alcancado tres vezes mais rapido, e e o que impede a rodada longa de
	// entupir a Prontera.
	areamonster "pprontera", .@x - 10, .@y - 10, .@x + 10, .@y + 10, "--ja--", .@qual, 6;

	if (rand(1000) < .MvpChance) {
		.@m = .Mvp[rand(15)];
		monster "pprontera", .@x, .@y, "--ja--", .@m, 1;
		announce "Anomalia Dimensional: algo grande atravessou a fenda em Prontera!", bc_all, 0xFF3030;
	}
	end;

OnFechaRodada:
	if (.rodada != 1) end;
	.rodada = 2;
	// Meio minuto para quem estava curando falar com o Varmunt e receber.
	.espera = 30;
	// killmonsterall, e nao killmonster com "all": o buildin compara o
	// label com strcmp(event,"All") EXATO (script.cpp:11486), entao "all"
	// minusculo nao mataria nada e nao diria nada.
	killmonsterall "pprontera";
	// As Pedras morreram junto com o killmonster; zerar os GIDs para o
	// tique nao tentar reergue-las enquanto se paga.
	for (.@i = 0; .@i < 4; .@i++) {
		.pedra[.@i] = 0;
	}
	mapannounce "pprontera", "As quatro Pedras estão inteiras. Falem com Varmunt, no centro.", bc_map, 0xFFCC00;
	end;

OnLimpaRodada:
	// Tira as marcas de quem ainda estiver la: sem isto o mini-mapa
	// continuaria apontando quatro Pedras que ja nao existem.
	for (.@i = 0; .@i < 4; .@i++) {
		viewpointmap "pprontera", 2, .px[.@i], .py[.@i], .@i + 1, 0xFF3030;
	}
	.rodada = 0;
	// killmonsterall, e nao killmonster com "all": o buildin compara o
	// label com strcmp(event,"All") EXATO (script.cpp:11486), entao "all"
	// minusculo nao mataria nada e nao diria nada.
	killmonsterall "pprontera";
	end;
}

//===== O MENSAGEIRO QUE ESTAVA NO CAMINHO ==================================
//
//= Este bloco NAO e do evento - e a conta que o evento teve de pagar. O
//= `Continental Messenger#01` do rAthena mora em `prontera 164,304`, e a
//= nossa Maquina Dimensional nasceu em `165,304`: uma celula ao lado.
//
//= Pior que a vizinhanca: aquele NPC tem AREA DE TOQUE `3,3`, um quadrado
//= de 7x7 celulas, e dispara uma fala na tela de quem passa. Isso cobria a
//= Maquina inteira e o Manouro, entao quem fosse girar a Maquina levava um
//= dialogo do Mensageiro por cima. Decisao do dono em 2026-08-26: mover o
//= Mensageiro para `150,306`, com a area junto.
//
//###### Por que NAO foi com `movenpc` #######################################
//
//= Era o caminho obvio - uma linha num OnInit - e ele NAO move a area. O
//= `npc_movenpc` (src/map/npc.cpp:5046) faz `map_moveblock` e mais nada:
//= nao chama `npc_unsetcells` nem `npc_setcells`. E a area de toque nao
//= mora no NPC, mora no MAPA: o `npc_setcells` (npc.cpp:4971) marca
//= `CELL_NPC` celula por celula em volta dele, no carregamento.
//
//= Ou seja, com `movenpc` o boneco andaria e o gatilho ficaria em 164,304
//= - a fala continuaria disparando no lugar antigo, em cima da Maquina, e
//= no lugar novo nao dispararia nada. Falha calada e exatamente ao
//= contrario do que se pediu.
//
//###### A receita de sempre, com uma sutileza ###############################
//
//= Entao e o caminho do CLAUDE.md secao 2: `disablenpc` no original e uma
//= duplicata nossa na coordenada nova, com o `3,3` repetido - e ai a area
//= nasce no lugar certo, porque o `npc_setcells` roda no carregamento da
//= duplicata.
//
//= A sutileza e o NOME, e ela morde. O script do Mensageiro descobre em
//= que cidade esta lendo o proprio nome:
//
//=     set .@area$,strnpcinfo(2);
//=     if (.@area$ == "01") { set .@area$,"Prontera"; }
//
//= O `strnpcinfo(2)` e o pedaco depois do `#`. Ou seja o sufixo `01` NAO e
//= enfeite: e o dado que faz o NPC dizer "Prontera". Uma duplicata chamada
//= `#01b` faria a variavel valer "01b", nenhum `if` casaria, e o texto na
//= tela citaria "01b" como se fosse o nome da cidade.
//
//= O que salva e que `strnpcinfo(2)` le o `nd->name` (script.cpp:9276) e o
//= nome UNICO e o `nd->exname`, que e `strnpcinfo(3)`. Sao campos
//= diferentes - entao da para manter o `#01` na parte visivel e pendurar
//= um nome unico proprio depois do `::`. E o que a linha abaixo faz.
//
//= Desligar este arquivo devolve o Mensageiro a 164,304 sozinho, porque o
//= `disablenpc` que o esconde mora aqui.
//
//===========================================================================
-	script	AnomaliaMudaMensageiro	FAKE_NPC,{
	end;

OnInit:
	// O original nao tem `::`, entao o nome unico dele e a linha inteira
	// (CLAUDE.md secao 5). Desligar pelo nome de exibicao seria o erro
	// classico aqui - so que neste caso os dois sao o mesmo texto.
	disablenpc "Continental Messenger#01";
	end;
}

// A conta do "dia logico": devolve quantos dias inteiros se passaram desde
// a epoca, cortando o dia as 4 DA MANHA em vez da meia-noite - e o "apos as
// 4 horas da manha" do bRO.
//
// O deslocamento de fuso e DESCOBERTO, nao escrito: compara a hora local
// que o servidor da (gettime(DT_HOUR), America/Sao_Paulo desde 2026-08-16)
// com a hora UTC derivada do gettimetick(2). Assim nao ha constante para
// envelhecer se o horario de verao voltar.
function	script	AnomaliaDia	{
	.@epoch = gettimetick(2);
	.@utc = (.@epoch % 86400) / 3600;
	.@off = gettime(DT_HOUR) - .@utc;
	if (.@off > 12) .@off -= 24;
	if (.@off < -12) .@off += 24;
	return (.@epoch + .@off * 3600 - 14400) / 86400;
}

// O Manouro, ao lado da Maquina: troca Amago, e vende pressa.
prontera,167,304,4	script	Manouro::ManouroDim	750,{
	mes "[Manouro]";
	mes "Eu não giro a Máquina. Eu fico do lado dela e recolho o que as";
	mes "pessoas jogam fora sem saber o que é.";
	next;
	mes "[Manouro]";
	mes "Âmago Dimensional, por exemplo. Você tem ^0000FF" + countitem(7925) + "^000000.";
	next;
	switch (select("Trocar 30 Âmagos por um Ovo de Freeoni.:Trocar 1 Âmago por um ovo menor.:Reabrir a fenda de hoje.:Nada disso.")) {
	case 1:
		if (countitem(7925) < 30) {
			mes "[Manouro]";
			mes "Trinta. Nem vinte e nove.";
			close;
		}
		if (checkweight(9111, 1) == 0) {
			mes "[Manouro]";
			mes "Não cabe mais nada em você. Volte mais leve.";
			close;
		}
		delitem 7925, 30;
		getitem 9111, 1;
		mes "[Manouro]";
		mes "Trinta âmagos viram um ovo, e esse ovo vira a coisa mais";
		mes "faminta que você já teve. Boa sorte.";
		close;
	case 2:
		if (countitem(7925) < 1) {
			mes "[Manouro]";
			mes "Sem âmago, sem conversa.";
			close;
		}
		// Menu e tabela saem da MESMA fonte, montados aqui num laco - a
		// regra 4.11, escrita com sangue no Teleportador da Ordem.
		setarray .@ovo[0], 9105, 9096;
		setarray .@ovo[2], -1;
		.@menu$ = "";
		for (.@i = 0; .@ovo[.@i] != -1; .@i++) {
			// O separador vai ANTES, e nao depois: um ":" sobrando no fim
			// da string cria uma opcao vazia no menu, clicavel.
			if (.@i > 0) .@menu$ = .@menu$ + ":";
			.@menu$ = .@menu$ + getitemname(.@ovo[.@i]);
		}
		.@e = select(.@menu$) - 1;
		if (checkweight(.@ovo[.@e], 1) == 0) {
			mes "[Manouro]";
			mes "Não cabe mais nada em você. Volte mais leve.";
			close;
		}
		delitem 7925, 1;
		getitem .@ovo[.@e], 1;
		mes "[Manouro]";
		mes "Pegue. E alimente, pelo amor de tudo.";
		close;
	case 3:
		.@dia = callfunc("AnomaliaDia");
		if (anomalia_dia != .@dia) {
			mes "[Manouro]";
			mes "A sua fenda de hoje ainda está aberta. Vá lá e resolva —";
			mes "não cobro por trabalho que você ainda pode fazer.";
			close;
		}
		if (anomalia_reset == .@dia) {
			mes "[Manouro]";
			mes "Já reabri uma para você hoje. Duas no mesmo dia, nem por";
			mes "todo o dinheiro do mundo.";
			close;
		}
		mes "[Manouro]";
		mes "Reabrir custa ^0000FF2 Moedas Novas^000000. Uma vez por dia, e";
		mes "não me pergunte como eu faço.";
		next;
		if (select("Pagar as 2 Moedas Novas.:Deixa quieto.") == 2) {
			close;
		}
		if (countitem(30998) < 2) {
			mes "[Manouro]";
			mes "Você não tem duas Moedas Novas.";
			close;
		}
		delitem 30998, 2;
		anomalia_reset = .@dia;
		anomalia_dia = 0;
		mes "[Manouro]";
		mes "Pronto. Fale com o Varmunt de novo — para a fenda, hoje ainda";
		mes "não aconteceu.";
		close;
	case 4:
		mes "[Manouro]";
		mes "Então não atrapalhe a fila.";
		close;
	}
}
