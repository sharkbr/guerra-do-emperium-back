//===== Guerra do Emperium ==================================================
//= Zelador dos Guardioes - os dezenove castelos-museu
//===========================================================================
//= Fica em prt_gld 153,133, no Feudo de Prontera (movido de 136,66 em
//= 2026-08-14, a pedido do dono). Um NPC so cuida dos dezenove: o comando
//= `guardian` recebe o mapa como argumento, entao nao ha motivo para um NPC
//= por castelo.
//===========================================================================
//
//  O QUE ELE FAZ
//  =============
//
//  Poe a DEFESA dos dezenove castelos que nao sao o Kriemhild em 100 e enche
//  cada um com os oito guardioes oficiais dele. Com a escala de
//  src/custom/guardiao_do_castelo.hpp ligada, defesa 100 e o patamar mais
//  alto: 15 milhoes de HP, 50% de reducao de dano recebido, ASPD 178 e 50.000
//  de ATQ por guardiao (numeros corrigidos em 2026-08-14 - o cabecalho tinha
//  ficado com a PRIMEIRA calibragem, descartada na mesma noite em que foi
//  escrito; ver HISTORICO.md, "a revisao, na mesma noite", e a tabela
//  completa em conf/guerra/battle_guerra.txt).
//
//  Sao campos de treino, nao castelos. O Kriemhild continua sendo o castelo de
//  verdade - a defesa e a economia dele sao as que a guilda dona investir, e
//  este arquivo nao encosta nele.
//
//  POR QUE PRECISA DE NPC NOSSO
//  ============================
//
//  Porque o rAthena NAO invoca guardiao em castelo sem dono. No `OnAgitInit`
//  do npc/guild/agit_main.txt, castelo com `CD_GUILD_ID == 0` cai num
//  `killmonsterall` e ganha Evil Druid, Khalitzburg e companhia; o
//  `donpcevent "Guardian#...::OnSpawnGuardians"` so existe no ramo do castelo
//  COM dono. Castelo vazio, no rAthena, e castelo sem guardiao nenhum.
//
//  Como os dezenove estao desligados em npc/scripts_guild.conf (sem Emperium,
//  sem Kafra, sem Gerente, sem os monstros de castelo vazio), aquele ramo nem
//  roda - e quem enche os mapas e este arquivo.
//
//  GUARDIAO TEMPORARIO, E ISSO IMPORTA
//  ===================================
//
//  O `guardian` e chamado SEM o indice do fim, o que produz o que o
//  doc/script_commands.txt chama de guardiao temporario. Nao e economia de
//  digitacao; e o que mantem estes guardioes fora da contabilidade do castelo:
//
//  - nao ocupam os oito slots `CD_ENABLED_GUARDIAN`, entao o Gerente do
//    Castelo do Kriemhild continua vendo os slots dele como sempre;
//  - nao passam pelo `mob_guardian_guildchange` (src/map/mob.cpp), que APAGA
//    guardiao de castelo sem dono. Aquela funcao so e chamada para guardiao
//    com `visible` ligado, ou seja para os comprados de verdade. Com indice,
//    os nossos morreriam na primeira vez que alguem tocasse na dona do
//    castelo, e o sumico seria calado.
//
//  O que se perde e o respawn automatico e o `guardianinfo`. O respawn vem do
//  temporizador abaixo; o `guardianinfo` nao faz falta aqui.
//
//  A REPOSICAO E POR CONFERENCIA, NAO POR EVENTO DE MORTE
//  ======================================================
//
//  A cada minuto o `S_Repoe` percorre os 152 lugares e pergunta `unitexists()`
//  ao id que guardou. Quem nao existe mais volta.
//
//  O caminho obvio seria o rotulo de morte do proprio `guardian`, mas ele
//  responde "morreu um" sem dizer QUAL - e sao 152. Daria para descobrir pelo
//  `killedgid`, so que ele e parametro de JOGADOR: guardiao que morresse sem
//  matador anexado (veneno, `@killmonster`, desconexao no meio) nao voltaria,
//  e a falha seria calada. A conferencia nao depende disso e ainda conserta
//  sozinha um `guardian` que tenha falhado no spawn, porque nesse caso o id
//  guardado e 0 e `unitexists(0)` e falso.
//
//  O rotulo `OnGuardiaoMorreu` existe assim mesmo, e nao faz nada: ele e o que
//  da NOME aos nossos guardioes, para o `killmonster` do `S_Limpa` conseguir
//  distinguir os nossos de qualquer outro monstro do mapa.
//
//  O PORQUE DO `S_Limpa` - DUAS ARMADILHAS QUE ELE RESOLVE
//  ======================================================
//
//  1. `@reloadscript` NAO MATA MONSTRO. O script volta com o `.gid` zerado
//     enquanto os 152 guardioes continuam de pe no mapa - sem a limpeza, o
//     `S_Repoe` os veria como ausentes e criaria 152 copias por cima.
//
//  2. `OnAgitInit` pode disparar mais de uma vez, e a segunda duplicaria tudo
//     pelo mesmo motivo. Por isso o S_Limpa vem antes do S_Repoe la tambem.
//
//  A ESCALA E RESOLVIDA NO SPAWN, e e por isso que o `OnInit` nao invoca nada
//  (so limpa). Na primeira subida ele roda ANTES de o char-server entregar os
//  dados de castelo: a defesa lida seria 0 e os 152 nasceriam no patamar mais
//  baixo. Corrigir depois nao da - o `guild_castledatasave` do rAthena so manda
//  recalcular guardiao com `visible` ligado (src/map/guild.cpp, `case
//  CD_CURRENT_DEFENSE`), e os nossos sao temporarios de proposito.
//
//  Medido em 2026-08-13: com o `OnInit` invocando, a subida gerava DUAS rodadas
//  de 152 e o log passava a mostrar 16 avisos por mapa - o resultado final era
//  o certo, mas nao havia como provar, de fora do jogo, que a limpeza do meio
//  tinha funcionado. Com uma rodada so, "152 avisos, 8 por mapa" e prova.
//
//  ONDE A ESCALA MORA - E O QUE ELA NAO E
//  ======================================
//
//  Nenhum numero de HP, dano, velocidade ou reducao esta neste arquivo. Eles
//  vivem em conf/guerra/battle_guerra.txt (bloco "O GUARDIAO CRESCE COM A
//  DEFESA DO CASTELO") e sao aplicados em C++, no spawn de qualquer guardiao,
//  venha deste NPC ou do Gerente do Castelo. Este arquivo so decide ONDE ha
//  guardiao e com QUE defesa.
//
//  E a reducao nao e "resistencia a humano": monstro nao tem resistencia por
//  raca no rAthena. E o `damagetaken`, que corta tudo. O cabecalho de
//  src/custom/guardiao_do_castelo.hpp explica por que.
//
//  O QUE MULTIPLICA, E ENGANA SE FOR LIDO SOZINHO
//  ==============================================
//
//  Mapa de castelo tem o mapflag `gvg_castle` PERMANENTE (npc/mapflag/gvg.txt)
//  e o `mapdata_flag_gvg2` so olha mapflag - entao os nossos
//  `gvg_*_attack_damage_rate: 20` valem la dentro 24 horas por dia, com ou sem
//  Guerra do Emperium aberta, e valem tambem quando o alvo e monstro. O dano
//  que entra num guardiao de defesa 100 e `bruto x 0,20 x 0,50`, ou seja 10%.
//  Com 15 milhoes de HP sao 150 MILHOES de dano bruto por guardiao, vezes oito.
//
//  O lado bom disso e o teste: como o mapflag nao depende da guerra, o numero
//  medido numa terca a tarde e o mesmo numero da guerra de domingo.
//
//  O SPRITE
//  ========
//
//  `EP17_2_GUARDIAN_PARTS` - id 20679. Conferido neste cliente em 2026-08-13,
//  pela receita de sempre: existe no `npcidentity.lub` do nosso data.grf, o
//  `jobname.lub` o resolve para `ep17_2_guardian_parts` e o `.spr` e o `.act`
//  estao no GRF. O id esta ACIMA da faixa de NPC do `npcdb_checkid` (que para
//  em 19999) e funciona mesmo assim porque 20679 e um monstro do nosso
//  mob_db: o `npc_parseview` (src/map/npc.cpp) cai no `mobdb_search_aegisname`
//  quando a constante nao e de NPC, e o `status_set_viewdata` testa
//  `mobdb_checkid` ANTES de `npcdb_checkid`.
//===========================================================================

prt_gld,153,133,4	script	Zelador dos Guardiões::ZeladorDosGuardioes	EP17_2_GUARDIAN_PARTS,{
	mes "[Zelador dos Guardiões]";
	mes "Estes castelos não têm mais dono, nem Emperium, nem tesouro. Só os guardiões ficaram - e ninguém mandou que parassem.";
	next;
	mes "[Zelador dos Guardiões]";
	mes "Cada um deles está no limite do que a defesa de um castelo pode dar. Se você quer saber o que enfrentaria numa guerra de verdade, e o preço de deixar a guilda inimiga investir, o lugar é aqui.";
	next;
	mes "[Zelador dos Guardiões]";
	mes "A defesa investida decide tudo. A cada dez pontos, o guardião ganha um degrau:";
	mes "- vida, até quinze milhões;";
	mes "- resistência, até cinquenta por cento;";
	mes "- velocidade de ataque;";
	mes "- e força de golpe.";
	next;
	mes "[Zelador dos Guardiões]";
	mes "Estes daqui estão todos em cem de defesa. Não há degrau acima.";
	next;
	callsub S_Relatorio;
	close;

// Lista os dezenove com a defesa que o castelo tem AGORA e quantos guardioes
// estao de pe. Serve de sonda: se a defesa aparecer diferente de 100, o
// `SetCastleData` do OnAgitInit nao pegou, e a escala do guardiao esta saindo
// no patamar errado sem nada denunciar.
S_Relatorio:
	for (.@c = 0; .@c < .castelos; .@c++) {
		.@vivos = 0;
		for (.@i = 0; .@i < 8; .@i++)
			if (unitexists(.gid[.@c * 8 + .@i])) .@vivos++;

		mes "- " + GetCastleName(.mapa$[.@c]) + ": defesa "
			+ GetCastleData(.mapa$[.@c], CD_CURRENT_DEFENSE)
			+ ", " + .@vivos + " de 8 de pé.";
	}
	return;

OnInit:
	// Os dezenove castelos-museu, na ordem em que o agit_main.txt os lista.
	// As quatro linhas de cada bloco saem do `Gld_Guard_Template` de
	// npc/guild/agit_main.txt sem uma virgula de diferenca - e o mesmo lugar de
	// onde os guardioes do Kriemhild nascem, e e de proposito: no dia em que o
	// rAthena mudar uma posicao, a divergencia aparece num diff.
	//
	// `.tipo` e 1 = Soldado, 2 = Arqueiro, 3 = Cavaleiro, como no arquivo
	// original; o de-para para o id do monstro e o `.mob` logo abaixo.

	// aldeg_cas01 - Neuschwanstein

	setarray .mob[1],1287,1285,1286;

	// Defesa que os museus recebem. 100 e o teto do investimento e o patamar
	// mais alto da escala.

		debugmes "Zelador dos Guardioes: .py tem " + getarraysize(.py) + " e devia ter " + .total;

	// O OnInit LIMPA mas NAO invoca, e a diferenca custou uma subida para ficar
	// clara. Invocar aqui funciona, mas na primeira subida o char-server ainda
	// nao mandou os dados de castelo: a defesa lida e 0 e os 152 nascem no
	// patamar mais baixo. O OnAgitInit logo abaixo derruba e refaz, entao o
	// resultado final e o certo - so que a subida passa a ter DUAS rodadas de
	// spawn, e ai o log deixa de servir de prova. Com uma rodada so, "152 avisos
	// de guardiao, 8 por mapa" e uma conferencia de verdade.
	//
	// Quem monta e o OnAgitInit (subida) ou o temporizador (depois de um
	// `@reloadscript`, que nao dispara OnAgitInit). O preco e ate um minuto de
	// mapa vazio depois de um reloadscript, e so.
	//
	// A limpeza fica porque `@reloadscript` NAO mata monstro: os guardioes da
	// instancia anterior continuam de pe e o `.gid` voltou zerado, entao sem ela
	// o S_Repoe os veria como ausentes e criaria 152 copias por cima.
	callsub S_Limpa;
	initnpctimer;
	end;

// O char-server acabou de entregar os dados de castelo, e este e o unico
// momento em que se pode garantir que o `gc->defense` da memoria do map-server
// e o do banco. O S_Limpa continua aqui por seguranca: OnAgitInit pode ser
// disparado mais de uma vez, e sem ele a segunda vez duplicaria tudo.
OnAgitInit:
	callsub S_Defesa;
	callsub S_Limpa;
	callsub S_Repoe;
	end;

OnTimer60000:
	callsub S_Defesa;
	callsub S_Repoe;
	setnpctimer 0;
	end;

// Existe para dar nome aos nossos guardioes - ver o cabecalho. Quem repoe e a
// conferencia do temporizador, entao aqui nao ha nada a fazer.
OnGuardiaoMorreu:
	end;

// O `if` nao e economia de linha: sem ele, o temporizador mandaria dezenove
// gravacoes de dado de castelo ao char-server a cada minuto, para sempre. O
// `GetCastleData` le a memoria do map-server e nao custa pacote nenhum, entao
// em regime a funcao passa a ser de graca e so escreve quando algo saiu do
// lugar.
S_Defesa:
	for (.@c = 0; .@c < .castelos; .@c++)
		if (GetCastleData(.mapa$[.@c], CD_CURRENT_DEFENSE) != .defesa)
			SetCastleData .mapa$[.@c], CD_CURRENT_DEFENSE, .defesa;
	return;

// Derruba os NOSSOS guardioes e so os nossos. O `killmonster` sem o terceiro
// argumento nao dispara o rotulo dos que morrem, que e o que se quer aqui - o
// terceiro argumento `1` e que FARIA disparar, ao contrario do que o nome
// sugere (doc/script_commands.txt, `*killmonster`).
S_Limpa:
	for (.@c = 0; .@c < .castelos; .@c++)
		killmonster .mapa$[.@c], "ZeladorDosGuardioes::OnGuardiaoMorreu";
	deletearray .gid[0], .total;
	return;

// Repoe quem faltar. Idempotente de proposito: pode ser chamado a qualquer
// hora, quantas vezes for, e so gasta um `guardian` por lugar vazio.
S_Repoe:
	for (.@n = 0; .@n < .total; .@n++) {
		if (unitexists(.gid[.@n]))
			continue;

		.@mob = .mob[.tipo[.@n]];
		.gid[.@n] = guardian(.mapa$[.@n / 8], .px[.@n], .py[.@n],
			getmonsterinfo(.@mob, MOB_NAME), .@mob,
			"ZeladorDosGuardioes::OnGuardiaoMorreu");
	}
	return;
}
