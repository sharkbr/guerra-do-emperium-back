// O teste do guardiao. `close2` ja foi chamado por quem convocou;
// daqui pra frente o jogador nao ve mais janela de dialogo ate o
// proximo `mes`.
// ------------------------------------------------------------
S_TesteDoGuardiao:
	warp .mapaTeste$, .pxTeste, .pyTeste;
	sleep2 3000;
	if (!playerattached()) return;

	mapannounce .mapaTeste$, "Suad: Invocando guardião em 10 segundos.", bc_map, 0xADD8E6;
	sleep2 10000;
	if (!playerattached()) return;

	// Idempotente por proposito: uma segunda tentativa, ou um
	// @reloadscript no meio de uma, nao pode empilhar guardiao.
	// Mesma razao do `S_Limpa` de guardioes_dos_castelos.txt.
	killmonster .mapaTeste$, "ComandanteMaram::OnGuardiaoTesteMorreu";

	monster .mapaTeste$, .pxGuardiao, .pyGuardiao, "Guardião Reforçado", 1287, 1,
		"ComandanteMaram::OnGuardiaoTesteMorreu";

	// O ATQ, o ASPD (AttackDelay/AttackMotion) e a precisao (Dex) vem
	// de db/guerra/mob_db_guerra.yml, lidos certo no spawn (nao da
	// pra fazer por setunitdata - ver o cabecalho, secao "o bug que
	// o setunitdata escondia"). Aqui so HP e reducao, que SAO
	// efetivos por setunitdata.
	.@n = getmapunits(BL_MOB, .mapaTeste$, .@gid[0]);
	if (.@n > 0) {
		.@mobgid = .@gid[0];
		setunitdata(.@mobgid, UMOB_MAXHP, 15000000);
		setunitdata(.@mobgid, UMOB_HP, 15000000);
		setunitdata(.@mobgid, UMOB_DAMAGETAKEN, 10);
	}

	mapannounce .mapaTeste$, "Suad: Você tem 60 segundos pra derrubar o guardião!", bc_map, 0xADD8E6;

	.@venceu = false;
	for (.@i = 0; .@i < 60; .@i++) {
		sleep2 1000;
		if (SalaSecretaOrdem >= 10) {
			.@venceu = true;
			break;
		}
		if (!playerattached())
			break;
		if (strcharinfo(3) != .mapaTeste$)
			break; // morreu (nosave + SavePoint ja tirou do mapa) ou saiu andando
	}

	// Idempotente de novo: mata o que sobrou, ganhe ou perca.
	killmonster .mapaTeste$, "ComandanteMaram::OnGuardiaoTesteMorreu";

	if (!playerattached())
		return;

	if (.@venceu) {
		dispbottom "Parabéns!! Você derrotou um guardião reforçado!!";
		sleep2 3000;
		if (playerattached())
			warp "cmd_in02", .retornoX, .retornoY;
		return;
	}

	if (strcharinfo(3) == .mapaTeste$) {
		// Ainda no mapa e vivo: os 60 segundos acabaram sem matar.
		// Morte de verdade ja tirou o personagem do mapa antes de
		// chegar aqui, e por isso nao repete a mensagem.
		dispbottom "Tente novamente!";
		sleep2 3000;
		if (playerattached())
			warp "cmd_in02", .retornoX, .retornoY;
	}
	return;

OnGuardiaoTesteMorreu:
	// Roda com o RID do PERSONAGEM QUE MATOU (doc/script_commands.txt,
	// *monster) - por isso da pra escrever direto na variavel dele.
	if (SalaSecretaOrdem == 9)
		SalaSecretaOrdem = 10;
	end;

OnPCDieEvent:
	// Dispara em TODA morte do servidor - o teste de mapa e o que
	// mantem isto barato fora do teste. Mesma receita do
	// `arena_de_combate.txt`.
	if (strcharinfo(3) != .mapaTeste$)
		end;
	killmonster .mapaTeste$, "ComandanteMaram::OnGuardiaoTesteMorreu";
	end;

	.pxGuardiao = 39;
	.pyGuardiao = 45;
	.retornoX = 77;
	.retornoY = 80;
	end;
