//===== Guerra do Emperium ===================================
//= Mestre de UP
//===== Onde =================================================
//= prontera 160,187 (praca da fonte, ao lado do Mestre de Classe)
//===== Descricao ============================================
//= Concede 50 niveis de base e 50 de classe de uma vez, apos duas
//= confirmacoes do jogador. Come a Maca da Inocencia de quem tiver
//= uma, mas nao a exige.
//===== Notas ================================================
//= A conversa tem TRES momentos, e a Maca so e tocada no terceiro:
//=  1) a oferta;
//=  2) o preco (a fala que menciona a Maca) e a confirmacao;
//=  3) a cobranca - e aqui, e so aqui, que o `countitem` roda.
//=
//= A Maca e PRECO COBRADO, NAO CONDICAO: quem a tem, perde; quem nao
//= a tem, recebe os niveis do mesmo jeito. Nenhum caminho do terceiro
//= momento recusa o jogador. Foi decidido assim de proposito, em
//= 2026-08-01 - antes disso a falta da Maca era recusa.
//=
//= CONSEQUENCIA a encarar de olho aberto: com isso o NPC deixa de ter
//= QUALQUER limite. Antes o inventario era o freio - uma Maca, um uso.
//= Agora nada impede reconversar: sao +50/+50 por conversa, e duas
//= conversas levam do nivel 1 ao 101. Na pratica ele virou "nivel
//= maximo a pedido, repetivel". Se um dia isso tiver de ser contido, o
//= lugar e aqui, e o instrumento e uma variavel de personagem
//= (`up_usos`) ou um cooldown - nao a Maca, que nao freia mais nada.
//=
//= O visual e o 10006 (JT_ACADEMY_MASTER, sprite `academy_master`),
//= conferido no npcidentity.lub, no jobname.lub e no data.grf DESTE
//= cliente - o .spr e o .act estao la. Nao confundir com o caso do
//= Mestre de Classe: la o 10605 caiu porque a faixa 10xxx deste
//= cliente para em 10508; o 10006 esta bem dentro dela.
//=
//= Os textos estao SEM ACENTO de proposito. O cliente roda em
//= langtype 0 (Coreia) e ainda nao testamos como ele desenha byte
//= acentuado. Ver PENDENCIAS.md, secao "Acentuacao no dialogo".
//=
//= A "Maca da Inocencia" e o item 30999 (Maca_Da_Inocencia), definido
//= em db/guerra/item_db.yml - arquivo nosso, encadeado pelo `Footer:
//= Imports:` de db/re/item_db.yml. NAO fica em db/import/item_db.yml:
//= o .gitignore do proprio rAthena ignora /db/import (linha 67), e um
//= item definido la sumiria num clone limpo.
//=
//= Do lado do cliente, a entrada [30999] do itemInfo.lua e posta pelo
//= ferramentas/instala_item.py - e la que moram o nome exibido e a
//= descricao. O cliente le esse arquivo so na inicializacao, entao
//= mexer nele exige fechar e reabrir; @reloaditemdb nao alcanca.
//= O item nao tem arte propria: o `resourceName` da entrada e copiado
//= byte a byte do da Maca comum, entao o desenho e o mesmo.
//=
//= NAO tem trava de uso unico, de proposito: pode ser usado quantas
//= vezes o jogador quiser - e, desde que a Maca deixou de ser
//= condicao, sem nada que o limite. Ver a CONSEQUENCIA la em cima.
//=
//= Por que atribuir BaseLevel/JobLevel e nao dar experiencia: o
//= pc_setparam (src/map/pc.cpp, SP_BASELEVEL e SP_JOBLEVEL) ja faz
//= o servico completo - trava no nivel maximo da classe, entrega os
//= pontos de atributo e de traco de cada nivel ganho, entrega os
//= pontos de habilidade, recalcula o personagem e avisa o grupo.
//=
//= Dois efeitos colaterais herdados desse mesmo codigo:
//=  1) a experiencia acumulada rumo ao proximo nivel e ZERADA (o
//=     setparam faz base_exp = 0 / job_exp = 0). Quem estava a 1%
//=     do nivel perde esse 1%. Irrelevante perto de +50.
//=  2) quem ja esta no teto da classe nao ganha nada, e o NPC nao
//=     reclama - a trava e silenciosa.
//============================================================

prontera,160,187,4	script	Mestre de UP	10006,{

	mes .npc$;
	mes "Ola viajante!";
	mes "Voce deve ter vindo falar comigo porque quer obter level mais rapido certo?";

	if (select(" ~ Sim, quero: ~ ^777777Nao, deixa pra la^000000") != 1)
		close;

	next;
	mes .npc$;
	mes "Meu metodo de UP consiste em uma ^0055FFilusao atemporal^000000 onde voce realiza todo o treinamento de uma vez so.";
	mes " ";
	// "se voce tiver uma": a fala nao pode prometer uma exigencia que o
	// terceiro momento nao cumpre mais, ou quem chega sem a Maca recusa
	// aqui uma oferta que ele teria recebido.
	mes "Mas ele consome sua ^BB0000Maca da Inocencia^000000 de uma vez por todas - se voce tiver uma.";
	mes " ";
	mes "Tem certeza que deseja continuar?";

	if (select(" ~ Sim, tenho certeza: ~ ^777777Nao, deixa pra la^000000") != 1)
		close;

	// Terceiro momento: a cobranca. A Maca e levada quando existe, mas
	// nao e condicao - ver o cabecalho. Nenhum caminho daqui recusa.
	next;
	mes .npc$;
	mes "Entao me entregue sua ^BB0000Maca da Inocencia^000000.";
	mes " ";
	mes "...";

	if (countitem(.Maca) > 0) {
		delitem .Maca,1;
		mes " ";
		mes "Voce a trouxe. Ela vai junto - e assim tem de ser.";
	} else {
		mes " ";
		mes "Voce nao tem nenhuma, viajante. Nao faz mal: a ilusao se abre com ou sem ela.";
	}

	mes " ";
	mes "Feche os olhos. Quando abri-los, o treinamento ja tera acontecido.";

	BaseLevel += .Niveis;
	JobLevel += .Niveis;

	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	close;

OnInit:
	.npc$ = "[Mestre de UP]";

	// Quantos niveis sao concedidos, de base e de classe.
	.Niveis = 50;

	// Maca da Inocencia (db/guerra/item_db.yml). Consumida a cada uso.
	.Maca = 30999;
	end;
}
