//===== Por que ele ja esteve desligado ======================
//= Ele passou o commit 7e81b45 desligado por `pvp_nocalcrank`,
//= por medo de um choque de timers. O medo era do bloco de
//= EXPULSAO que vem junto: quando o pvp_point fica NEGATIVO, o
//= pc_dead arma um timer de 1 segundo que manda o jogador para o
//= `save_point`. E esse timer e armado ANTES do nosso - o
//= OnPCDieEvent so dispara no fim do status_damage
//= (src/map/status.cpp, "Always run NPC scripts for players
//= last") -, entao num empate de 1000ms contra 1000ms quem vence
//= e o do servidor, e o morto cairia no ponto de retorno.
//=
//= DUAS COISAS FAZEM O CHOQUE NAO ACONTECER, e e por isso que o
//= rank pode voltar:
//=
//=   1) O pvp_point NAO CHEGA A NEGATIVO AQUI. Ele comeca em 5 e
//=      a morte tira 5: a primeira morte para em ZERO, e o teste
//=      do pc_dead e `< 0`. Negativar exigiria morrer DUAS VEZES
//=      na mesma visita, e o OnPCDieEvent abaixo expulsa em TODA
//=      morte - a segunda so vem depois de reentrar, e reentrar
//=      devolve os 5 pontos.

//=      1000ms do servidor: nos chegamos primeiro. E quando o
//=      timer do servidor enfim dispara, ele nao faz nada - o
//=      pc_respawn comeca com `if (!pc_isdead(sd)) return;` e o
//=      `recovery` do nosso script ja levantou o jogador.
//=
//= (1) e o motivo, (2) e a rede. Levar `.Espera` de volta para
//= 1000 ou mais desfaz a rede: manter abaixo de 1000.

prontera,147,180,6	script	Arena de Combate	10028,{
	mes "[Arena de Combate]";
	mes "Deseja entrar na gloriosa e";
	mes "eterna arena de combate?";
	next;
	if (select("Sim:Não") == 2) {
		mes "[Arena de Combate]";
		mes "A arena fica aqui, esperando.";
		close;
	}
	mes "[Arena de Combate]";
	mes "Boa sorte!";
	close2;
	warp "pvp_n_1-5",0,0;
	end;

OnInit:
	.Mapa$ = "pvp_n_1-5";   // a arena
	.Volta$ = "prontera";   // e para onde o morto volta
	.VoltaX = 155;
	.VoltaY = 183;
	.Espera = 800;          // ms entre a morte e o teleporte. Abaixo dos
	                        // 1000ms do rAthena de proposito: ver a secao
	                        // "Por que ele ja esteve desligado".

	// O fantasma da arena - a conta de bot do openkore. O grupo e o
	// Id 20 de conf/guerra/groups_guerra.yml; 1869 e o Flame Skull do
	// db/re/mob_db.yml. Ver a secao "O visual do fantasma" no
	// cabecalho antes de mexer em qualquer um dos dois.
	.Fantasma = 20;
	.Caveira = 1869;

	// Nao ha `disablenpc` nenhum aqui desde 2026-08-12, e a ausencia e
	// deliberada: 147,180 e celula vazia. Ate aquele dia esta porta
	// ficava em 154,187, em cima do Guia de Prontera, e precisava
	// desliga-lo. Ver a secao "Este arquivo nao toca NPC de terceiros"
	// no cabecalho antes de por um de volta.
	end;

OnPCLoginEvent:
	// A caveira do fantasma, ligada uma vez por sessao. O teste de
	// grupo corta na primeira linha para todo jogador de verdade,
	// entao este rotulo nao custa nada no login de ninguem.
	//
	// Por que aqui e nao numa lista de personagens: o pedido era que
	// o visual fosse da CONTA. Qualquer char do grupo Fantasma nasce
	// caveira, e trocar o personagem do bot nao mexe neste arquivo.
	//
	// Reaplicar por cima de um status que ja veio salvo do banco e
	// de graca: o `transform` encerra o anterior antes de comecar o
	// novo (buildin_montransform, src/map/script.cpp).
	if (getgroupid() != .Fantasma)
		end;
	transform .Caveira, INFINITE_TICK;
	end;

OnPCDieEvent:
	// Dispara em TODA morte do servidor, inclusive por monstro em
	// qualquer mapa - o teste de mapa e o que mantem isto barato
	// fora da arena.
	if (strcharinfo(3) != .Mapa$)
		end;

	// O `sleep2` deixa a tela mostrar a morte antes do teleporte.
	// Sem ele o jogador e arrancado no mesmo quadro em que cai e nao
	// entende o que aconteceu. Fica logo abaixo do segundo que o
	// pc_respawn_timer do rAthena usa, para vencer a corrida com ele.
	.@cid = getcharid(0);
	sleep2 .Espera;

	// Deslogou durante a espera, ou clicou "Reiniciar" e ja saiu por
	// conta propria. Nos dois casos nao ha o que teleportar, e fazer
	// mesmo assim arrancaria o jogador de onde ele ja esta vivo.
	if (!playerattached())
		end;
	if (strcharinfo(3) != .Mapa$)
		end;

	// `recovery` levanta E cura, que e o que o respawn normal faz (o
	// pc_respawn chama pc_setrestartvalue). Levantar ANTES de
	// teleportar, para o cliente nao receber um cadaver andando.
	recovery 0, .@cid, 1;

	// E a caveira do fantasma volta. A morte e a UNICA coisa que
	// derruba o SC_MONSTER_TRANSFORM, e ela derruba antes deste
	// rotulo rodar: o status_change_clear do status_damage limpa
	// primeiro, e so depois o rAthena chama os eventos de jogador
	// ("Always run NPC scripts for players last", src/map/status.cpp).
	// Ou seja, quando esta linha executa nao ha status para
	// atrapalhar - e ela precisa vir DEPOIS do `recovery`, porque
	// status nenhum inicia em quem esta morto.
	if (getgroupid() == .Fantasma)
		transform .Caveira, INFINITE_TICK;

	warp .Volta$, .VoltaX, .VoltaY;
	end;
}
