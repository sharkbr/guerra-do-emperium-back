gef_fild10,242,202,0	script	Dimensional Gorge Piece	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Batalha dos Orcs";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "Só quem está entre os níveis ^ff000030 e 80^000000 pode entrar nesta masmorra.";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "Grupo conferido. Quer reservar a entrada para a "+.@md_name$+"?";
			next;
			switch(select("Reservar a "+.@md_name$+":Entrar na masmorra:Cancelar")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "Grupo: "+ getpartyname(.@party_id);
					mes "Líder: "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+" ^000000 - a reserva falhou.";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000 - tentando reservar a entrada";
				mes "Depois de reservar, escolha 'Entrar na masmorra' no menu para entrar na "+.@md_name$+".";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name$ + " Entrar na memória:Cancelar") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "Se a masmorra já estiver aberta, você pode entrar.";
		next;
		if(select("Entrar na "+.@md_name$+":Cancelar") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ffTodo o registro da Batalha dos Orcs foi apagado. Dá para abrir ou entrar de novo.^000000";
		erasequest 12059;
		close;
	}
	mes "Para abrir a masmorra é preciso ser o líder do grupo e ter pelo menos 2 pessoas nele.";
	close;
L_Enter:
	switch(instance_enter("Batalha dos Orcs")) {
	case IE_OTHER:
		mes "Alguma coisa deu errado, e nem eu sei o quê.";
		close;
	case IE_NOINSTANCE:
		mes "A memória da Batalha dos Orcs não existe.";
		mes "A memória foi desfeita pelo líder do grupo, ou o tempo dela acabou. Tente de novo daqui a 2 horas.";
		close;
	case IE_NOMEMBER:
		mes "Só quem está no grupo pode entrar na memória.";
		close;
	case IE_OK:
		mapannounce "gef_fild10", getpartyname( getcharid(1) ) +" - o aventureiro "+strcharinfo(0)+" entrou na Batalha dos Orcs.",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}

			}
		}

			areamonster .@map$,41,91,51,81,"High Orc",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
