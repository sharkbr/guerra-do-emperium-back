// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[Edgar]";
		mes "Vai para Alberta de novo? Te faço o mesmo desconto e cobro só 250 zeny, igual à última vez.";
		mes "Que tal?";
		next;
		if (select("Beleza~!:Por que você é tão bonzinho comigo?!") == 1) {
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Hum...";
				mes "Isso não";
				mes "dá.";
				mes "Por que não vai";
				mes "arrumar mais dinheiro?";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "É o meu jeito. E também porque esse seu sorriso endiabrado me lembra o meu querido filho loiro, que saiu de casa anos atrás para ser marinheiro no próprio navio. Que Deus o proteja, onde quer que esteja.";
		close;
	}
	mes "[Edgar]";
	mes "A minha cidade, Izlude, é ligada a Alberta pelo porto do oeste. O vaivém entre as duas é tanto que eu quase virei albertano. Hehehe~";
	next;
	mes "[Edgar]";
	mes "Tem um sujeito que eu conheço bem, o Phelix, que mora em Alberta. Esse cara é muito pão-duro... Cobra por tudo!";
	next;
	mes "[Edgar]";
	mes "Mas ele é gente boa e gosta de ajudar os outros. Tem bom coração e te dá uma força se você pagar o preço dele.";
	next;
	mes "[Edgar]";
	mes "Ultimamente o pessoal de Alberta anda dizendo que ele quer mesmo ajudar e que essa história de pedir Jellopy é só fachada. Bom, vale a pena ver o que ele tem a oferecer.";
	next;
	if (select("Pode me dizer como chegar a Alberta?:Encerrar conversa") == 1) {
		mes "[Edgar]";
		mes "Hein? Bom, dá para ir a pé. Mas, se tiver dinheiro, eu sugiro pegar um navio.";
		next;
		if (select("Certo, entendi.:Mas eu cansei de andar e estou duro!") == 1) {
			mes "[Edgar]";
			mes "Beleza, se cuida~";
			close;
		}
		mes "[Edgar]";
		mes "Tá...";
		mes "Não quer andar E está sem dinheiro, mas ainda quer ir para lá? Ai, ai...";
		next;
		mes "[Edgar]";
		mes "Tá bom, tá bom. Como sou capitão de navio, posso te levar por um preço mais baixo. Que tal 250 zeny?";
		next;
		if (select("Beleza~!:Ah, que roubo!!") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Hum...";
				mes "Isso não";
				mes "dá.";
				mes "Vai lá arrumar";
				mes "mais um pouco.";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "Puxa vida,";
		mes "se você acha";
		mes "que isso é roubo...";
		close;
	}
	mes "[Edgar]";
	mes "Tá, beleza.";
	mes "Até mais~";
	close;
}
