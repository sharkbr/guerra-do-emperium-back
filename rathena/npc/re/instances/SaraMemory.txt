dali,139,118,4	script	Dimensional Device#sara	10007,{
	if (BaseLevel < 99) {
		mes "[Aventureiro Leon]";
		mes "Ei! Você é novo demais para brincar com isso! Volte quando estiver pelo menos no ^0000FFNível 99^000000!";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[Aventureiro Leon]";
		mes "Antes de sair brincando com isso, por que não vem aqui conversar comigo?";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Memórias de Sarah";

		if (!.@party_id) {
			mes "[Aventureiro Leon]";
			mes "Antes de entrar você precisa montar um grupo! Não sabe como? Digite /organize NOMEDOGRUPOAQUI.";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Ligar o dispositivo dimensional.:Usar o dispositivo dimensional.:Cancelar.";
		else
			set .@menu$, ":Usar o dispositivo dimensional.:Cancelar.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "Nome do Grupo: "+.@p_name$;
				mes "Líder do Grupo: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- Falha na reserva!";
				close;
			}
			mes "^FF0000O dispositivo dimensional liga sem problema. Use o dispositivo para entrar na fenda do espaço-tempo.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "Ocorreu um erro desconhecido.";
				close;
			case IE_NOINSTANCE:
				mes "A masmorra memorial "+.@md_name$+" não existe.";
				mes "O líder do grupo ainda não gerou a masmorra.";
				close;
			case IE_NOMEMBER:
				mes "Só os membros registrados podem entrar na instância "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+", do grupo "+.@p_name$+", entrou na instância "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000O líder do grupo não ligou o dispositivo dimensional.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[Aventureiro Leon]";
		mes "Acho que umas 20 horas depois da última vez que você entrou na fenda ela vai estar estável o bastante para você atravessar de novo.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ Memórias de Sarah ] O tempo de espera foi zerado. Já dá para usar o dispositivo dimensional de novo.^000000";
		close;
	}
}
