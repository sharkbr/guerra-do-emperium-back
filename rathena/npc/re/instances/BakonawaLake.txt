ma_scene01,174,179,4	script	Taho	541,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Você está carregando itens demais. Volte depois de tirar alguns.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Você está carregando peso demais. Volte depois de aliviar a carga.";
		close;
	}
	if (BaseLevel < 140) {
		mes "[Taho]";
		mes "Ei. Este lugar é perigoso demais para você. Não tem como se proteger aqui. Saia e volte para a vila agora mesmo!";
		close;
	}
	if (malaya_bakona1 < 7 && malaya_bakona2 < 15) {
		mes "[Taho]";
		mes "Aquele Bakonawa vai matar todos nós. Ah, precisamos abatê-lo, mas infelizmente não estamos preparados.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$, "Lago de Bakonawa";

	set .@baku_time, checkquest(12278,PLAYTIME);
	if (.@baku_time == -1) {
		if (checkquest(12279) == -1) setquest 12279;
	}
	if (!.@party_id) {
		mes "[Taho]";
		mes "Não há ninguém que possa ir com você? E mesmo se quiser ir sozinho, volte depois de montar um grupo.";
		close;
	}
	if (.@baku_time == -1) {
		if (is_party_leader() == true) {
			mes "[Taho]";
			mes "Você é o líder dos caçadores de Bakonawa? O caminho não está bom, então temos de descer o penhasco por uma corda. Tudo bem?";
			next;
			set .@i, select("Trança uma corda.:Vou descer agora.:Cancelar.");
		} else {
			mes "[Taho]";
			mes "Você veio abater o Bakonawa. O líder do seu grupo tem de me ajudar a trançar a corda da descida.";
			next;
			set .@i, select(":Vou descer agora.:Cancelar.");
		}
		switch(.@i) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "[Taho]";
				mes "Ah, a corda caiu. Vou ter de fazer outra.";
				close;
			}
			mes "[Taho]";
			mes "Estou trançando; quando eu terminar você pode descer.";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "[Taho]";
				mes "Alguma coisa deu errado, e nem eu sei o quê.";
				close;
			case IE_NOINSTANCE:
				mes "[Taho]";
				mes "A corda ainda não está trançada. Espere um pouco.";
				close;
			case IE_NOMEMBER:
				mes "[Taho]";
				mes "Civis não entram.";
				close;
			case IE_OK:
				mapannounce "ma_scene01","O aventureiro "+strcharinfo(0)+" do grupo "+getpartyname(.@party_id)+" está entrando na masmorra, "+.@md_name$+".",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
				setquest 12278;
				//warp "1@ma_b",64,51;
				close;
			}
		case 3:
			close;
		}
	} else if (.@baku_time == 0 || .@baku_time == 1) {
		mes "[Taho]";
		mes "A corda que descia até o Lago já arrebentou. Estamos limpando a área, então volte daqui a algum tempo.";
		close;
	} else if (.@baku_time == 2) {
		mes "[Taho]";
		mes "A área já está limpa. Agora dá para descer de novo, com uma corda.";
		erasequest 12278;
		close;
	}
}
