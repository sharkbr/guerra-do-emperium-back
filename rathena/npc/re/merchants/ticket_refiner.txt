// Main NPC :: safety_Ref_NPC
//============================================================
prontera,184,177,6	script	Refine Master	851,{
	disable_items;
	setarray .@cert_weapon[0],
		6864,	// Guarantee_Weapon_19Up
		6875,	// Guarantee_Weapon_18Up
		6874,	// Guarantee_Weapon_17Up
		6873,	// Guarantee_Weapon_16Up
		6872,	// Guarantee_Weapon_15Up
		6871,	// Guarantee_Weapon_14Up
		6870,	// Guarantee_Weapon_13Up
		6584,	// Guarantee_Weapon_12Up
		6238,	// Guarantee_Weapon_11Up
		6228,	// Guarantee_Weapon_9Up
		6229,	// Guarantee_Weapon_8Up
		6230,	// Guarantee_Weapon_7Up
		6231,	// Guarantee_Weapon_6Up
		6456;	// Guarantee_Weapon_5Up
	setarray .@cert_armor[0],
		6865,	// Guarantee_Armor_19Up
		6881,	// Guarantee_Armor_18Up
		6880,	// Guarantee_Armor_17Up
		6879,	// Guarantee_Armor_16Up
		6878,	// Guarantee_Armor_15Up
		6877,	// Guarantee_Armor_14Up
		6876,	// Guarantee_Armor_13Up
		6585,	// Guarantee_Armor_12Up
		6239,	// Guarantee_Armor_11Up
		6232,	// Guarantee_Armor_9Up
		6233,	// Guarantee_Armor_8Up
		6234,	// Guarantee_Armor_7Up
		6235,	// Guarantee_Armor_6Up
		6457;	// Guarantee_Armor_5Up
	setarray .@cert_level[0],19,18,17,16,15,14,13,12,11,9,8,7,6,5;
	.@size_cert = getarraysize(.@cert_weapon);

	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@cert_weapon[.@i]) > 0 || countitem(.@cert_armor[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}

	if (.@check == 0) {
		mes "[Mestre do Refino]";
		mes "Olá!";
		mes "Tudo bem?";
		mes "Eu sou especialista";
		mes "em refinar equipamento,";
		mes "mas não trabalho mais.";
		next;
		switch(select("Então vou seguir meu caminho.:Hmm... fiquei curioso.")) {
		case 1:
			mes "[Mestre do Refino]";
			mes "Se cuide, aventureiro.";
			close;
		case 2:
			mes "[Mestre do Refino]";
			mes "Na verdade, de vez em quando eu ainda refino para quem chega com um ^006400Pergaminho^000000...";
			mes "Tchau, tchau~!";
			close;
		}
	}
	emotion ET_SURPRISE;
	mes "[Mestre do Refino]";
	mes "Saudações!";
	mes "Eu refino a peça até o ^006400mesmo nível do seu pergaminho^000000.";
	mes "E pode ficar tranquilo: não há chance nenhuma de quebrar.";
	next;
	if(select("Volto depois.:Refinar com o pergaminho.") == 1) {
		mes "[Mestre do Refino]";
		mes "Certo.";
		mes "Volte quando quiser.";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Qual peça você quer refinar?";
	next;
	setarray .@position$[1],"Cabeça (topo)","Armadura","Mão esquerda","Mão direita","Capa","Sapatos","Acessório 1","Acessório 2","Cabeça (meio)","Cabeça (baixo)";
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ += (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : .@position$[.@i] + "- [Vazio]") + ":";

	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Mestre do Refino]";
		mes "Você precisa estar usando a peça que quer refinar.";
		close;
	}
	if (!getequipisenableref(.@part)) {
		emotion ET_OTL;
		mes "[Mestre do Refino]";
		mes "Ah, sinto muito.";
		mes "Esta peça não pode ser refinada.";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_WEAPON ){
		switch( getequipweaponlv( .@part ) ){
			case 1:
			case 2:
			case 3:
			case 4:
				copyarray .@tickets[0], .@cert_weapon[0], .@size_cert;
				.@type$ = "Arma";
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_ARMOR ){
		switch( getequiparmorlv( .@part ) ){
			case 1:
				copyarray .@tickets[0], .@cert_armor[0], .@size_cert;
				.@type$ = "Armadura";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	.@check = 0;
	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@tickets[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}
	if (.@check == 0) {
		emotion ET_THINK;
		mes "[Mestre do Refino]";
		mes "Para refinar esta ^006400"+.@type$+"^000000, volte com um ^006400Pergaminho de "+.@type$+"^000000.";
		mes "Até mais!";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Escolha qual ^006400Pergaminho de "+.@type$+"^000000 você quer usar.";
	next;
	.@menu$ = "";
	for ( .@i = 0; .@i < .@size_cert; ++.@i )
		.@menu$ += getitemname(.@tickets[.@i]) + ":";
	.@select = select(.@menu$)-1;
	.@ticket_lv = .@cert_level[.@select];
	.@ticket_id = .@tickets[.@select];

	if (countitem(.@ticket_id) == 0) {
		emotion ET_QUESTION;
		mes "[Mestre do Refino]";
		mes getitemname(.@ticket_id)+" não está na sua bolsa. Será que ficou no armazém?";
		mes "Confira de novo, por favor.";
		mes "Até mais!";
		close;
	}
	if (getequiprefinerycnt(.@part) >= .@ticket_lv) {
		emotion ET_PROFUSELY_SWEAT;
		mes "[Mestre do Refino]";
		mes "^8B4513Esta peça já está refinada até onde o seu pergaminho alcança.^000000";
		mes "Traga uma peça refinada abaixo do que o pergaminho dá.";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Vou refinar a ^006400"+getequipname(.@part)+"^8B4513 até o +"+.@ticket_lv+"^000000, usando o ^006400"+getitemname(.@ticket_id)+"^000000.";
	mes "Posso seguir?";
	next;
	if(select("Não.:Sim.") == 1) {
		emotion ET_THINK;
		mes "[Mestre do Refino]";
		mes "Ah, mudou de ideia.";
		mes "Tudo bem.";
		mes "Pode voltar depois.";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Ótimo.";
	mes "Como quiser!";
	mes "Eu tenho um jeito bem particular de refinar...";
	mes ".......ka bum!";
	specialeffect EF_SUI_EXPLOSION;
	if (countitem(.@ticket_id) < 1) {
		next;
		mes "Erro!";
		mes "Por favor, avise a administração.";
		close;
	}
	delitem .@ticket_id,1;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes "[Mestre do Refino]";
		emotion ET_FRET;
		mes "Espera um pouco...";
		mes "Acha que eu sou bobo?!";
		mes "Você trocou a peça enquanto eu não estava olhando! Sai daqui!";
		close;
	}
	successrefitem .@part, .@ticket_lv - getequiprefinerycnt(.@part);
	next;
	emotion ET_DELIGHT;
	mes "[Mestre do Refino]";
	mes "Pronto, aqui está~";
	mes "Muito bem, ^0000FF"+strcharinfo(0)+"^000000!";
	mes "Parabéns pela sua reluzente "+.@type$+".";
	mes "Ficou ÓTIMO!";
	mes "Até a próxima~!";
	close;
}
