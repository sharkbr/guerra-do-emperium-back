// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[Oficial de Canal]";
	mes "Izlude é um lugar bem movimentado, não é?";
	mes "Estou aqui para deixar Izlude mais tranquila!";
	mes "Posso enviar você para outra cópia de Izlude.";
	mes "Quer ir para outra cópia?";
	next;
	set .@i, select(getarg(0)+":Deixa pra lá");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
