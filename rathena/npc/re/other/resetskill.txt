izlude,127,175,3	script	Hipnotizadora#novice	124,{
	mes "[Hipnotizadora]";
	mes "Saudações, aventureiro.";
	mes "Sou membro da Academia";
	mes "de Hipnose, enviada a Izlude";
	mes "para oferecer o serviço de Reset";
	mes "a certos personagens de Primeira Classe por um preço ótimo: de graça!";
	next;
	mes "[Hipnotizadora]";
	mes "Por enquanto eu ofereço resets";
	mes "de atributos e habilidades sem limite,";
	mes "mas há duas condições que";
	mes "precisam ser cumpridas. Primeiro, você";
	mes "precisa estar abaixo do ^FF0000Nível de Base 50^000000.";
	next;
	mes "[Hipnotizadora]";
	mes "Segundo, você precisa ser";
	mes "^FF0000Espadachim, Noviço, Mago,";
	mes "Gatuno, Arqueiro, Mercador,";
	mes "Taekwon (menino ou menina),";
	mes "Justiceiro ou Ninja^000000";
	mes "para ter direito.";
	mes "E então, alguma dúvida?";
	next;
	switch(select("^FF0000O que é o Reset?:Resetar atributos:Resetar habilidades^000000:Cancelar")) {
	case 1:
		mes "[Hipnotizadora]";
		mes "O Reset permite ao aventureiro";
		mes "redistribuir os seus pontos";
		mes "de atributo e de habilidade se não";
		mes "estiver contente com eles.";
		next;
		mes "[Hipnotizadora]";
		mes "Antes de fazer";
		mes "o Reset, você precisa";
		mes "zerar o peso";
		mes "de todos os itens que o seu";
		mes "personagem carrega. Pode guardar";
		mes "o que sobrar no Armazém Kafra.";
		next;
		mes "[Hipnotizadora]";
		mes "Ah, também é importante";
		mes "tirar o seu ^FF0000Carrinho^000000";
		mes "se estiver com um.";
		mes "Senão a hipnose não";
		mes "funciona, ou sai pela culatra...";
		close;
	case 2:
		set .@str$,"Atributos";
	case 3:
		if (.@str$ == "")
			set .@str$,"Habilidades";
		mes "[Hipnotizadora]";
		mes "Tem certeza de que";
		mes "deseja prosseguir com";
		mes "o meu serviço de ^FF0000Reset de "+.@str$+"^000000?";
		next;
		if(select("Sim:Cancelar") == 1)
			break;
	case 4:
		mes "[Hipnotizadora]";
		mes "Obrigada, e boa";
		mes "sorte nas suas aventuras.";
		mes "Viaje em segurança~";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[Hipnotizadora]";
			mes "Oh! Por favor, tire o seu";
			mes "Carrinho antes de prosseguir";
			mes "com o Reset de "+.@str$+".";
			mes "Obrigada pela compreensão~";
			close;
		}
		if (Weight != 0) {
			mes "[Hipnotizadora]";
			mes "Se veio pelo meu Reset de "+.@str$;
			mes "lembre-se, por favor,";
			mes "de que não dá para";
			mes "resetar "+.@str$+" enquanto o";
			mes "^FF0000peso dos itens do seu";
			mes "inventário não chegar a 0^000000.";
			next;
			mes "[Hipnotizadora]";
			mes "Que tal colocar as suas";
			mes "coisas no Armazém Kafra";
			mes "por enquanto? Assim você";
			mes "guarda tudo em segurança.";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[Hipnotizadora]";
			mes "Sinto muito, mas personagens";
			mes "com Nível de Base acima";
			mes "de 50 não podem usar o Reset de";
			mes .@str$+" que eu ofereço.";
			close;
		}
		mes "[Hipnotizadora]";
		mes "Obrigada por usar";
		mes "o meu serviço de Reset de "+.@str$+". Ah, e boa";
		mes "sorte nas suas";
		mes "viagens,";
		mes "aventureiro.";
		if (.@str$ == "Atributos")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[Hipnotizadora]";
		mes "Sinto muito, mas a sua";
		mes "classe não tem direito";
		mes "ao Reset de "+.@str$+" que eu";
		mes "ofereço. Eu só posso";
		mes "oferecer Reset de "+.@str$+" para as";
		mes "seguintes classes...";
		next;
		mes "[Hipnotizadora]";
		mes "^FF0000Espadachim, Noviço,";
		mes "Mago, Gatuno, Arqueiro,";
		mes "Mercador, Taekwon";
		mes "(menino ou menina),";
		mes "Justiceiro e Ninja^000000.";
		close;
	}
}
