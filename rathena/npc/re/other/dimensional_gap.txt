	next;
	switch(select("Abrir o armazém por 200 Zeny.:Não, obrigado.")) {
	case 1:
		if (Zeny > 199) {
			if(!callfunc("F_CanOpenStorage")){
				mes "[Gerente de Logística]";
				mes "Você ainda não está habilitado a usar o Armazém.";
			}
			else {
				Zeny -= 200;
				openstorage;
			}
		}
		else {
			mes "[Gerente de Logística]";
			mes "Você precisa de pelo menos 200 zeny para usar o Armazém.";
		}
		break;
	case 2:
		mes "[Gerente de Logística]";
		mes "Me avise se mudar de ideia.";
	}
	close;
}
