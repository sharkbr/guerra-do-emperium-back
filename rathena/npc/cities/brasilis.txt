-	script	::Crewman_bra2	-1,{
	mes "[Marinheiro]";
	mes "Ei, você já ouviu falar de um lugar chamado Brasilis?";
	mes "É uma cidade tropical, quente como o deserto e chuvosa ao mesmo tempo. Um lugar muito misterioso.";
	next;
	mes "[Marinheiro]";
	mes "Descobrimos faz pouco tempo uma nova rota marítima para chegar lá com facilidade.";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "São só 10.000 zeny pela ida e volta, e 1.000 para VIP! E aí, quer ir?"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "São só 10.000 zeny pela ida e volta! E aí, quer ir?";
	}
	next;
	switch(select("Me leve para Brasilis!:Vou ficar por aqui.")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[Marinheiro]";
			mes "Eu disse "+callfunc("F_InsertComma",.@cost)+" zeny.";
			close;
		}
		else {
			mes "[Marinheiro]";
			mes "Beleza~!! Vamos nessa~!";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[Marinheiro]";
		mes "Se um dia se interessar, é só me avisar que eu te levo lá.";
		close;
	}
}

brasilis,137,77,5	script	Vendedor de Sorvete	85,{
	mes "[Vendedor de Sorvete]";
	mes "Chega~chega~";
	mes "Sorvete é o lanche perfeito para um dia quente~";
	mes "São só ^3355FF100 zeny^000000~";
	mes "Sorvete~";
	mes "Olha o sorvete!";
	next;
	switch(select("Quero um!:Sorvete?:Cancelar.")) {
	case 1:
		mes "[Vendedor de Sorvete]";
		mes "Como tem muita gente querendo um sorvete geladinho, dá para pedir só 5 de cada vez.";
		mes "E aí, quantos você quer?";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[Vendedor de Sorvete]";
				mes "Nenhum?";
				mes "Então sai da frente, que eu tenho fregueses para atender.";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[Vendedor de Sorvete]";
				mes "Uau.";
				mes "Você pediu demais.";
				mes "Se comer mais de 5 vai acabar lutando com um monstro dentro da barriga. Calma aí, amigo.";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[Vendedor de Sorvete]";
			mes "Poxa~! Seu dinheiro não dá.";
			mes "São só ^3355FF100 zeny^000000~ Fala sério!";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[Vendedor de Sorvete]";
			mes "Parece que você está carregando coisa demais.";
			mes "Alivie a mochila antes de comprar.";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[Vendedor de Sorvete]";
		mes "'Sorvete é...";
		mes "Espera, você não sabe";
		mes "o que é sorvete?";
		mes "Debaixo de que pedra";
		mes "você estava vivendo?";
		next;
		mes "[Vendedor de Sorvete]";
		mes "Nem vou começar a falar de como isso soa estranho.";
		mes "Enfim, leve o seu sorvete aqui mesmo, enquanto está bem geladinho.";
		close;
	case 3:
		mes "[Vendedor de Sorvete]";
		mes "Não perca a chance de comer o melhor sorvete de todas as terras~!";
		close;
	}
}
