	mes "Atenção, atenção!";
	mes "Venha velejar ao vento";
	mes "num navio fascinante!";

	case 1:
		if (Zeny < 150) {
			mes "[Marinheiro]";
			mes "150 Zeny!";
			mes "Só 150 Zeny para embarcar!";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;

		if (Zeny < 500) {
			mes "[Marinheiro]";
			mes "500 Zeny!";
			mes "Só 500 Zeny para embarcar!";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
