-	script	::Airship_Staff_izlude	-1,{
	mes "[Assistente do Aeroplano]";
	mes "Bem-vindo! Deseja embarcar";
	mes "no Aeroplano que faz o voo";
	mes "com escalas";
	mes "em Juno e Rachel?";
	next;
	if (select("Embarcar no Aeroplano para Juno/Rachel.:Cancelar.") == 1) {
		mes "[Assistente do Aeroplano]";
		mes "A tarifa de embarque é de";
		mes "1.200 zeny. Mas a tarifa";
		mes "é dispensada se você usar";
		mes "um Bilhete para Aeroplano. E então,";
		mes "ainda deseja embarcar?";
		next;
		if (select("Sim:Não") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[Assistente do Aeroplano]";
			mes "Sinto muito, mas você não";
			mes "tem os 1.200 zeny";
			mes "da tarifa de embarque.";
			close;
		}
	}
	mes "[Assistente do Aeroplano]";
	mes "Obrigado e";
	mes "volte sempre.";
	mes "Tenha um bom dia~";
	close;
}
