PurchaseItem:

	.@item_id = getarg(0);
	.@req_setting = getarg(1);
	.@item_type = getarg(2);

	mes "[Logística]";

	if (!.@item_type)
		mes "Você selecionou ^ff0000"+ getitemname(.@item_id) +"^000000.";
	else if (.@item_type == 1)
		mes "Há apenas um "+ callfunc("F_GetWeaponType",.@item_id) +": "+ getitemname(.@item_id) +".";
	else if (.@item_type == 2) {
		mes "Você selecionou '"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "A armadura para quem passa do nível 80 é: ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 3) {
		mes "You have selected 'Mass-Production "+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "Há apenas um de Produção em Massa "+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 4) {
		mes "You have selected 'Popularized "+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "Há apenas um Popular "+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}

	if (!.@req_setting)
		mes "Custa ^0000ff2.000^000000 Pontos KVM, e "+ .@name$ +", seus pontos são: ^580080"+ kvm_point +"^000000.";
	else if (.@req_setting == 4)
		mes "Custa ^0000ff1.200^000000 Pontos KVM, e "+ .@name$ +", seus pontos são: ^580080"+ kvm_point +"^000000.";
	else
		mes "Custa ^0000ff"+ .@prices[.@req_setting] +"^000000 Pontos KVM, e "+ .@name$ +", seus pontos são: ^580080"+ kvm_point +"^000000.";

	mes "Tem certeza de que quer este item?";
	next;
	switch(select("Não, não vou comprar.:Sim, vou comprar.")) {
	case 1:
		mes "[Logística]";
		mes "Você selecionou 'Não vou comprar'.";
		mes "Ao comprar um item, tenha cuidado: não há reembolso.";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[Logística]";
			mes "Você comprou um "+ getitemname(.@item_id) +".";

			if (!.@req_setting)
				mes "Você precisa de ^0000ff2.000^000000 Pontos KVM para comprar este item.";
