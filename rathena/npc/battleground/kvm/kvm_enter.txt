// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	KVM Mercenary Officer#1	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[Recrutador do Exército Croix]";
		mes "Sei que você já se alistou com os Croix. Volte e junte-se a eles. Não aceitamos traidores lutando por nós.";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[Recrutador do Exército Guillaume]";
		mes "Parece que você participou de uma batalha KVM há pouco. É preciso esperar 5 minutos para se alistar de novo.";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[Recrutador do Exército Guillaume]";
	mes "Mostre a eles a verdadeira força dos Guillaume!";
	next;
	switch(select("Vou lutar com você.:Encerrar conversa.")) {
	case 1:
		mes "[Recrutador do Exército Guillaume]";
		mes "Mostre a eles o quanto somos fortes.";
		mes "Hoje todos ouvirão o grito de triunfo dos Guillaume!";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[Recrutador do Exército Guillaume]";
		mes "Nós venceremos!";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	Guillaume Knight#kvm	417,{
	mes "[Cavaleiro Guillaume]";
	mes "Olá.";
	mes "O que você quer saber?";
	next;
	switch(select("Alistar-se no KVM.:O que é o KVM?:Como participo do KVM?:Quero saber meus Pontos Kreiger.")) {
	case 1:
		mes "[Cavaleiro Guillaume]";
		if (getbattleflag("feature.bgqueue"))
			mes "Se está pronto para a batalha, clique no ^0000FFbotão de batalha da interface^000000 para se alistar no KVM!";
		else {
			mes "As inscrições ainda não estão abertas.";
			mes "Para se alistar, procure um Oficial Mercenário do KVM.";
		}
		close;
	case 2:
		mes "[Cavaleiro Guillaume]";
		mes "KVM é a abreviação de Kreiger Von Midgard.";
		mes "Aventureiro, você sabia que o caminho para o novo mundo foi aberto?";
		next;
		mes "[Cavaleiro Guillaume]";
		mes "Nós, da Administração Guillaume, temos vários planos para avançar ao novo mundo.";
		mes "E o KVM é um deles.";
		next;
		mes "[Cavaleiro Guillaume]";
		mes "Significa que nós, um entre muitos países do continente de Midgard,";
		mes "decidimos empregar muitos aventureiros para o avanço imediato ao novo mundo.";
		next;
		mes "[Cavaleiro Guillaume]";
		mes "Por isso, para selecionar os melhores aventureiros, realizamos o KVM.";
		mes "Esperamos resposta de muitos aventureiros.";
		next;
		mes "[Cavaleiro Guillaume]";
		mes "E daremos recompensas pela participação!";
		mes "Quanto às recompensas, procure um Oficial de Logística do KVM.";
		close;
	case 3:
		mes "[Cavaleiro Guillaume]";
		if (getbattleflag("feature.bgqueue")) {
			mes "Basicamente, o KVM é uma batalha 3 contra 3.";
			mes "Aliste-se clicando no ^0000FFbotão de batalha da interface^000000, em grupo ou individualmente.";
		} else {
			mes "Basicamente, o KVM é uma batalha 5 contra 5.";
			mes "Primeiro você se alista com um recepcionista do KVM, em grupo ou individualmente.";
		}
		next;
		mes "[Cavaleiro Guillaume]";
		if (getbattleflag("feature.bgqueue"))
			mes "A inscrição em grupo é para quando você pretende entrar no KVM com seu grupo ou clã,";
		else
			mes "A inscrição em grupo é para quando você pretende entrar no KVM com seu grupo,";
		mes "e a individual é para quando você pretende entrar sozinho.";
		next;
		mes "[Cavaleiro Guillaume]";
		if (getbattleflag("feature.bgqueue"))
			mes "Atenção: ao cancelar, você ^0000FFnão poderá se alistar de novo por 1 minuto^000000.";
		else {
			mes "Aliste-se com um oficial do KVM; ele o contatará depois, quando você estiver em Prontera.";
			next;
			mes "[Cavaleiro Guillaume]";
			mes "Depois você entra e segue as instruções no campo de batalha.";
			next;
			mes "[Cavaleiro Guillaume]";
			mes "Porém, saiba que fora do escritório do KVM ele não consegue contatá-lo.";
		}
		close;
	case 4:
		mes "[Cavaleiro Guillaume]";
		mes "Seus Pontos Kreiger são:";
		mes ""+kvm_point+".";
		close;
	}
}
