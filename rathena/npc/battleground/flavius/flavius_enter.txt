//============================================================
bat_room,133,178,5	script	Flavius Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- Espera aí!! -";
		mes "- Você está carregando -";
		mes "- itens demais. -";
		mes "- Tente de novo -";
		mes "- depois de aliviar o peso. -";
		close;
	}
	mes "[Oficial do Exército Guillaume]";
	mes "Vamos mostrar o poder do Exército Guillaume a esses Croix fedorentos!";
	next;
	switch(select("Quero entrar no seu exército!:Encerrar conversa")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[Oficial do Exército Guillaume]";
			mes "Fico feliz que queira lutar conosco contra o Exército Croix, mas sinto muito: não posso mandar um novato como você morrer num campo de batalha cruel.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[Oficial do Exército Guillaume]";
				mes "Fico feliz que queira lutar conosco contra o Exército Croix, mas sinto muito: não posso mandar um novato como você morrer num campo de batalha cruel.";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[Oficial do Exército Guillaume]";
					mes "Parece que você acabou de voltar do campo de batalha.";
					mes "É cedo demais para você voltar. Vá descansar e deixe os inimigos conosco!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[Oficial do Exército Guillaume]";
						mes "Recebi um relatório informando que um corpo de elite já foi enviado ao campo de batalha contra os Croix.";
						mes "Que tal aguardar um pouco? Sugiro afiar as armas e preparar os suprimentos até lá.";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[Oficial do Exército Guillaume]";
							mes "Um corpo de elite já está a postos para ser enviado ao campo de batalha.";
							mes "Que tal aguardar um pouco? Sugiro afiar as armas e preparar os suprimentos até lá.";
						}
						else {
							mes "[Oficial do Exército Guillaume]";
							mes "Você parece mesmo pronto para a batalha!";
							mes "Mostre aos Croix o que é medo de verdade!";
							mes "Hoje nosso grito de vitória vai ecoar por todo o campo de batalha!";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[Oficial do Exército Guillaume]";
		mes "Hoje seremos vitoriosos!";
		break;
	}
	close;
}

bat_room,135,178,5	script	Guillaume Knight#3	417,{

	next;
	mes "[Cavaleiro Guillaume]";
	mes "Os cristais dos dois exércitos são protegidos por barricadas especiais que não podem ser destruídas por ataque direto.";
	next;

	next;
	mes "[Cavaleiro Guillaume]";
	mes "Os cristais são imunes a todo tipo de habilidade; o ataque físico é a única forma de destruir o cristal inimigo.";
	next;
	mes "[Cavaleiro Guillaume]";
	if (getbattleflag("feature.bgqueue"))
		mes "Se está pronto, comece clicando no ^0000FFbotão de batalha da interface^000000.";
	else {
		mes "Lembre-se: ao entrar numa batalha você recebe uma marca que indica por quanto tempo não poderá participar do mesmo tipo de batalha.";
		mes "Você pode conferir o tempo apertando Alt+U.";
	}
	next;
	mes "[Cavaleiro Guillaume]";
	if (getbattleflag("feature.bgqueue"))
		mes "Tenha em mente que, ao cancelar, ^0000FFnão poderá se alistar de novo por 1 minuto^000000!";
	else
		mes "Pronto para a batalha? Então aliste-se com o recrutador ao meu lado!";
	close;
}
