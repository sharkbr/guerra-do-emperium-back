	mes "Você escolheu ^3131FF"+getitemname(.@Items[.@i])+"^000000.";
	switch(.@Items[.@i]) {
		case 2720: mes "Este item é exclusivo das classes Espadachim e Mestre Taekwon."; break;
		case 2721: mes "Este item é exclusivo da classe Gatuno."; break;
		case 2722: mes "Este item é exclusivo da classe Noviço."; break;
		case 2723: mes "Este item é exclusivo da classe Mago."; break;
		case 2724: mes "Este item é exclusivo da classe Arqueiro."; break;
		case 2725: mes "Este item é exclusivo da classe Mercador."; break;
		case 2733: mes "Este item é exclusivo do Justiceiro."; break;
		default: break;
	}
	mes "Você pode trocar por este item com ^FF0000"+.@Items[.@i+1]+" "+getitemname(7828)+" ou "+.@Items[.@i+1]+" "+getitemname(7829)+"^000000.";
	mes "Deseja trocar?";
	next;
	switch(select("Não trocar:Trocar")) {
	case 1:
		mes "[Erundek]";
		mes "Precisa de mais tempo para conferir os itens?";
		break;
	case 2:
		mes "[Erundek]";
		mes "Qual Emblema você quer trocar?";
		mes "Você precisa de ^3131FF"+.@Items[.@i+1]+" Emblemas^000000 para trocar.";
		next;
		if (.@item[0] < 12269 || .@item[0] > 12273 ) {
			mes "[Erundek]";
			mes "Lembre-se: os itens de recompensa do campo de batalha são ^FF0000vinculados ao personagem^000000. Tem certeza de que quer este item?";
			next;
		}
		set .@j, select("Emblema da Bravura:Emblema do Valor:Cancelar");
		mes "[Erundek]";
		if (.@j == 3) {
			mes "Você cancelou a troca.";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "Obrigado pela troca.";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "Você não tem "+getitemname(.@cost)+"s.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Bendito Guillaume!";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Bendito Croix!";
		}
		mes "Vamos celebrar nossa gloriosa vitória!";
		mes strcharinfo(0)+", it's a sign reflecting victory.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Você perdeu, mas se dedicou a esta batalha.";
			mes "Esta é a recompensa de Guillaume Marollo pela sua dedicação!";
			mes "Encare esta derrota como lição; da próxima você certamente vencerá.";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Ah, "+strcharinfo(0)+" Don't be sad.";
			mes "Mesmo sem vencer, demos o nosso melhor.";
			mes "Este é um presente real de Croix; não esqueça esta batalha. Venceremos a próxima.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
