-	script	Itens para Assistentes#dummy::MercMerchant	-1,{
	mes "[Itens para Assistentes]";
	mes "Olá, eu vendo itens";
	mes "que os Assistentes podem";
	mes "usar. Procura alguma";
	mes "coisa em";
	mes "especial?";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,".",.@len-3);
	mes "[Itens para Assistentes]";
	mes getitemname(.@item[.@m]);
	mes "custa, cada um, " + .@str_cost$ + " zeny.";
	mes "Quantos você quer?";
	next;
	input .@input;
	if (!.@input) {
		mes "[Itens para Assistentes]";
		mes "Mudou de ideia?";
		mes "Tudo bem, pode voltar";
		mes "quando quiser";
		mes "comprar Poções para Assistentes.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[Itens para Assistentes]";
		mes "Sinto muito, mas você";
		mes "só pode comprar até";
		mes "10.000 poções";
		mes "de uma vez. Digite";
		mes "um número de 1 a 10.000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[Itens para Assistentes]";
		mes "Sinto muito, mas você não";
		mes "tem zeny para tantas";
		mes "poções. Bem, eu vou";
		mes "estar aqui quando você";
		mes "quiser comprar alguma coisa";
		mes "para os seus Assistentes.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[Itens para Assistentes]";
		mes "Se eu te desse tantas";
		mes "poções, você não";
		mes "conseguiria carregá-las.";
		mes "Volte depois de";
		mes "liberar espaço";
		mes "no seu inventário.";
		close;
	}
	mes "[Itens para Assistentes]";
	if (.@input == 1) 
		mes "Aqui está: 1x " + getitemname(.@item[.@m]) + ".";
	else {
		mes "Aqui está a sua encomenda:";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "Obrigado, e volte";
	mes "sempre que precisar de mais";
	mes "poções para os seus Assistentes.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Lanceiro","Espadachim","Arqueiro";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[Verificador]";
	mes "Digite a senha.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[Verificador]";
	if (.@i == -2)
		mes "Erro.";
	else if (.@i == 0)
		mes "Número errado";
	else if (.@i == 1) {
		mes "Escolha.";
		next;
		switch( select( "Desligar os NPCs de Assistente","Ligar os NPCs de Assistente" ) ) {
		case 1:
			mes "Os NPCs foram desligados.";
			disablenpc "Gerente de Assistentes#" + .@name$[.@type];
			disablenpc "Itens para Assistentes#" + .@name$[.@type];
			break;
		case 2:
			mes "Os NPCs foram ligados.";
			enablenpc "Gerente de Assistentes#" + .@name$[.@type];
			enablenpc "Itens para Assistentes#" + .@name$[.@type];
			break;
		}
	}
	close;
}
